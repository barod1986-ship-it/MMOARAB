-	script	::PR_Officer_moscovia	-1,{
	mes "[مندوب موسكوفيا]";
	mes "موسكوفيا! هل سمعت بها من قبل؟";
	mes "إنها جنة تمتد وسط البحار الشاسعة...";
	mes "مرحبا بك في موسكوفيا،";
	mes "موطن المغامرات والأسرار.";
	next;
	switch(select("معلومات عن موسكوفيا:السفر إلى موسكوفيا:إلغاء")) {
		case 1:
			mes "[مندوب موسكوفيا]";
			mes "موسكوفيا مملكة جميلة";
			mes "تقع على جزيرة إلى الشمال من";
			mes "مملكة رون-ميدغارتس.";
			next;
			mes "[مندوب موسكوفيا]";
			mes "أنا واثق أنك ستفتن";
			mes "بمناظر موسكوفيا الساحرة";
			mes "وقصرها البديع،";
			mes "كأنك دخلت حكاية خيالية.";
			next;
			mes "[مندوب موسكوفيا]";
			mes "لقد انقضى شتاؤنا الطويل،";
			mes "وحل الربيع الدافئ،";
			mes "ويسعدني الآن أن أريك";
			mes "روعة موطننا.";
			next;
			mes "[مندوب موسكوفيا]";
			mes "إن كنت تفكر في زيارة موسكوفيا،";
			mes "فهذه فرصتك!";
			mes "سأتولى إرشادك لتستمتع";
			mes "برحلة رائعة إليها.";
			close;

		case 2:
			mes "[مندوب موسكوفيا]";
			mes "حسنا، لنبدأ.";
			mes "تبلغ أجرة الرحلة إلى موسكوفيا";
			mes "10,000 زيني.";
			mes "أما رحلة العودة،";
			mes "فهي مجانية.";
			next;
			mes "[مندوب موسكوفيا]";
			mes "هل أنت مستعد للمغادرة؟";
			next;
			if (select("لننطلق!:إلغاء") == 2) {
				mes "[مندوب موسكوفيا]";
				mes "لا بأس إن كنت مشغولا الآن.";
				mes "عد إلي متى شئت،";
				mes "فأنا مستعد دائما لإرشاد المسافرين إلى موسكوفيا.";
				close;
			}
			if (Zeny < 10000) {
				mes "[مندوب موسكوفيا]";
				mes "عذرا، لا تملك";
				mes "ما يكفي من الزيني.";
				mes "تحتاج إلى 10,000 زيني";
				mes "للسفر إلى موسكوفيا.";
				mes "عد إلي عندما يتوفر المبلغ.";
				close;
			} else {
				mes "[مندوب موسكوفيا]";
				mes "حسنا، سننطلق حالا.";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[مندوب موسكوفيا]";
			mes "لا بأس إن كنت مشغولا الآن.";
			mes "عد إلي متى شئت،";
			mes "فأنا مستعد دائما لإرشاد المسافرين إلى موسكوفيا.";
			close;
	}
}
