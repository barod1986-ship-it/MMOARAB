-	script	بحار::Crewman_bra2	-1,{
	mes "[البحار]";
	mes "مهلا، هل سمعت عن مدينة تدعى برازيليس؟";
	mes "إنها مدينة استوائية حارة كالصحراء، لكنها كثيرة الأمطار أيضا. إنها مدينة غامضة حقا.";
	next;
	mes "[البحار]";
	mes "اكتشفنا مؤخرا طريقا بحريا جديدا يجعل الوصول إليها سهلا.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "تكلفة الرحلة ذهابا وإيابا 10,000 زيني فقط، أو 1,000 زيني للأعضاء المميزين! هل ترغب في السفر؟"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "تكلفة الرحلة ذهابا وإيابا 10,000 زيني فقط! هل ترغب في السفر؟";
	}
	next;
	switch(select("السفر إلى برازيليس!:سأبقى هنا.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[البحار]";
			mes "قلت إن الأجرة "+callfunc("F_InsertComma",.@cost)+" زيني.";
			close;
		}
		else {
			mes "[البحار]";
			mes "رائع~! لننطلق~!";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[البحار]";
		mes "حسنا. إن غيرت رأيك فأخبرني، وسأوصلك إلى هناك.";
		close;
	}
}

brasilis,137,77,5	script	صانع مثلجات::Ice-Cream Maker	85,{
	mes "[صانع مثلجات]";
	mes "اقتربوا~ اقتربوا~";
	mes "المثلجات هي الوجبة الخفيفة المثالية في يوم حار~";
	mes "سعر الواحدة ^3355FF100 زيني^000000 فقط~";
	mes "مثلجات~";
	mes "مثلجات باردة ولذيذة! تفضلوا!";
	next;
	switch(select("أعطني واحدة!:ما هي المثلجات؟:إلغاء.")) {
	case 1:
		mes "[صانع مثلجات]";
		mes "لكثرة الراغبين في المثلجات الباردة، لا يمكنك طلب أكثر من 5 في كل مرة.";
		mes "كم واحدة تريد؟";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[صانع مثلجات]";
				mes "لا تريد شيئا؟";
				mes "حسنا، ابتعد قليلا، فلدي زبائن آخرون.";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[صانع مثلجات]";
				mes "ماذا؟";
				mes "طلبت عددا كبيرا جدا.";
				mes "إن أكلت أكثر من خمس، فقد تضطر إلى قتال وحش داخل معدتك. تمهل يا صاح.";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[صانع مثلجات]";
			mes "يا صاح~! لا تملك ما يكفي من الزيني.";
			mes "سعر الواحدة ^3355FF100 زيني^000000 فقط~ بجدية!";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[صانع مثلجات]";
			mes "يبدو أنك تحمل أشياء كثيرة.";
			mes "خفف حمولة حقيبتك قبل أن تشتري.";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[صانع مثلجات]";
		mes "المثلجات هي...";
		mes "لحظة، ألا تعرف";
		mes "ما هي المثلجات؟";
		mes "تحت أي صخرة";
		mes "كنت تعيش؟";
		next;
		mes "[صانع مثلجات]";
		mes "لن أبدأ حتى بشرح مدى غرابة ما قلته.";
		mes "على أي حال، اشتر مثلجاتك من هنا وهي لا تزال باردة ولذيذة.";
		close;
	case 3:
		mes "[صانع مثلجات]";
		mes "لا تفوت فرصة تذوق ألذ مثلجات في البلاد كلها~!";
		close;
	}
}
