				end;

			close;

		mes "[أيباكثينغ]";
		mes "هل حلمت يوما";
		mes "بأرض جميلة يعمها";
		mes "الصفاء الروحي؟ لا بد أنك";
		mes "كنت تحلم";
		mes "بأيوثايا، يا صديقي.";
