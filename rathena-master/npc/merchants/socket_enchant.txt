function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[سايابليم]";
	mes "إضافة فتحة إلى "+ getitemname(.@item_id) +"، صحيح؟";
	mes "تحتاج إلى ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "":"") +"^000000، "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "":"") + "^000000، ") : "") + "بالإضافة إلى رسم خدمة قدره ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 زيني^000000.";
	mes "ولا تنس إحضار "+ getitemname(.@item_id) +" نفسه.";
	next;
	mes "[سايابليم]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "إذا كانت العناصر المطلوبة والزيني متوافرة، يمكنني بدء المحاولة الآن.";
		mes "لكن النجاح غير مضمون.";
		mes "لذلك يجب توضيح المخاطر أولا...";
		next;
		mes "[سايابليم]";
		mes "إذا فشلت المحاولة، فسيدمر ^FF0000السلاح^000000، وستدمر أيضا ^FF0000كل البطاقات المركبة فيه^000000.";
	}
	else // armor
	{
		mes "إذا كانت المواد المطلوبة والدرع ورسم الخدمة متوافرة، فيمكننا بدء محاولة إضافة الفتحة.";
		mes "لكن يجب توضيح المخاطر أولا.";
		next;
		mes "[سايابليم]";
		mes "إذا فشلت المحاولة، فسيدمر ^FF0000الدرع^000000 مع ^FF0000مستوى ترقيته^000000 و^FF0000كل البطاقات المركبة فيه^000000.";
	}
	mes "هل تريد متابعة محاولة إضافة الفتحة؟";
	next;
	switch(select("بدء محاولة إضافة الفتحة:إلغاء"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[سايابليم]";
				mes "حسنا، لنبدأ العمل!";
				mes "لعل الحظ يحالفك في هذه المحاولة.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[سايابليم]";
					mes "نجحت العملية!";
					mes "أضيفت الفتحة بإتقان. تهانينا!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[سايابليم]";
					mes "إلى اللقاء، وبالتوفيق.";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[سايابليم]";
					mes "آه... يؤسفني أن المحاولة فشلت.";
					mes "لقد نفذت العملية بالطريقة الصحيحة، لكن النجاح يعتمد على الحظ.";
					mes "هذه هي نتيجة المحاولة هذه.";
					mes "لا تيأس؛ قد تنجح المحاولة المقبلة.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[سايابليم]";
					mes "أتمنى لك حظا أفضل في المرة المقبلة.";
					close;
				}
			}
			else
			{
				mes "[سايابليم]";
				mes "لا يمكنني بدء محاولة إضافة الفتحة لأن بعض المتطلبات ناقصة.";
				mes "تحقق من وجود القطعة والمواد المطلوبة والزيني.";
				close;
			}
		case 2:
			mes "[سايابليم]";
			mes "لا بأس، خذ وقتك في التفكير.";
			mes "يمكنك العودة عندما تكون مستعدا.";
			mes "لكن تذكر أن المخاطرة جزء من متعة المغامرة.";
			close;
	}
}
