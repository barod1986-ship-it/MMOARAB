	if (checkweight(1201,1) == 0) {
		mes "- انتظر قليلا! -";
		mes "- تحمل الآن -";
		mes "- عددا كبيرا من الأغراض. -";
		mes "- عد إلي لاحقا -";
		mes "- بعد إيداع بعضها في مخزن كافرا. -";
		close;
	}
	mes "[الصيدلي العجوز]";
	mes "هممم...";
	mes "ما الذي أتى بك إلى هنا...؟";
	next;
	switch(select("صنع جرعة:التحدث معه:معلومات صناعة الجرعات:إلغاء")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "[الصيدلي العجوز]";
			mes "لماذا تحمل كل هذه الأغراض؟!";
			mes "لا تحمل إلا ما تحتاج إليه!";
			next;
			mes "[الصيدلي العجوز]";
			mes "لا توجد مساحة كافية لتحمل الجرعات...";
			mes "ضع بعض أغراضك في المخزن أولا!";
			close;
		} else {
			mes "[الصيدلي العجوز]";
			mes "هل أحضرت جميع المواد؟ اختر الجرعة التي تريد صنعها.";
			next;
			switch(select("جرعة حمراء:جرعة برتقالية:جرعة صفراء:جرعة بيضاء:جرعة زرقاء:جرعة خضراء:إلغاء")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "[الصيدلي العجوز]";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "أيها المشاغب! جئت من دون المواد المطلوبة؟!";
					mes "اغرب عن وجهي!";
					close;
				}
				if (Zeny < 3) {
					mes "أيها المشاغب! جئت من دون المواد المطلوبة؟!";
					mes "اغرب عن وجهي!";
					close;
				}
				mes "كم جرعة تريد؟";
				next;
				switch(select("صنع أكبر عدد ممكن:تحديد الكمية:إلغاء")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "[الصيدلي العجوز]";
						mes "أيها المشاغب! لا تملك المواد أو الزيني الكافي لصنع الجرعات!";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "[الصيدلي العجوز]";
					mes "أدخل عددا من 1 إلى 100، أو أدخل '0' للإلغاء. بموادك الحالية يمكن صنع نحو "+.@max+" جرعة.";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "[الصيدلي العجوز]";
						mes "غيرت رأيك إذن؟!";
						close;
					}
					if (.@amount > 100) {
						mes "[الصيدلي العجوز]";
						mes "قلت إن الحد الأقصى 100 جرعة!";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "[الصيدلي العجوز]";
						mes "أيها المشاغب! لا تملك المواد أو الزيني الكافي لصنع الجرعات!";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "[الصيدلي العجوز]";
					mes "ماذا؟!";
					mes "غررر...";
					mes "تبا!";
					mes "اغرب عن وجهي!";
					close;
				}
				mes "[الصيدلي العجوز]";
				mes "انتهيت. خذ الجرعات، وتذكر أن الإفراط في تناول الدواء ليس جيدا.";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "[الصيدلي العجوز]";
				mes "ماذا؟!";
				mes "غررر...";
				mes "تبا!";
				mes "اغرب عن وجهي!";
				close;
			}
		}
	case 2:
		mes "[الصيدلي العجوز]";
		mes "يمكن للأدوية أن تعزز قدرة الجسد على التعافي، لكن أثرها يبقى محدودا. *تنهد* أتذكر أيامي حين كنت شابا... لا بد أنني أهرم.";
		next;
		mes "[الصيدلي العجوز]";
		mes "في النهاية، الجرعة دواء فحسب؛ لا أكثر ولا أقل.";
		close;
	case 3:
		mes "[الصيدلي العجوز]";
		mes "هممم...";
		mes "أنتم الشبان كثيرو الأسئلة... لكن بما أنك سألت، فسأشرح لك.";
		next;
		mes "[الصيدلي العجوز]";
		mes "للأعشاب أثر مفيد بمفردها، لكن تحويلها إلى جرعات بتقنياتي الخاصة يجعل أثرها أقوى بكثير.";
		next;
		mes "[الصيدلي العجوز]";
		mes "إن طلبت بأدب فسأصنع الجرعات لك، لكن ليس مجانا... لا تقلق، فرسومي بسيطة.";
		next;
		mes "[الصيدلي العجوز]";
		mes "جرعة حمراء - ^0098E5عشبتان حمراوان، زجاجة فارغة، ورسم 2 زيني.^000000";
		mes "جرعة برتقالية - ^0098E5عشبة حمراء، وعشبة صفراء، وزجاجة فارغة، ورسم 5 زيني.^000000";
		mes "جرعة صفراء - ^0098E5عشبتان صفراوان، وزجاجة فارغة، ورسم 10 زيني.^000000";
		next;
		mes "[الصيدلي العجوز]";
		mes "جرعة بيضاء - ^0098E5عشبتان بيضاوان، وزجاجة فارغة، ورسم 20 زيني.^000000";
		mes "جرعة زرقاء - ^0098E5عشبتان زرقاوان، وزجاجة فارغة، ورسم 30 زيني.^000000";
		mes "جرعة خضراء - ^0098E5عشبتان خضراوان، وزجاجة فارغة، ورسم 3 زيني.^000000";
		close;
	case 4:
		mes "[الصيدلي العجوز]";
		mes "يا لك من شخص ممل... إن كان لديك ما تقوله فقله!";
		close;
	}
	
L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "[الصيدلي العجوز]";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "أيها المشاغب! جئت من دون المواد المطلوبة؟!";
		mes "اغرب عن وجهي!";
		close;
	}
	if (Zeny < 3) {
		mes "أيها المشاغب! جئت من دون المواد المطلوبة؟!";
		mes "اغرب عن وجهي!";
		close;
	}
	mes "كم جرعة تريد؟";
	next;
	switch(select("صنع أكبر عدد ممكن:تحديد الكمية:إلغاء")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "[الصيدلي العجوز]";
			mes "أيها المشاغب! لا تملك المواد أو الزيني الكافي لصنع الجرعات!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max; 
		break;
	case 2:
		mes "[الصيدلي العجوز]";
		mes "أدخل عددا من 1 إلى 100، أو أدخل '0' للإلغاء. بموادك الحالية يمكن صنع نحو "+.@max+" جرعة.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[الصيدلي العجوز]";
			mes "غيرت رأيك إذن؟!";
			close;
		}
		if (.@amount > 100) {
			mes "[الصيدلي العجوز]";
			mes "قلت إن الحد الأقصى 100 جرعة!";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "[الصيدلي العجوز]";
			mes "أيها المشاغب! لا تملك المواد أو الزيني الكافي لصنع الجرعات!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount; 
		break;
	case 3:
		mes "[الصيدلي العجوز]";
		mes "ماذا؟!";
		mes "غررر...";
		mes "تبا!";
		mes "اغرب عن وجهي!";
		close;
	}
	mes "[الصيدلي العجوز]";
	mes "انتهيت. خذ الجرعات، وتذكر أن الإفراط في تناول الدواء ليس جيدا.";
	close;
}
