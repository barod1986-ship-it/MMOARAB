payon,157,146,6	script	سونبي#cash::Suhnbi#cash	85,{
	disable_items;
	mes "[سونبي]";
	mes "أنا حداد أسلحة متخصص.";
	mes "يمكنني تطوير جميع أنواع الأسلحة،";
	mes "والدروع والمعدات، لذا أخبرني";
	mes "بما تريد مني تطويره.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i])) {
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@equipped,1;
		}
		set .@menu$, .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "[سونبي]";
		mes "لا أظن أنني أستطيع تطوير أي من معداتك...";
		close;
	}
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) //custom check
		close;
	if (!getequipisenableref(.@part)) {
		mes "[سونبي]";
		mes "ابحث عن حداد آخر؛ فهذا العنصر غير قابل للتطوير.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[سونبي]";
		mes "همم... لقد بلغ هذا العنصر أقصى مستوى تطوير. لا أستطيع العمل عليه أكثر.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@price = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	// Make sure you have the necessary items and Zeny to refine your items
	// Determines chance of failure and verifies that you want to continue.
	callsub S_RefineValidate,.@itemtype,.@material,.@price,.@part,.@refineitemid,.@refinerycnt;

	mes "[سونبي]";
	mes "طنغ! طنغ! طنغ!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[سونبي]";
		mes "تفضل! انتهيت.";
		mes "مضى وقت طويل منذ أن طورت "+((.@itemtype == IT_WEAPON)?"سلاحا":"درعا")+" بهذا الإتقان. ها قد أصبح أقوى!";
		close;
	}
	failedrefitem .@part;
	next;
	emotion (!rand(5))?ET_MONEY:ET_HUK;
	mes "[سونبي]";
	mes "أووووووووووووف!!!";
	next;
	mes "[سونبي]";
	mes "...";
	mes ".....";
	mes ".......هوهوهوهوهو~";
	mes "........كان القرار قرارك، وهذه حدود قدرتي؛ لا مجال للندم.";
	close;

S_RefineValidate:
	.@itemtype = getarg(0);
	.@item_req = getarg(1);
	.@price = getarg(2);
	.@part = getarg(3);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch(.@equip_lv) {
				case 1:
					set .@price, .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch( .@equip_lv ){
				case 1:
					set .@price, .@price * 40;
					break;
				case 2:
					set .@price, .@price * 50;
					break;
				case 3:
					set .@price, .@price * 2;
					break;
				case 4:
					set .@price, .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		// TODO:
		close;
	}

	mes "[سونبي]";
	if (.@itemtype == IT_WEAPON)
		mes "أتريد تطوير سلاح من المستوى "+ .@equip_lv +"؟";
	mes "يتطلب تطويره قطعة من ^ff9999"+ getitemname(.@item_req) +"^000000 و "+ .@price +" زيني.";
	mes "هل تريد المتابعة؟";
	next;
	if(select("نعم:لا") == 1) {
		if (getequippercentrefinery(.@part) < 100) {
			if (.@itemtype == IT_WEAPON) {
				mes "[سونبي]";
				mes "يا للعجب!";
				mes "يبدو أن هذا السلاح";
				mes "خضع للتطوير بالفعل...";
				mes "مرات عديدة...";
				mes "وقد ينكسر إذا";
				mes "حاولت تطويره مجددا.";
				next;
				mes "وإذا انكسر،";
				mes "فلن تتمكن من استخدامه مجددا!";
				mes "وستفقد جميع البطاقات المركبة فيه وخصائصه ^ff0000نهائيا^000000!";
				mes "^ff0000وسيتحطم السلاح نفسه أيضا!^000000";
				mes "هل ما زلت تريد المتابعة؟";
				next;
				if(select("نعم:لا") == 2) {
					mes "[سونبي]";
					mes "جيد.";
					mes "فإذا انكسر السلاح نتيجة مخاطرة غير محسوبة، سيتعكر مزاجي أيضا.";
					close;
				}
			} else {
				mes "[سونبي]";
				mes "هيهي... لديك جرأة كبيرة على محاولة تطوير هذه المعدات.";
				mes "تعرف أن هذه المحاولة محفوفة بالمخاطر، أليس كذلك؟";
				next;
				mes "إذا تحطمت معداتك الدفاعية، فلن تتمكن من استخدامها مجددا.";
				mes "وستختفي البطاقات المركبة فيها وخصائصها ^ff0000نهائيا^000000.";
				//mes "Everything will disappear. As in... GONE!";
				mes "هل ما زلت تريد المتابعة؟";
				next;
				if(select("نعم:لا") == 2) {
					mes "[سونبي]";
					mes "يا له من تردد... لقد أضعت وقتي الثمين.";
					mes "ارحل من هنا.";
					close;
				}
			}
		}
		if (countitem(.@item_req) > 0 && Zeny > .@price) {
			delitem .@item_req,1;
			set Zeny, Zeny - .@price;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@part, getarg(4)) ||
				callfunc("F_IsEquipRefineHack", .@part, getarg(5)) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
				mes "[سونبي]";
				emotion ET_FRET;
				mes "مهلا...";
				mes "أتظن أنني غبي؟!";
				mes "بدلت العنصر حين لم أكن أنظر! اخرج من هنا!";
				close;
			}

			return;
		}
		mes "[سونبي]";
		mes "أهذا كل ما لديك؟";
		mes "آسف، لكنني لا أستطيع البدء من دون جميع المواد، كما أن عملي يستحق أجرا، أليس كذلك؟";
		close;
	}
	mes "[سونبي]";
	mes "لا حيلة لي إن لم ترغب في المتابعة...";
	close;
}
