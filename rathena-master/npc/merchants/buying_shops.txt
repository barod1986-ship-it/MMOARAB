que_job01,68,84,1	script	تاجر السوق السوداء#Buying::Black Marketeer#Buying	881,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "[السيد جاس]";
		mes "حتى شخص مثلي يراك جشعا جدا!";
		mes "خفف حمولة حقيبتك أولا.";
		close;
	}
	if (getskilllv("ALL_BUYING_STORE") == 1) {
		mes "[السيد جاس]";
		mes "لقد أبرمت عقدا مع هيو بالفعل.";
		mes "لا شأن لي بك إذن.";
		close;
	}
	mes "[السيد جاس]";
	mes "لا بد أنك بحاجة ماسة إلى شيء حتى جئت تبحث عني.";
	mes "ما الذي تريده؟";
	next;
	switch(select("شراء رخصة متجر الشراء بالجملة:من أنت؟:لا شيء!")) {
	case 1:
		mes "[السيد جاس]";
		mes "كنت أعلم ذلك!";
		mes "بالطبع، أستطيع صنعها لك.";
		mes "رخصتي تشبه الرخصة الأصلية التي يحصل عليها التجار من ذلك الوغد، السيد هيو!";
		next;
		mes "[السيد جاس]";
		mes "والأفضل أنك لا تحتاج إلى تعلم أي مهارة لاستخدام رخصتي.";
		mes "كم رخصة تريد؟";
		next;
		mes "[السيد جاس]";
		mes "لا أستطيع صنع أكثر من 10 رخص في المرة الواحدة.";
		mes "سعر الرخصة الواحدة 500 زيني.";
		next;
		while(1) {
			input .@input;
			mes "[السيد جاس]";
			if (.@input == 0) {
				mes "ألست بحاجة إلى الرخص؟";
				close;
			} else if (.@input > 10) {
				mes "لا أستطيع صنع أكثر من 10 رخص في المرة الواحدة.";
				next;
			} else {
				mes "سيبلغ السعر "+(.@input*500)+" زيني.";
				if (Zeny < (.@input*500)) {
					mes "لكن رصيدك لا يكفي.";
					next;
					mes "[السيد جاس]";
					mes "ألا تعرف أبسط قواعد التجارة؟ لكل شيء ثمن.";
					mes "إن أردت شيئا، فعليك دفع ثمنه.";
				} else {
					mes "ها... ها ها ها!";
					mes "يا هيو، سأنتزع منك تجارة الرخص. سترى!";
					mes "*ضحكة مكتومة*";
					getitem 12548,.@input; //Buy_Market_Permit2
					set Zeny,Zeny-(.@input*500);
				}
				close;
			}
		}
	case 2:
		mes "[السيد جاس]";
		mes "غادرت موطني منذ زمن بعيد.";
		mes "لا معنى للسؤال عن هويتي؛ لم يبق لي سوى الكراهية.";
		next;
		mes "[السيد جاس]";
		mes "...";
		mes "هيو تاجر فاسد لا يعرف أخلاقيات التجارة.";
		mes "وهدفي الوحيد هو القضاء عليه.";
		next;
		mes "[السيد جاس]";
		mes "آه، لقد أكثرت من الشراب... *فواق*";
		mes "إنه تأثير الكحول فحسب.";
		mes "انس كل ما قلته.";
		close;
	case 3:
		mes "[السيد جاس]";
		mes "حسنا! لا داعي للصراخ.";
		mes "دعني وشأني ما دمت لا تريد شيئا.";
		close;
	}
}

					mes "ألغيت الصفقة.";
					mes "أتمنى لك يوما طيبا.";

		} else {

			mes "يبدو أنك...";

			mes "[السيد هيو]";
			mes "هل تعلم؟ أصدرت نقابتنا رخصة تتيح للأفراد شراء السلع من الآخرين.";

			mes "هذا غير عملي، أليس كذلك؟";
			next;
			mes "[السيد هيو]";
			mes "الشراء بكميات كبيرة مهم لنا نحن التجار،";
			mes "لذلك قدمت إلى النقابة خطة مبتكرة، ثمرة عشر سنوات من خبرتي في الشراء.";
