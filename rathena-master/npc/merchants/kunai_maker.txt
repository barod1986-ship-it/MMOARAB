		case 4: callfunc "Kunai_Trade",13253,2,7524,1,13256; break;

function	script	Kunai_Trade	{
	setarray .@item_req[0], getarg(0), getarg(2);
	setarray .@req_amount[0], getarg(1), getarg(3);
	.@item_id = getarg(4);

	mes "[كاشين]";
	mes "يمكنك استبدال";
	mes .@req_amount[0] +" "+ getitemname(.@item_req[0]) +" و";
	mes .@req_amount[1] +" "+ getitemname(.@item_req[1]) +" مقابل";
	mes "مجموعة من 10 "+ getitemname(.@item_id) +".";
	next;
	mes "[كاشين]";
	mes "يمكنني تجهيز 500 مجموعة كوناي كحد أقصى في كل صفقة.";
	mes "أدخل ''0'' لإلغاء المبادلة.";
	mes "كم مجموعة كوناي تريد؟";
	next;
	input .@amount;
	if(.@amount < 1 || .@amount > 500) {
		mes "[كاشين]";
		mes "لا يمكن تنفيذ هذا العدد.";
		mes "يمكنك طلب عدد من";
		mes "1 إلى 500 مجموعة،";
		mes "أو إدخال 0 للإلغاء.";
		close;
	}
	if (countitem(.@item_req[0]) < .@amount*.@req_amount[0] || countitem(.@item_req[1]) < .@amount*.@req_amount[1]) {
		mes "[كاشين]";
		mes "لا تملك المواد";
		mes "الكافية لهذه المبادلة.";
		mes "تحقق من الكميات";
		mes "المطلوبة ثم عد إلي.";
		close;
 	}
	if (checkweight(.@item_id, .@amount*10) == 0) {
		mes "[كاشين]";
		mes "لا توجد مساحة كافية";
		mes "في حقيبتك لحفظ";
		mes "الكوناي الجديدة.";
		mes "أفرغ بعض المساحة";
		mes "أولا ثم عد إلي.";
		close;
	}
	mes "[كاشين]";
	mes "ممتاز، المواد والكميات";
	mes "كلها صحيحة.";
	mes "سنكمل المبادلة الآن.";
	mes "ستجد هذه الكوناي";
	mes "مفيدة في معاركك.";
	mes "استخدمها بحكمة.";
	delitem .@item_req[0],.@req_amount[0]*.@amount;
	delitem .@item_req[1],.@req_amount[1]*.@amount;
	getitem .@item_id,10*.@amount;
	close;
}
