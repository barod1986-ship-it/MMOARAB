function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[تاجر المبتدئين]";
		mes "لا تملك الكمية";
		mes "المطلوبة من "+ getitemname(.@item_req) +"";
		mes "لإتمام المبادلة.";
		mes "اجمع المزيد من "+ getitemname(.@item_req) +" ثم عد إلي.";
		close;
	}
	else {
		mes "[تاجر المبتدئين]";
		mes "دعني أتحقق من";
		mes "كمية "+ getitemname(.@item_req) +"";
		mes "التي تحملها...";
		next;
		mes "[تاجر المبتدئين]";
		mes "الكمية التي لديك:";
		mes ""+ countitem(.@item_req) +" من "+ getitemname(.@item_req) +".";
		mes "يمكنك استبدالها بما يصل إلى";
		mes ""+ (countitem(.@item_req)/.@req_amount) +" من الجرعات الحمراء.";
		next;
		mes "[تاجر المبتدئين]";
		mes "هل تريد";
		mes "إتمام المبادلة؟";
		next;
		if (select("إتمام المبادلة.:إلغاء.") == 1) {
			mes "[تاجر المبتدئين]";
			mes "حدد عدد الجرعات الحمراء";
			mes "التي تريد استلامها";
			mes "مقابل ما لديك من";
			mes ""+ getitemname(.@item_req) +".";
			next;
			mes "[تاجر المبتدئين]";
			mes "يمكنك طلب ما يكفي";
			mes "للحصول على جرعة واحدة";
			mes "وحتى 100 جرعة حمراء";
			mes "في الصفقة الواحدة. أدخل";
			mes "'0' إن أردت الإلغاء.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[تاجر المبتدئين]";
					mes "ألغيت المبادلة؟";
					mes "لا بأس.";
					mes "يمكنك العودة";
					mes "متى غيرت رأيك.";
					close;
				}
				else if (.@input > 100) {
					mes "[تاجر المبتدئين]";
					mes "لا يمكنني إعطاؤك أكثر";
					mes "من 100 جرعة حمراء";
					mes "في صفقة واحدة.";
					mes "أدخل عددا آخر.";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[تاجر المبتدئين]";
				mes "لا تملك مواد كافية للحصول على هذا العدد من الجرعات الحمراء. تحقق من الكمية المتاحة وحاول مجددا.";
				close;
			}
			mes "[تاجر المبتدئين]";
			mes "تمت المبادلة!";
			mes "تحقق من الجرعات الحمراء";
			mes "التي استلمتها. شكرا لك، كانت صفقة موفقة~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[تاجر المبتدئين]";
		mes "همم~";
		mes "حسنا.";
		mes "لكن الجرعات الحمراء";
		mes "مفيدة جدا";
		mes "في بداية مغامرتك.";
		close;
	}
}
