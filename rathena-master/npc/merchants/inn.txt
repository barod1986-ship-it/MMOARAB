//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "أهلا بك في";
	mes getarg(1) + ".";
	mes "كيف يمكنني خدمتك؟";
	next;
	switch(select("حفظ نقطة العودة:الإقامة - 5,000 زيني:إلغاء")) {
	case 1:
		mes .@npc_name$;
		mes "تم حفظ نقطة";
		mes "عودتك هنا.";
		mes "شكرا لك،";
		mes "ونرجو زيارتك مجددا.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "عذرا، رسوم الإقامة 5,000 زيني. يرجى إحضار المبلغ الكافي في زيارتك القادمة.";
			close;
		}
		mes "شكرا لك.";
		mes "نتمنى لك";
		mes "إقامة مريحة~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
