	mes "وحتى عند النجاح، ستزول ترقية "+ ((getiteminfo(.@item_id, ITEMINFO_LOCATIONS)&EQP_HAND_R) ? "السلاح" : "الدرع") +" وبطاقاته،";
	mes "لأن القطعة ستستبدل بنسخة جديدة ذات فتحة.";
	next;
	switch(select("بدء إضافة الفتحة:التراجع"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no .@req_amount[0] and .@req_amount[1] provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[ليابليم]";
				mes "حسنا، لنبدأ العمل!";
				mes "تمن أن يحالفنا الحظ.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect EF_LORD;
					mes "[ليابليم]";
					mes "رائع، نجحت العملية!";
					mes "يبدو العمل متقنا جدا. تهانينا!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[ليابليم]";
					mes "أراك في المرة القادمة!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[ليابليم]";
					mes "واه! ...أنا آسف جدا، لقد فشلت.";
					mes "لقد بذلت كل ما أستطيع.";
					mes "لكن الحظ لم يحالفنا هذه المرة.";
					mes "لا تحبط كثيرا،";
					mes "وجرب في المرة القادمة.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[ليابليم]";
					mes "أراك في المرة القادمة!";
					close;
				}
			}
			else
			{
				mes "[ليابليم]";
				mes "لا يمكنني البدء؛ لم تحضر جميع المتطلبات.";
				mes "جهزها كلها ثم عد إلي.";
				close;
			}
		case 2:
			mes "[ليابليم]";
			mes "أراك في المرة القادمة.";
			close;
	}
}
