//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "أنا خبير في ترقية المعدات.";
	mes "أستطيع ترقية مختلف الأسلحة والدروع، لذا";
	mes "اختر القطعة التي تريد ترقيتها.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "لا أرى أي قطعة مجهزة يمكنني ترقيتها...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "لا توجد في هذا الموضع";
		mes "قطعة مجهزة";
		mes "يمكنني ترقيتها.";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "لا يمكن ترقية";
		mes "هذه القطعة إطلاقا...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "لا يمكنني ترقية هذه القطعة";
		mes "أكثر من ذلك. لقد بلغت";
		mes "الحد الأقصى للترقية!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "تحتاج ترقية هذه القطعة إلى";
		mes "قطعة واحدة من ^003366"+getitemname(.@material)+"^000000 و";
		mes "رسم خدمة قدره " + .@price + " زيني.";
		mes "هل تريد المتابعة؟";
		next;
		if(select("نعم:لا") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "حسنا...";
			mes "لا داعي إلى";
			mes "العجلة. خذ وقتك.";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "انتبه! إذا تابعت";
			mes "ترقية هذه القطعة، فقد";
			switch(.@material) {
			case 985:
				mes "تتحطم! ستفقد ^FF0000المعدة^000000 وكل ^FF0000البطاقات^000000 والخصائص الخاصة المضافة إليها ^FF0000نهائيا^000000.";
				break;
			default:
				mes "تتحطم، وستفقد ^FF0000السلاح^000000 وكل ^FF0000البطاقات^000000 الموجودة فيه،";
				mes "وأي خصائص خاصة مضافة إليه.";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "أوضح لك الأمر للمرة الأخيرة:";
			mes "إذا تحطمت القطعة،";
			mes "فلن تتمكن من استعادتها.";
			mes "هناك احتمال حقيقي";
			mes "^FF0000لفقدانها إلى الأبد^000000.";
			mes "هل ما زلت تريد المتابعة؟";
			next;
			if(select("نعم:لا") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "قرار حكيم...";
				mes "قد أكون خبيرا في الترقية، لكن الخطأ وارد دائما.";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "لا تملك ما يكفي من";
			mes "الزيني أو مادة "+getitemname(.@material)+" المطلوبة...";
			mes "أحضر المتطلبات كاملة ثم عد.";
			mes "سأكون هنا.";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "انتظر لحظة...";
			mes "هل تظن أنني لن ألاحظ؟!";
			mes "لقد بدلت القطعة أثناء العمل. اخرج من هنا!";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "يا إلهي!";
				mes "لا... ليس مرة أخرى!";
				mes "أنا آسف، لكنك تعرف أن الإتقان يحتاج إلى تدريب.";
				mes "أليس كذلك؟ هيه هيه...";
			} else if(.@lose == 2) {
				mes "لااااا!";
				mes "لقد تحطمت القطعة!";
				mes "أ-أنا آسف!";
			} else {
				mes "تبا!";
				mes "لم تتحمل القطعة";
				mes "مزيدا من الطرق والتقسية!";
				mes "آسف على ذلك...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "مثالي!";
			mes "هيه هيه!";
			mes "مرة أخرى،";
			mes "عمل متقن";
			mes "من يد الأستاذ~";
		} else if(.@win == 2) {
			mes "نجحت الترقية!";
			mes "مرة أخرى تتجلى";
			mes "موهبتي المذهلة";
			mes "وتتألق اليوم.";
		} else {
			mes "هيه هيه!";
			mes "انتهيت.";
			mes "لا شك أن النتيجة";
			mes "ستنال رضاك.";
			mes "إتقان تام ومطلق~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "يمكنني ترقيتها إلى الحد الآمن أو تكرار المحاولة عددا تحدده. الخيار لك.";
		next;
		.@menu2 = select("الترقية إلى الحد الآمن.","تحديد عدد المحاولات.","إلغاء.");

		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "كم مرة تريد أن أحاول ترقية القطعة؟";

		if (.@refinecnt < 1 || .@refinecheck > 10) {

			next;
			if(select("نعم...","لا...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "كما تشاء. لن أتابع.";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "كما تشاء. لن أتابع.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";

	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){

		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "انتهيت من العمل. عد مرة أخرى قريبا.";
	close;
}

	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "أبيع معدنين";
	mes "يستخدمان في ترقية الأسلحة.";
	mes "^007777فراكون^000000 لأسلحة المستوى 1،";
	mes "و^007777إمفيريتاركون^000000";
	mes "لأسلحة المستوى 2.";
	next;
	switch(select("فراكون - 200 زيني:إمفيريتاركون - 1,000 زيني:السؤال عن المعادن الأخرى")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "معادن أخرى؟";
		mes "تحتاج الأسلحة الأعلى مستوى والدروع إلى معادن خاصة للترقية. لكن";
		mes "العثور على أوريديكون وإيلونيوم ليس سهلا...";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "كم وحدة تريد شراءها؟";
	mes "أدخل الرقم 0 للإلغاء.";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "ألغيت";
			mes "الصفقة.";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "يمكنك";
			mes "شراء 500 وحدة كحد أقصى.";
			mes "لا يمكنني بيع";
			mes "أكثر من ذلك.";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "همم...";
		mes "لا تملك";
		mes "زينيا كافيا لشراء";
		mes ""+ .@input +" وحدة.";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "لا توجد مساحة كافية";
		mes "في حقيبتك. أودع أغراضك الزائدة في مخزن كافرا ثم عد.";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "تفضل.";
	mes "شكرا";
	mes "لتعاملك معنا.";
	close;
}

//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "أهلا بك!";
	mes "هل لديك";
	mes "معدات تالفة";
	mes "تريد مني إصلاحها؟";
	mes "يمكنك الاعتماد علي!";
	next;
	switch(select("نعم، لدي معدات تالفة.:لا أحتاج إلى الإصلاح الآن.")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "ممتاز!";
			mes "يبدو أنك تعتني بمعداتك جيدا؛ لا توجد لديك أي قطعة تالفة.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "لو كان الجميع مثلك، فسأفقد عملي! هاها!";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "همم...";
		mes "دعني أفحص معداتك...";
		mes "وجدت";
		mes "" + .@checkitem + " قطعة تالفة.";
		mes "هل تريد إصلاحها؟";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "يكلف إصلاح القطعة الواحدة " + .@repairprice + " زيني، وتبلغ تكلفة إصلاح جميع القطع " + .@totalcost + " زيني. هل أبدأ؟";
		next;
		switch(select("نعم:لا")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "انتظر...";
				mes "لا تملك زينيا كافيا لدفع تكلفة الإصلاح.";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "انتهيت! حاول أن تحافظ على معداتك مستقبلا؛ فلها عمر أيضا.";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "هناك خطأ ما. جهز المعدات التي تريد إصلاحها ثم عد إلي.";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "كما تشاء، لكن ترك المعدات تالفة ليس فكرة جيدة. أصلحها في أقرب وقت.";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "حسنا...";
		mes "لا تحتاج";
		mes "إلى خدمتي";
		mes "ما دامت معداتك";
		mes "سليمة.";
		close;
	}
}
