-	script	::merchant_13_2	-1,{
	set .@name$,strnpcinfo(1);
	set .@n$,"["+ .@name$ +"]";
	if (.@name$ == "Merchant of Manuk") {
		set .@n$,"[تاجر مانوك]";
	}
	if (.@name$ == "Merchant of Splendide") {
		set .@n$,"[تاجر سبلينديد]";
	}
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "يبدو أن حمولة أغراضك زائدة.";
			mes "ضع بعض أغراضك في المخزن ثم عد.";
			close;
		}
		mes "أهلا بك.";
		mes "ما الذي ترغب في شرائه؟";
		next;
		if (strnpcinfo(3) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("عرض أوصاف السلع:فرصة مانوك:شجاعة مانوك:إيمان مانوك:إلغاء");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("عرض أوصاف السلع:شراء مربى فاكهة بينغويكولا:شراء مربى عسل لوسيولا:شراء دموع كورنوس:إلغاء");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "إليك أوصاف السلع المتاحة.";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "يمكنني بيعك ثلاث وحدات من "+getitemname(.@Items[.@i])+" مقابل ^3131FFعملة واحدة^000000.";
			next;
			switch(select("شراء:إلغاء")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "شكرا لتعاملك معنا.";
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "لا تملك العملة المطلوبة.";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "عد إذا غيرت رأيك.";
		close;
	}
	else {
		if (strnpcinfo(3) == "Merchant of Manuk") {
			mes "رتت أود د";
			mes "كيوو هد إز د إرر";
		}
		else {
			mes "بورويه ألا";
			mes "تاسنار أندو إي رو";
		}
		close;
	}
}
