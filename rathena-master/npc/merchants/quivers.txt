	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[المخترع جاكس]";
		mes "مهلا، تحمل";
		mes "أغراضا كثيرة جدا. أودع بعضها في مخزن كافرا، ثم عد إلي.";
		close;
	}
	mes "[المخترع جاكس]";
	mes "أنا جاكس.";
	mes "ومن دون غرور،";
	mes "أستطيع القول إنني ربما";
	mes "^663300أعظم مخترع في هذا العصر^000000.";
	next;
	mes "[المخترع جاكس]";
	mes "لقد صنعت هذه المرة";
	mes "اختراعا";
	mes "استثنائيا حقا، وأسميه... ^663300الجعب السحرية^000000! وسيسجل التاريخ";
	mes "هذه الثورة في عالم الأسهم!";
	emotion ET_BEST;
	next;
	mes "[المخترع جاكس]";
	mes "درست السحر والجعب سنوات طويلة، وعملت ليلا ونهارا حتى توصلت إلى طريقة لضغط الأسهم بالسحر! تتيح لك الجعب السحرية حمل سهام أكثر بوزن أقل!";
	next;
	mes "[المخترع جاكس]";
	mes "هل تود تجربة إحدى جعب الأسهم التي صنعتها؟ لا شك أن شخصا مثلك سيقدر عبقريتي!";
	next;
	switch(select("جعبة:جعبة سهام حديدية:جعبة سهام فولاذية:جعبة سهام أوريديكون:جعبة سهام نارية:جعبة سهام فضية:جعبة سهام الرياح:جعبة سهام حجرية:جعبة سهام كريستالية:جعبة سهام الظل:جعبة سهام غير مادية:جعبة سهام صدئة")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Is there";
	//mes "anything you want?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[المخترع جاكس]";
		mes "رائع!";
		mes "يمكنني وضع "+ .@amount +" من "+ getitemname(.@item_id) +" في جعبة مقابل ^FF3131"+ .@zeny_req +" زيني^000000.";
		next;
		switch(select("تعبئة أكبر عدد ممكن من الجعب:صنع جعبة واحدة:إلغاء")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "عدد الأسهم";
			mes "المتاحة: ^3131FF"+.@arrows+" ^000000";
			mes "أقصى عدد";
			mes "من الجعب";
			mes "التي يمكن صنعها: ^3131FF"+.@quiver+" ^000000";
			mes "التكلفة الإجمالية: ^3131FF"+.@arrow_zeny01+" زيني^000000";
			next;
			mes "[المخترع جاكس]";
			mes "هل تريد";
			mes "صنع أكبر عدد ممكن";
			mes "من الجعب باستخدام الأسهم";
			mes "التي تحملها الآن؟";
			next;
			if (select("نعم:إلغاء") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[المخترع جاكس]";
					mes "انتهيت!";
					mes "تذكر: ^FF0000لا يمكنك فتح الجعبة إذا بلغ وزن ما تحمله 90% من الحد الأقصى^000000.";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[المخترع جاكس]";
					mes "لذلك...";
					mes "راقب دائما";
					mes "مقدار الوزن";
					mes "الذي تحمله";
					mes "من حين إلى آخر،";
					mes "ولن تواجه مشكلة.";
					close;
				}
				else {
					mes "[المخترع جاكس]";
					mes "لا تملك زينيا كافيا. لا يمكنني منح هذه الجعب مجانا بعد سنوات";
					mes "قضيتها في تطوير هذه التقنية الثورية!";
					close;
				}
			}
			mes "[المخترع جاكس]";
			mes "هل غيرت رأيك؟";
			mes "وأنت على بعد خطوة من امتلاك";
			mes "إحدى هذه الجعب الرائعة؟";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[المخترع جاكس]";
				mes "انتهيت!";
				mes "تذكر: ^FF0000لا يمكنك فتح الجعبة إذا بلغ وزن ما تحمله 90% من الحد الأقصى^000000.";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[المخترع جاكس]";
				mes "لذلك...";
				mes "راقب دائما";
				mes "مقدار الوزن";
				mes "الذي تحمله";
				mes "من حين إلى آخر،";
				mes "ولن تواجه مشكلة.";
				close;
			}
			else {
				mes "[المخترع جاكس]";
				mes "ألا تملك حتى";
				mes ""+ .@zeny_req +" زيني؟";
				mes "يا للأسف... لم أكن";
				mes "أعلم أنك تعاني هذا القدر من...";
				mes "^333333الفاقة^000000.";
				close;
			}
		case 3:
			mes "[المخترع جاكس]";
			mes "ماذا...؟";
			mes "ألا تدرك أن هذا الاختراع سيغير طريقة حمل الأسهم إلى الأبد؟! هذا هو المستقبل!";
			close;
		}
	}
	else {
		mes "[المخترع جاكس]";
		mes "تتسع هذه الجعبة إلى "+ .@amount +" سهم. صنعتها بطريقتي السرية،";
		mes "لذلك يكون وزن الجعبة بأسهمها أقل من وزن الأسهم منفردة.";
		next;
		mes "[المخترع جاكس]";
		mes "إنها معجزة علمية! أحضر "+ .@amount +" من الأسهم المطلوبة و"+ .@zeny_req +" زيني لكل جعبة لتجربها بنفسك.";
		close;
	}
}
