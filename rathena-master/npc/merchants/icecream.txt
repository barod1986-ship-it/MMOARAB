-	script	صانع مثلجات::IceCreamer	-1,{
	mes "[صانع مثلجات]";
	mes "آيس كريم طازج مصنوع من ثلوج لوتي!";
	mes "تذوقه الآن، فلن أبيعه لوقت طويل!";
	mes "^3355FF100 زيني^000000 فقط للواحدة،";
	mes "آيس كريم لذيذ!";
	next;
	if(select("أريد آيس كريم!:إلغاء الشراء") == 2) {
		mes "هل تريد حقا إلغاء الشراء؟";
		mes "لن أستمر في البيع طويلا،";
		mes "وما إن تنفد الكمية،";
		mes "فلن يتبقى منه شيء!";
		close;
	}
	mes "[صانع مثلجات]";
	mes "آيس كريم طازج مصنوع من ثلوج لوتي!";
	mes "الجميع يريد الآيس كريم اللذيذ،";
	mes "لكن الكمية المتاحة محدودة،";
	mes "لذلك يمكنك شراء 5 فقط في كل مرة!";
	next;
	while(true) {
		input .@input;
		if(.@input < 1) {
			mes "[صانع مثلجات]";
			mes "إذا كنت لا تريد الشراء،";
			mes "فدع الزبون التالي";
			mes "يتقدم لإتمام طلبه.";
			mes "شكرا لك.";
			close;
		}
		if(.@input > 5) {
			mes "[صانع مثلجات]";
			mes "مهلا!";
			mes "هذا طلب كبير جدا.";
			mes "يمكنك شراء 5 فقط في كل مرة.";
			mes "فتناول أكثر من ذلك";
			mes "قد يجعلك تكثر من زيارة";
			mes "إلى الحمام هذه الليلة.";
			next;
			continue;
		}
		break;
	}
	if(Zeny<.@input*100) {
		mes "[صانع مثلجات]";
		mes "يبدو أن ما معك من زيني لا يكفي.";
		mes "السعر ^3355FF100 زيني^000000 لكل حبة آيس كريم.";
		close;
	}
	if(checkweight(536,.@input) == 0) {
		mes "[صانع مثلجات]";
		mes "يبدو أن حقيبتك ممتلئة.";
		mes "الآيس كريم خفيف،";
		mes "لكن عليك تخفيف حملك";
		mes "قبل الشراء.";
		close;
	}
	set Zeny,Zeny-100*.@input;
	getitem 536,.@input;	// Ice_Cream
	close;
}

// Alberta
alberta,120,45,2	duplicate(IceCreamer)	صانع مثلجات#1::Ice Cream Maker#1	85
