function Dyes {
	setarray .@DyeName$[0],"اللون الافتراضي","البرتقالي","البنفسجي","الأحمر","الأسود","الأخضر","الأزرق","الأبيض","الأصفر";
	setarray .@DyeItems[0],0,980,981,975,983,979,978,982,976;
	set .@menu$,"- اللون الافتراضي:";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+2)
		set .@menu$, .@menu$+"- "+.@DyeName$[getarg(.@i)]+":";
	set .@menu$, .@menu$+"- ^777777إلغاء^000000";
	set .@s, select(.@menu$)-1;
	if (.@s == (getargcount()/2)+1) goto L_End;
	mes .n$;
	if (.@s == 0) {
		mes "يمكنني إعادة لون ملابسك إلى";
		mes "اللون الافتراضي مجانا.";
		mes "هل تريد المتابعة؟";
		next;
		switch(select("نعم:لا")) {
		case 1:
			mes .n$;
			setlook 7,0;
			mes "أعيد لون ملابسك إلى "+.@DyeName$[0]+".";
			close;
		case 2:
			goto L_End;
		}
	}
	if (!countitem(.@DyeItems[getarg((.@s-1)*2)])) {
		mes "أحتاج إلى الأصباغ المناسبة لصبغ ملابسك. راجع قائمة الأسعار لمعرفة النوع المطلوب.";
		close;
	}
	if (Zeny < 10000) {
		mes "آسف، لا تملك 10,000 زيني المطلوبة.";
		close;
	}
	delitem .@DyeItems[getarg((.@s-1)*2)],1;
	set Zeny, Zeny-10000;
	setlook 7, ((.@s)?(getarg((.@s-1)*2+1)):0);
	mes "صُبغت ملابسك باللون "+((.@s)?.@DyeName$[getarg((.@s-1)*2)]:.@DyeName$[0])+".";
	close;
}
