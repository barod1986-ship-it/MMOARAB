alde_alche,24,188,3	script	تاجر النقابة::Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- لحظة من فضلك! -";
		mes "- تحمل حاليا عددا كبيرا -";
		mes "- من الأغراض. -";
		mes "- عد بعد إيداع بعضها -";
		mes "- في مخزن كافرا. -";
		close;
	}
	mes "[غيفر آل شارب]";
	mes "مرحبا بك في";
	mes "اتحاد الخيميائيين.";
	mes "كيف يمكنني خدمتك اليوم؟";
	next;
	switch(select("شراء مواد.:شراء دليل صنع.:مغادرة.")) {
	case 1:
		mes "[غيفر آل شارب]";
		mes "ما الذي تريده؟";
		next;
		if (select("وعاء الدواء - 8 زيني:إلغاء.") == 1) {
			mes "[غيفر آل شارب]";
			mes "كم وعاء تريد؟";
			mes "أدخل 0 لإلغاء العملية.";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[غيفر آل شارب]";
					mes "ألغيت الصفقة.";
					mes "أهلا بك في أي وقت.";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[غيفر آل شارب]";
					mes "يجب ألا يتجاوز العدد";
					mes "2000 وعاء.";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[غيفر آل شارب]";
				mes "ليس لديك";
				mes "ما يكفي من الزيني.";
				mes "تحقق من رصيدك";
				mes "ثم حاول مجددا.";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[غيفر آل شارب]";
				mes "لا يمكنك حمل";
				mes "هذه الكمية كلها.";
				mes "تحقق من المساحة المتاحة";
				mes "في حقيبتك.";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[غيفر آل شارب]";
			mes "شكرا لتعاملك معنا.";
			mes "أهلا بك مجددا.";
			close;
		}
		mes "[غيفر آل شارب]";
		mes "حسنا.";
		mes "عد عندما تحتاج";
		mes "إلى شراء مواد";
		mes "متعلقة بالخيمياء.";
		mes "أهلا بك في أي وقت.";
		close;
	case 2:
		mes "[غيفر آل شارب]";
		mes "أي دليل تحتاج؟";
		mes "سعر معظم الأدلة 100,000 زيني، لكن بعض الأدلة الخاصة أغلى.";
		next;
		switch(select("دليل صنع الجرعات:دليل صنع الكحول:دليل صنع القنبلة الزجاجية:دليل صنع زجاجة الحمض:دليل صنع زجاجة النبات:دليل صنع زجاجة مارين سفير:دليل صنع الطلاء اللامع:دليل صنع الجرعات المكثفة:إلغاء الصفقة.")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[غيفر آل شارب]";
			mes "حسنا...";
			mes "عد عندما تحتاج";
			mes "إلى شراء أحد";
			mes "أدلة الصنع.";
			close;
		}
	case 3:
		mes "[غيفر آل شارب]";
		mes "حسنا.";
		mes "أتمنى لك يوما طيبا.";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[غيفر آل شارب]";
		mes "ليس لديك";
		mes "ما يكفي من الزيني.";
		mes "تحقق من رصيدك";
		mes "ثم حاول مجددا.";
		close;
	}
	mes "[غيفر آل شارب]";
	mes getitemname(.@item_id) +"?";
	mes "سعره "+ .@zeny_req +" زيني.";
	next;
	if (select("شراء.:إلغاء.") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[غيفر آل شارب]";
		mes "شكرا لك";
		mes "على تعاملك معنا.";
		close;
	}
	mes "[غيفر آل شارب]";
	mes "أهلا بك";
	mes "في أي وقت.";
	close;
}
