payon,173,238,5	script	جايد#pay::Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FFمهلا! تحمل حاليا أغراضا كثيرة. ضع بعض أغراضك في مخزن كافرا ثم عد إلي.^000000";
		close;
	}
	mes "[جايد]";
	mes "أحضر لي جوهرتين";
	mes "من اللون نفسه،";
	mes "وسأبدلهما بجوهرة";
	mes "من لون مختلف.";
	mes "اختر نوع المبادلة التي تريدها.";
	next;
	switch(select("جوهرتان زرقاوان مقابل جوهرة حمراء.:جوهرتان حمراوان مقابل جوهرة صفراء.:جوهرتان صفراوان مقابل جوهرة زرقاء.")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[جايد]";
		mes "هاه...!";
		mes "أتمزح معي؟";
		mes "لا أستطيع إجراء المبادلة";
		mes "إذا لم تسلمني";
		mes "جوهرتين على الأقل من";
		mes getitemname(.@item_req) +"!";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[جايد]";
		mes "يمكنني صنع";
		mes "عدد "+ .@gems +" من "+ getitemname(.@item_id);
		mes "باستخدام "+ getitemname(.@item_req);
		mes "الموجودة في حقيبتك حاليا.";
		mes "ماذا تريد أن تفعل؟";
		next;
		switch(select("استبدل أكبر كمية ممكنة.:سأحدد الكمية.:إلغاء.")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[جايد]";
			mes "كم جوهرة تريد";
			mes "الحصول عليها؟";
			mes "أقصى كمية يمكنك";
			mes "إدخالها هي 100.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[جايد]";
					mes "لا تريد أي جوهرة؟";
					mes "يبدو أنك";
					mes "غيرت رأيك...";
					close;
				}
				else if (.@input > 100) {
					mes "[جايد]";
					mes "همم...";
					mes "طلبت منك إدخال";
					mes "كمية لا تزيد";
					mes "على 100، أتذكر؟";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[جايد]";
					mes "همم...";
					mes "لا تملك ما يكفي";
					mes "من الجواهر لهذه المبادلة...";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[جايد]";
			mes "حسنا، لا مشكلة.";
			mes "عد إلي في أي وقت.";
			close;
		}
		mes "[جايد]";
		mes "تفضل.";
		mes "لا تتردد في العودة";
		mes "متى احتجت إلى المبادلة.";
		mes "همم، لماذا تنظر إلي هكذا؟";
		mes "هل يوجد شيء على وجهي؟";
		close;
	}
}
