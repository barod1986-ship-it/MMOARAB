//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[موظف شركة كول إيفنت]";
	mes "أهلا بك في شركة كول إيفنت.";
	mes "يسعى موظفونا دائما";
	mes "إلى تقديم خدمة تفوق";
	mes "توقعاتك. كيف يمكنني";
	mes "مساعدتك اليوم؟";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[موظف شركة كول إيفنت]";
		mes "تم حفظ نقطة عودتك";
		mes "هنا";
		mes getarg(1)+".";
		mes "شكرا لاستخدامك";
		mes "خدمات شركة كول إيفنت~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[موظف شركة كول إيفنت]";
		if(!callfunc("F_CanOpenStorage")){
			mes "عذرا، لاستخدام";
			mes "خدمة المخزن، يجب أن تبلغ";
			mes "مهارة المبتدئ الأساسية";
			mes "المستوى 6.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "عذرا، لا تملك";
			mes "ما يكفي من الزيني لاستخدام";
			mes "خدمة المخزن؛ رسوم";
			mes "استخدام المخزن 40 زيني.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "سأفتح مخزنك";
		mes "الشخصي على الفور.";
		mes "شكرا لدعمك شركة";
		mes "كول إيفنت باستخدام";
		mes "خدماتنا. نتمنى لك يوما سعيدا~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "يرجى اختيار";
		mes "وجهتك.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		.@destination_display$ = .@destination$;
		if (.@destination_display$ == "Veins") set .@destination_display$,"فينس";
		else if (.@destination_display$ == "Rachel") set .@destination_display$,"راشيل";
		if (select("الانتقال إلى "+ .@destination_display$ +" - "+ .@cost +" زيني:إلغاء") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[موظف شركة كول إيفنت]";
				mes "عذرا، لا تملك";
				mes "ما يكفي من الزيني لاستخدام";
				mes "خدمة النقل. تكلفة الانتقال";
				mes "إلى "+ .@destination_display$ +" هي "+.@cost+" زيني.";
			}
		}
		close2;
		break;
	case 4:
		mes "[موظف شركة كول إيفنت]";
		if(BaseClass != Job_Merchant){
			mes "عذرا، خدمة";
			mes "استئجار عربة الدفع";
			if (checkre(0))
				mes "متاحة فقط لوظائف التاجر.";
			else {
				mes "متاحة فقط للتجار،";
				mes "والحدادين، وسادة الحدادين،";
				mes "والخيميائيين والكيميائيين الحيويين.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "لا يمكنك استئجار عربة دفع";
			mes "قبل تعلم مهارة عربة الدفع.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "لديك عربة دفع";
			mes "مجهزة بالفعل.";
			mes "للأسف، لا يمكننا";
			mes "تأجير أكثر من عربة واحدة";
			mes "لكل عميل في الوقت نفسه.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "تبلغ رسوم استئجار";
		mes "عربة الدفع 800 زيني.";
		mes "هل ترغب في استئجارها؟";
		next;
		if(select("استئجار عربة دفع:إلغاء") == 1){
			if(Zeny < 800){
				mes "[موظف شركة كول إيفنت]";
				mes "عذرا، لا تملك";
				mes "ما يكفي من الزيني";
				mes "لدفع رسوم استئجار";
				mes "عربة الدفع البالغة 800 زيني.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[موظف شركة كول إيفنت]","شركة كول إيفنت");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[موظف شركة كول إيفنت]";
		mes "تسعى شركة كول إيفنت دائما";
		mes "إلى تقديم أفضل الخدمات";
		mes "لعملائها. ساعدنا";
		mes "على بلوغ الأفضل بتزويدنا";
		mes "بآرائك وملاحظاتك";
		mes "الصادقة. شكرا لك.";
		close2;
	}
	cutin "", 255;
	end;
}
