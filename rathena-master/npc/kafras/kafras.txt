aldeba_in,96,181,4	script	كافرا ليلاه::Kafra Service	113,{
	cutin "kafra_05",2;
	mes "[كافرا ليلاه]";
	mes "همم...؟";
	mes "أوه، أهلا بك في";
	mes "المقر الرئيسي لشركة كافرا.";
	mes "هل تحتاج";
	mes "إلى شيء؟";
	next;
	switch(select("حفظ نقطة العودة:استخدام المخزن:استئجار عربة دفع:إلغاء")) {
	case 1:
		mes "[كافرا ليلاه]";
		mes "تم حفظ نقطة عودتك";
		mes "هنا داخل";
		mes "المقر الرئيسي لشركة";
		mes "كافرا. شكرا لك.";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[كافرا ليلاه]";
		mes "يرجى الاستفادة من";
		mes "خدمات كافرا المتاحة";
		mes "في جميع أنحاء";
		mes "ميدغارد. شكرا لك على";
		mes "زيارة مقر كافرا الرئيسي.";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[كافرا ليلاه]";
			mes "عذرا، ولكن يبدو";
			mes "أنك لا تملك";
			mes "20 زيني المطلوبة";
			mes "لرسوم استخدام المخزن...";
			close3;
		}
		mes "[كافرا ليلاه]";
		mes "رغم أن هذه المنشأة";
		mes "مخصصة حصرا";
		mes "لتدريب موظفات كافرا";
		mes "والأعمال الإدارية،";
		mes "سأفتح لك المخزن.";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[كافرا ليلاه]";
		mes "في المرات القادمة، يرجى";
		mes "طلب الخدمة من موظفة كافرا";
		mes "المناوبة إذا أردت استخدام";
		mes "أي من خدمات كافرا.";
		mes "شكرا لاستخدامك خدماتنا.";
		callfunc("F_CheckKafCode");    //check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "[كافرا ليلاه]";
		mes "أعتذر، لكنني";
		mes "لست في الخدمة الآن. كنت";
		mes "سأساعدك لو استطعت، لكنني";
		mes "لا أملك عربات دفع متاحة.";
		mes "لم لا تطلب المساعدة من";
		mes "موظفة كافرا أخرى؟";
		next;
		mes "[كافرا ليلاه]";
		mes "تنتشر موظفات كافرا";
		mes "في جميع أنحاء";
		mes "قارة ميدغارد،";
		mes "وستجد عددا كبيرا منهن";
		mes "في الخارج في ألديباران.";
		close2;
		break;
	case 4:
		mes "[كافرا ليلاه]";
		mes "^666666*آه...*^000000";
		mes "هذا جيد، لأنني";
		mes "في استراحتي";
		mes "حاليا. وكان اختيار";
		mes "''إلغاء'' تصرفا";
		mes "صائبا منك.";
		close2;
		break;
	}
	cutin "",255;
	end;
}
