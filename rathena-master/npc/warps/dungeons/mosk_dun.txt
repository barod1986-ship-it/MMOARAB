mosk_dun02,53,220,4	script	#babayagain	WARPNPC,1,1,{
	end;
OnTouch:
	if (mos_nowinter == 11) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "[جندي]";
		mes "توقف مكانك!";
		next;
		mes "[جندي]";
		mes "لدينا معلومات";
		mes "تفيد بأنك كنت تتواصل سرا";
		mes "مع";
		mes "بابا ياغا الرهيبة.";
		next;
		mes "[جندي]";
		mes "إن كان لديك ما تقوله،";
		mes "فتحدث إلى القيصر.";
		mes "تعال معنا.";
		close2;
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		set mos_nowinter,12;
		warp "mosk_in",125,92;
		end;
	}
	else if (mos_nowinter == 12 || mos_nowinter == 13) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "[جندي]";
		mes "توقعنا";
		mes "أن تظهر مجددا.";
		mes "ما شأنك مع القيصر؟";
		next;
		if(select("الهرب.:شرح الموقف.") == 1) {
			mes "["+strcharinfo(0)+"]";
			mes "يجب أن أخرج من هنا...";
			close2;
			warp "mosk_dun02",100,100;
		}
		else {
			mes "- أشرح ما حدث حتى الآن. -";
			next;
			mes "[جندي]";
			mes "إن كان لديك ما تقوله، فتحدث";
			mes "إلى القيصر.";
			mes "إذا غادرت هذا المكان،";
			mes "فسنعدك عدوا لنا.";
			close2;
		}
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		end;
	}
	else {
		warp "mosk_in",215,41;
		end;
	}
}
