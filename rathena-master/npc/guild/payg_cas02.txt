-	script	القصر القرمزي#1::OutsideFlagsPA2	722,{

	set .@GID, GetCastleData("payg_cas02",1);

	if (.@GID == 0) {
		mes "[ مرسوم مملكة رون-ميدغارتس الإلهية ]";
		mes " ";
		mes "1. بناء على مرسوم مملكة رون-ميدغارتس الإلهية، ";
		mes "نعلن أنه";
		mes "ليس لهذه القلعة مالك رسمي.";
		mes " ";
		mes "2. من يستطيع ";
		mes "تجاوز كل المحن";
		mes "وتدمير الإمبريوم،";
		mes "سيمنحه الملك";
		mes "ملكية هذه القلعة.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ صوت يتردد ]";
		mes "أيها الشجعان...";
		mes "هل ترغبون في العودة إلى قلعة نقابتكم؟";
		next;
		if(select("العودة إلى قلعة النقابة.:إلغاء.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("payg_cas02",1)) {
					warp "payg_cas02",278,251;
				end;
			}
			end;
		}
		close;
	}
	mes "[ مرسوم مملكة رون-ميدغارتس الإلهية ]";
	mes " ";
	mes "1. بناء على مرسوم مملكة رون-ميدغارتس الإلهية، ";
	mes "نقر بأن هذه القلعة";
	mes "ملك خاص لنقابة ^ff0000"+GetGuildName(.@GID)+"^000000.";
	mes " ";
	mes "2. سيد نقابة ^ff0000"+GetGuildName(.@GID)+"^000000 هو:";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "إذا كان هناك من يعترض على هذا،";
	mes "فليثبت قوته وشرفه بنصل فولاذي.";
	close;

OnRecvCastlePy02:
	FlagEmblem GetCastleData("payg_cas02",1);
	end;
}
