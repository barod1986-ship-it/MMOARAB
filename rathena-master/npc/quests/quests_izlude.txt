// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[إدغار]";
		mes "إذن، هل ستتجه إلى ألبرتا مجددا؟ سأمنحك الخصم نفسه، ولن أطلب منك سوى 250 زيني، كما في المرة الماضية.";
		mes "ما رأيك في ذلك؟";
		next;
		if (select("حسنا إذن~!:لماذا أنت لطيف معي إلى هذا الحد؟!") == 1) {
			if (Zeny < 250) {
				mes "[إدغار]";
				mes "همم...";
				mes "هذا المبلغ";
				mes "غير كاف.";
				mes "اذهب واجمع";
				mes "بعض المال أولا.";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[إدغار]";
		mes "هذه طبيعتي فحسب. كما أن ابتسامتك المشاكسة تذكرني بابني الحبيب ذي الشعر الأشقر، الذي غادر المنزل قبل سنوات ليصبح بحارا على متن سفينته الخاصة. ليحفظه الله أينما كان.";
		close;
	}
	mes "[إدغار]";
	mes "ترتبط بلدتي إزلود بألبرتا عبر الميناء الغربي. حركة السفر بيننا نشطة جدا، حتى إنني أكاد أعد نفسي من أهل ألبرتا. هيهيهي~";
	next;
	mes "[إدغار]";
	mes "أعرف رجلا يدعى فيليكس ويعيش في ألبرتا. إنه بخيل حقا... ويتقاضى رسوما على كل شيء!";
	next;
	mes "[إدغار]";
	mes "لكنه في الواقع حسن القلب ويحب مساعدة الآخرين. وسيقدم لك دعمه إن دفعت الثمن الذي يطلبه.";
	next;
	mes "[إدغار]";
	mes "يقول أهل ألبرتا مؤخرا إنه يحاول فعلا مساعدة الناس، وأن اشتراطه إحضار حبات الجيلوبي ليس سوى ذريعة. يجدر بك أن ترى ما يقدمه.";
	next;
	if (select("كيف يمكنني الوصول إلى ألبرتا؟:إنهاء المحادثة") == 1) {
		mes "[إدغار]";
		mes "هاه؟ يمكنك الذهاب سيرا على الأقدام. أما إن كان معك بعض المال، فأنصحك بركوب سفينة.";
		next;
		if (select("حسنا، فهمت.:لكنني سئمت المشي ولا أملك مالا!") == 1) {
			mes "[إدغار]";
			mes "حسنا إذن، اعتن بنفسك~";
			close;
		}
		mes "[إدغار]";
		mes "حسنا...";
		mes "لا تريد المشي، ولا تملك المال، ومع ذلك تريد الوصول إلى هناك؟ يا للعجب...";
		next;
		mes "[إدغار]";
		mes "حسنا، حسنا. بصفتي قبطان سفينة، يمكنني نقلك إلى هناك بسعر مخفض. ما رأيك في 250 زيني؟";
		next;
		if (select("حسنا إذن~!:باه، هذا استغلال!!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[إدغار]";
				mes "همم...";
				mes "هذا المبلغ";
				mes "غير كاف.";
				mes "اذهب واجمع";
				mes "المزيد من المال.";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[إدغار]";
		mes "يا للعجب...";
		mes "إن عددت هذا";
		mes "استغلالا...";
		close;
	}
	mes "[إدغار]";
	mes "حسنا إذن.";
	mes "أراك لاحقا~";
	close;
}
