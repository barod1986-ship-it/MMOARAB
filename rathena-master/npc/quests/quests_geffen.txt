geffen,122,77,4	script	نيا#yagu::Nia#yagu	700,{
	mes "[نيا]";
	mes "مرحبا، أنا نيا، التلميذة";
	mes "الخامسة لماوونغ،";
	mes "العبقري في الألعاب. ما رأيك";
	mes "في أن تلعب معي";
	mes "لعبة مطابقة الأرقام؟";
	next;
	switch(select("ما لعبة مطابقة الأرقام؟:التدرب على اللعبة:بدء اللعبة:صاحب الرقم القياسي:إلغاء")) {
	case 1:
		mes "[نيا]";
		mes "حسنا، الهدف من";
		mes "لعبة مطابقة الأرقام";
		mes "هو تخمين الرقم الذي";
		mes "أفكر فيه. لكن توجد";
		mes "بعض القواعد التي تقيد";
		mes "الأرقام التي أختارها.";
		next;
		mes "[نيا]";
		mes "يجب أن يتكون الرقم الذي";
		mes "أختاره من ثلاث خانات،";
		mes "ولا يجوز أن تكون أي منها 0،";
		mes "كما لا يمكنني تكرار";
		mes "الرقم نفسه في أكثر من";
		mes "خانة واحدة.";
		next;
		mes "[نيا]";
		mes "فمثلا، لا يمكنني";
		mes "اختيار 103 أو 112،";
		mes "أو 252 أو 701؛ فكل منها";
		mes "إما يحتوي على 0، أو يكرر";
		mes "الرقم نفسه في أكثر من";
		mes "خانة. هل فهمت الآن؟";
		next;
		mes "[نيا]";
		mes "لديك خمس محاولات فقط،";
		mes "وبعد كل تخمين سأخبرك";
		mes "بعدد الأرقام الصحيحة،";
		mes "ثم بعدد الأرقام الموجودة";
		mes "في مواضعها الصحيحة.";
		next;
		mes "[نيا]";
		mes "لنفترض مثلا أن";
		mes "الرقم الذي اخترته هو 168.";
		mes "وأن تخمينك الأول كان 678؛";
		mes "عندها يكون الرقمان 6 و8 صحيحين، لكن";
		mes "الرقم 8 وحده في موضعه الصحيح.";
		next;
		mes "[نيا]";
		mes "أنصحك أن تجرب";
		mes "وضع التدريب";
		mes "أولا. وبعد أن تعتاد";
		mes "اللعبة، خض";
		mes "الجولة الحقيقية. ولا تنس أن السرعة تؤثر في نتيجتك!";
		next;
		mes "[نيا]";
		mes "وإذا حققت";
		mes "رقما قياسيا جديدا في";
		mes "لعبة مطابقة الأرقام، فسوف";
		mes "نسجل اسمك،";
		mes "فلا تضيع فرصة";
		mes "نيل الشهرة وربما الثروة أيضا~";
		close;
	case 2:
		mes "[نيا]";
		mes "رائع، لنبدأ جولة";
		mes "تدريبية في مطابقة الأرقام.";
		mes "حاول تخمين رقمي، حسنا؟";
		mes "وللإلغاء، أدخل رقما أكبر من 1,000.";
		set .@practice,1;
		break;
	case 3:
		mes "[نيا]";
		mes "رائع، لنبدأ لعبة مطابقة الأرقام!";
		mes "لديك خمس محاولات لتخمين الرقم الذي أفكر فيه.";
		mes "وللإلغاء، أدخل رقما أكبر من 1,000.";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[نيا]";
		mes "يحمل ^ff0000" + .@name_record$ + "^000000";
		mes "الرقم القياسي في لعبة";
		mes "مطابقة الأرقام بزمن قدره";
		mes "^ff0000" + .@score_min + " دقيقة، " + .@score_sec + " ثانية^000000.";
		close;
	case 5:
		mes "[نيا]";
		mes "اعتن بنفسك،";
		mes "أيها المغامر. وإذا رغبت";
		mes "في لعب مطابقة الأرقام،";
		mes "فتعال وتحدث";
		mes "إلي، حسنا؟ أراك لاحقا~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[نيا]";
				mes "لقد ألغيت";
				mes "لعبة مطابقة الأرقام.";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[نيا]";
			mes "أدخلت الرقم 0";
			mes "في إحدى الخانات، أو كررت";
			mes "القيمة نفسها في";
			mes "أكثر من خانة.";
			mes "أدخل تخمينا آخر.";
			next;
		}

		set .@retry, .@retry+1;
		mes "[نيا]";
		mes "^ff0000"+strcharinfo(0)+"^000000،";
		mes "كان تخمينك ^0000ff"+.@input+"^000000.";
		mes "مهلا قليلا";
		mes "حتى أحسب النتيجة.";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[نيا]";
				mes "نعم، الرقم الذي";
				mes "اخترته كان ^ff0000"+.@yagu+"^000000!";
				mes "تهانينا! لقد";
				mes "فزت في هذه الجولة";
				mes "التدريبية، ويمكنك الآن";
				mes "خوض الجولة الحقيقية~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[نيا]";
					mes "رائع! لقد خمنت";
					mes "رقمي الصحيح، وهو";
					mes "^ff0000" + .@yagu + "^000000. أنهيت الجولة في";
					mes "^ff0000" + .@total_min + " دقيقة، " + .@total_sec + " ثانية^000000،";
					mes "وبذلك حطمت الرقم القياسي السابق";
					mes "البالغ ^ff0000" + .@score_min + " دقيقة، " + .@score_sec + " ثانية^000000.";
					next;
					mes "[نيا]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "كان يحمل الرقم القياسي السابق،";
					mes "أما الرقم القياسي الجديد";
					mes "في لعبة مطابقة الأرقام";
					mes "فقد سجل باسمك. تهانينا";
					mes "على هذا الإنجاز!";
					close;
				}
				mes "[نيا]";
				mes "أحسنت، لقد خمنت";
				mes "رقمي! والإجابة كانت";
				mes "بالطبع ^ff0000" + .@yagu + "^000000. وقد";
				mes "توصلت إلى الرقم الصحيح";
				mes "في ^ff0000" + .@total_min + " دقيقة، " + .@total_sec + " ثانية^000000.";
				next;
				mes "[نيا]";
				mes "يحمل ^ff0000" + .@name_record$ + "^000000";
				mes "الرقم القياسي الحالي";
				mes "في مطابقة الأرقام بزمن";
				mes "قدره ^ff0000" + .@score_min + " دقيقة، " + .@score_sec + " ثانية^000000.";
				mes "حظا موفقا، وآمل أن";
				mes "تحطم هذا الرقم يوما ما~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[نيا]";
				mes "في تخمينك ^ff0000" + .@ball + "^000000 من";
				mes "الأرقام الصحيحة، ويوجد منها ^ff0000" + .@strike + "^000000";
				mes "في";
				mes "مواضعها";
				mes "الصحيحة.";
				mes "حسنا، حاول مجددا~";
				next;
			} else {
				mes "[نيا]";
				mes "في تخمينك الأخير ^ff0000" + .@ball + "^000000 من";
				mes "الأرقام الصحيحة، ويوجد منها";
				mes "^ff0000" + .@strike + "^000000 في";
				mes "مواضعها";
				mes "الصحيحة.";
				switch(.@retry) {
					case 1: mes "بقيت لك أربعة تخمينات."; break;
					case 2: mes "بقيت لك ثلاثة تخمينات."; break;
					case 3: mes "بقي لك تخمينان."; break;
					case 4: mes "بقي لك تخمين واحد فقط..."; break;
					case 5: mes "كانت الإجابة ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
