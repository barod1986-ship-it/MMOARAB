payon_in03,188,146,5	script	ماركس هانسن#juice::Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- انتظر لحظة! -";
		mes "- أنت تحمل الآن -";
		mes "- أغراضا أكثر من اللازم. -";
		mes "- عد إلي لاحقا -";
		mes "- بعد وضع بعض الأغراض في مخزن كافرا. -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[التاجر ماركس هانسن]";
		mes "أهلا بك.";
		mes "هل جئت أنت أيضا";
		mes "لتحويل الفاكهة إلى عصير؟";
		next;
		switch(select("صنع عصير.:معلومات عن معالجة الفاكهة.:إلغاء.")) {
		case 1:
			mes "[التاجر ماركس هانسن]";
			mes "أي نوع من عصير الفاكهة تريد صنعه؟";
			next;
			switch(select("عصير التفاح:عصير الموز:عصير الجزر:عصير العنب:إلغاء")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[التاجر ماركس هانسن]";
				mes "حسنا...";
				mes "أراك لاحقا.";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[التاجر ماركس هانسن]";
				mes "للأسف...";
				mes "ليست لديك كل المواد المطلوبة. لصنع "+getitemname(.@juice)+"، أحتاج إلى 1 "+getitemname(.@fruit)+" و1 زجاجة فارغة، إضافة إلى رسوم قدرها 3 زيني.";
				next;
				mes "[التاجر ماركس هانسن]";
				mes "عندما تحضر كل شيء، سأخلط الفاكهة وأصنع لك عصيرا لذيذا.";
				close;
			}

			mes "[التاجر ماركس هانسن]";
			mes "كم زجاجة تريد؟";
			next;
			switch(select("أقصى كمية ممكنة.:أريد تحديد الكمية.:إلغاء.")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[التاجر ماركس هانسن]";
				mes "اختر رقما لا يتجاوز 100. للإلغاء، أدخل '0'. يمكنك صنع ما يصل إلى " + countitem(.@fruit) + " زجاجة عصير.";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[التاجر ماركس هانسن]";
						mes "حسنا...";
						mes "عد إلي لاحقا.";
						close;
					}
					else if (.@input > 100) {
						mes "[التاجر ماركس هانسن]";
						mes "لا يمكن صنع أكثر من 100 زجاجة. اختر كمية أخرى.";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[التاجر ماركس هانسن]";
				mes "حسنا...";
				mes "عد إلي لاحقا.";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[التاجر ماركس هانسن]";
				mes "للأسف...";
				mes "لا تملك كل المواد المطلوبة، لذلك لا أستطيع مساعدتك. اجمع ما ينقصك ثم عد إلي.";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[التاجر ماركس هانسن]";
			mes "تفضل! عصير طازج ولذيذ كما وعدتك. سيكون منعشا وشهيا جدا.";
			next;
			mes "[التاجر ماركس هانسن]";
			mes "حسنا...";
			mes "عد إلي لاحقا.";
			close;
		case 2:
			mes "[التاجر ماركس هانسن]";
			mes "قبل أن يبني البشر مجتمعاتهم الواسعة، كانوا يقطفون الفاكهة من الأشجار ليعيشوا. ولعل الفاكهة كانت نعمة من الطبيعة مكنتنا من البقاء في هذا العالم.";
			next;
			mes "[التاجر ماركس هانسن]";
			mes "ومع ازدهار الحياة، يبدو أن الجيل الجديد لم يعد يقبل على الفاكهة. لذلك بدأت أبحث عن طريقة تجعل تناولها أسهل.";
			next;
			mes "[التاجر ماركس هانسن]";
			mes "ثم أدركت أن تحويل الفاكهة إلى عصير يجعل تناولها أسهل وطعمها ألذ.";
			next;
			mes "# معلومات عصائر الفاكهة #";
			mes "^CC4E5C- عصير التفاح -^000000";
			mes "تفاحة x 1، زجاجة فارغة x 1، 3 زيني.";
			mes "^E3CF57- عصير الموز -^000000";
			mes "موز x 1، زجاجة فارغة x 1، 3 زيني.";
			mes "^ED9121- عصير الجزر -^000000";
			mes "جزر x 1، زجاجة فارغة x 1، 3 زيني.";
			mes "^CC00FF- عصير العنب -^000000";
			mes "عنب x 1، زجاجة فارغة x 1، 3 زيني.";
			close;
		case 3:
			mes "[التاجر ماركس هانسن]";
			mes "مهلا!";
			mes "عند زيارة أحد،";
			mes "لا بد من التحدث إليه!";
			close;
		}
	}
	else {
		mes "[التاجر ماركس هانسن]";
		mes "أهلا بك.";
		mes "كيف أستطيع مساعدتك؟";
		next;
		if (select("تحدث:إلغاء") == 1) {
			mes "[التاجر ماركس هانسن]";
			mes "قبل أن يبني البشر مجتمعاتهم الواسعة، كانوا يقطفون الفاكهة من الأشجار ليعيشوا. ولعل الفاكهة كانت نعمة من الطبيعة مكنتنا من البقاء في هذا العالم.";
			next;
			mes "[التاجر ماركس هانسن]";
			mes "ومع ازدهار الحياة، يبدو أن الجيل الجديد لم يعد يقبل على الفاكهة. لذلك بدأت أبحث عن طريقة تجعل تناولها أسهل.";
			next;
			mes "[التاجر ماركس هانسن]";
			mes "ثم أدركت أن تحويل الفاكهة إلى عصير يجعل تناولها أسهل وطعمها ألذ.";
			close;
		}
		mes "[التاجر ماركس هانسن]";
		mes "مهلا!";
		mes "عند زيارة أحد،";
		mes "لا بد من التحدث إليه!";
		close;
