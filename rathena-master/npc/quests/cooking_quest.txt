S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[مادلين تشو]";
	mes "كم عدد "+ (.@item_id == 12125 ? "عدة الطهي الخارجية":"عدة الطهي المنزلية");
	mes "تريد";
	mes "شراءها؟ وللإلغاء";
	mes "أدخل";
	mes "الرقم 0.";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[مادلين تشو]";
			mes "غيرت رأيك؟";
			mes "إن احتجت لاحقا";
			mes "إلى شراء "+ getitemname(.@item_id);
			mes "فعد";
			mes "إلي في أي وقت.";
			close;
		}
		else if (.@sell > 100) {
			mes "[مادلين تشو]";
			mes "آسفة، لكنني";
			mes "لا أبيع أكثر من";
			mes "100 "+ getitemname(.@item_id);
			mes "في المرة الواحدة.";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[مادلين تشو]";
		mes "آسفة، لكنك";
		mes "لا تملك ثمن هذا العدد من";
		mes getitemname(.@item_id);
		mes "تحقق من رصيدك";
		mes "قبل شراء الأدوات~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[مادلين تشو]";
		mes "آسفة، لكن ليست لديك";
		mes "مساحة كافية في";
		mes "المخزون لهذا العدد من";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[مادلين تشو]";
	mes "تفضل~";
	mes "بالتوفيق في";
	mes "تدريبك على الطهي!";
	return;
}
