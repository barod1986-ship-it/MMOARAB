		mes "^3355FFتوقف مكانك!";
		mes "أنت تحمل عناصر كثيرة ولا تملك مساحة كافية في الحقيبة لاستلام المكافأة. أفرغ بعض المساحة ثم عد.";
		close;
	}
	emotion ET_HUK, playerattached();
	emotion ET_HUK;
	mes "[الدجاجة ييل]";
	mes "كوك-كوووك؟";
	mes "كوك كوك كوك!";
	next;
	if (rand(1,3) == 2) {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FFما إن شعرت";
		mes "بوجودك حتى ركضت";
		mes "الدجاجة مبتعدة بسرعة.^000000";
		if (qskill_bard > 1 && qskill_bard < 7) {
			mes "^3355FFوجدت ^333333بيضة واحدة^3355FF في";
			mes "المكان الذي كانت";
			mes "ييل تعشش فيه.^000000";
			set qskill_bard,qskill_bard+1;
			getitem 574,1; //Egg
		}
	}
	else {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FFما إن شعرت";

}
