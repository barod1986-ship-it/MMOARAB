		delitem 1088,1; // Morocc Potion
		set SAG_SK,100;
		skill .@Skill[.@i],1,0;
		next;
		mes "[ميشونا]";
		mes strcharinfo(0)+"...";
		mes "يسعدني أنك حفظت";
		mes "مهارة تغيير العنصر إلى";
		mes .@Skill$[.@i]+" بنجاح.";
		mes "آمل أن تفيدك هذه المهارة";
		mes "في القتال. إلى اللقاء~";
		close;
	}
	else {
		mes "[ميشونا]";
		mes "إن كان لديك أصدقاء";
		mes "من الحكماء أو العلماء لم";
		mes "يتعلموا المهارات من لفائف";
		mes "السيد بارموندت،";
		mes "فأرجو أن تدلهم علي.";
		next;
		mes "[ميشونا]";
		mes "أثق أنك وجدت";
		mes "أن هذه المهارات";
		mes "المعتمدة على العناصر";
		mes "مفيدة جدا في القتال.";
		mes "فالمعرفة تتحول حقا";
		mes "إلى قوة مع مرور الوقت...";
		close;
	}
