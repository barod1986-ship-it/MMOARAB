			mes "تتذكر طريقة استخدامها الآن؛";
			mes "مجرد رؤيتي";
			mes "كفيل بإنعاش الذكريات اللازمة";
			mes "لاستخدام المهارة مجددا~";
			if (getskilllv("BS_GREED") == 0)
				skill "BS_GREED",1,SKILL_PERM;
			BLACK_SK2 = 3;
			close;
		}
		if (BLACK_SK2 == 3) {
			mes "[غودمان]";
			mes "كيف تتقدم مهارتك";
			mes "في الحدادة؟";
			mes "عندما ترهقك الحرفة،";
			mes "يمنحك السفر فرصة جيدة";
			mes "للاسترخاء والتفكير";
			mes "وجمع المواد.";
			if (getskilllv("BS_GREED") == 0)	// debug
				skill "BS_GREED",1,SKILL_PERM;
			close;
		}
	}
	if (BLACK_SK2 == 0) {
		mes "[غودمان]";
		mes "مرحبا. أنا معلم مهارات";
		mes "للحدادين، أرسلتني";
		mes "نقابة الحدادين. يدعوني";
		mes "الجميع غودمان. وأنت";
		if (Upper == 1)
			mes "من فئة سيد الحدادين، صحيح؟";
		else
			mes "حداد، صحيح؟";
		mes "هاه، إنه سؤال لا يحتاج إلى جواب.";
		next;
		mes "[غودمان]";
		mes "لدي خبر جيد. أنا مسؤول عن";
		mes "تعليم مهارة طورتها";
		mes "نقابتا الحدادين والتجار،";
		mes "وتسمى ^FF0000جشع^000000. القرار";
		mes "لك في تعلمها،";
		mes "لكنني أنصحك بذلك.";
		next;
		mes "[غودمان]";
		mes "يمكنني، إن رغبت،";
		mes "أن أبدأ تعليمك مهارة";
		mes "جشع حالا. فما";
		mes "رأيك؟";
		next;
		if (select("فكرة جيدة!:ربما لاحقا~") == 1) {
			mes "[غودمان]";
			mes "هيه! اختيار موفق.";
			mes "تلتقط مهارة جشع تلقائيا";
			mes "كل العناصر ضمن مساحة";
			mes "5×5 خلايا حول المستخدم.";
			mes "إنها مفيدة لجمع أكوام العناصر المتناثرة دفعة واحدة.";
			next;
			mes "[غودمان]";
			mes "أولا، يجب أن";
			mes "أختبر مدى أهليتك";
			mes "لتعلم المهارة. املأ حقيبتك بأي عناصر حتى";
			mes "لا تبقى فيها مساحة";
			mes "لعنصر وزنه 500.";
			next;
			mes "[غودمان]";
			mes "بعد إنهاء";
			mes "المهمة، عد";
			mes "وتحدث إلي. حسنا،";
			mes "حظا موفقا~";
			BLACK_SK2 = 1;
			close;
		}
		mes "[غودمان]";
		mes "فهمت. لا بأس بأن";
		mes "تنشغل بأمور أخرى،";
		mes "لكن تذكر أن الوقت";
		mes "ثروة، وعليك";
		mes "أن تنفقه دائما فيما";
		mes "يستحق. مفهوم؟";
		close;
	}
	if (BLACK_SK2 == 1) {
		if (MaxWeight - Weight >= 5000) {
			mes "[غودمان]";
			mes "همم. عدت";
			mes "مبكرا، ولم";
			mes "تكمل الهدف الذي";
			mes "حددته لك. استمع جيدا.";
			next;
			mes "[غودمان]";
			mes "املأ حقيبتك";
			mes "بأي عناصر حتى";
			mes "لا تبقى فيها مساحة";
			mes "لعنصر وزنه 500.";
			mes "بعد إنهاء المهمة،";
			mes "عد وأبلغني.";
			close;
		}
		mes "[غودمان]";
		mes "عدت بهذه السرعة؟";
		mes "حسنا، دعني أفحص";
		mes "ما تحمله.";
		mes "لنر... الحقيبة، الحقيبة...";
		next;
		mes "[غودمان]";
		mes "أحسنت، لقد نجحت.";
		mes "هل أدركت الآن أن";
		mes "الوقت والخبرة المبذولين في";
		mes "جمع هذه العناصر أهم";
		mes "من أسعارها بالزيني؟";
		next;
		mes "[غودمان]";
		if (Upper == 1)
			mes "أؤمن بأن أسياد الحدادين";
		else
			mes "أؤمن بأن الحدادين";
		mes "يجب أن يكونوا قبل كل شيء";
		mes "حرفيين. أما الثراء والشهرة";
		mes "فمجرد نتيجة جانبية.";
		if (Upper == 1)
			mes "أسياد الحدادين الشرفاء يعرفون أن القيمة الحقيقية في جودة الصنعة...";
		else
			mes "الحدادون الشرفاء يعرفون أن القيمة الحقيقية في جودة الصنعة...";
		next;
		if (Upper == 1) {
			mes "[غودمان]";
			mes "إذا لم تبالغ في";
			mes "تقدير قيمة العنصر";
			mes "بالزيني، فسيصبح";
			mes "التخلي عنه";
			mes "أسهل بكثير. راقبني.";
			mes "أرأيت؟ يمكنك فعل ذلك أيضا.";
			next;
		}
		mes "[غودمان]";
		mes "المهارة التي تعلمتها";
		mes "تدعى جشع، لكنني آمل";
		mes "أن تستخدمها بأسلوب";
		mes "يخالف معنى اسمها.";
		mes "لا تنس ذلك، وآمل";
		mes "أن تصنع روائع حقيقية.";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[غودمان]";
	mes "كيف تستخدم";
	mes "مهارة جشع؟ آمل";
	mes "أن تكون مفيدة";
	mes "في جهودك لصنع معدات";
	mes "أعلى جودة";
	mes "لرفاقك جميعا.";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
