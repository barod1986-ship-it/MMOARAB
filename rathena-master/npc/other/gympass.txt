payon,173,141,4	script	كابوس مفتول العضلات#GymPass::Ripped Cabus#GymPass	899,{
	mes "[مفتول العضلات]";
	mes "مرحبا. لم يعد الناس";
	mes "يمارسون نشاطا بدنيا كما";
	mes "في السابق. حتى مع القتال";
	mes "لكسب العيش، قد يبقى الجسد";
	mes "ضعيفا ومترهلا في بعض";
	mes "المواضع. هل المعنى واضح؟";
	next;
	mes "[مفتول العضلات]";
	mes "تدرب معي، وأضمن";
	mes "أن تزداد قدرتك على رفع";
	mes "المزيد من أغراضك وحملها.";
	mes "كل ما أحتاجه هو ^FF0000بطاقة النادي الرياضي^000000";
	mes "في كل مرة، وعندها يمكننا";
	mes "بدء التدريب.";
	next;
	mes "[مفتول العضلات]";
	mes "لكن لا داعي للحماس الزائد:";
	mes "فمهما كان مقدار التدريب";
	mes "الذي سنخوضه، تظل هناك";
	mes "خطورة من الإفراط. هل سمعت";
	mes "عن شخص صار مفتول العضلات";
	mes "أكثر من اللازم؟";
	mes "لم تسمع، لأنهم ماتوا. واضح؟";
	next;
	mes "[مفتول العضلات]";
	mes "أرى أن التدريب الجاد معي";
	mes "آمن، ويمكنه زيادة";
	mes "سعة حمل العناصر لديك";
	mes "حتى ^FF000010 مرات^000000.";
	mes "إذن... هل نبدأ التمرين؟";
	next;
	switch(select("نعم:لا:همم، اختفى أثر تدريبي.")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[مفتول العضلات]";
				mes "رائع، أرى أن";
				mes "بطاقة النادي الرياضي معك.";
				mes "حسنا، اتبع حركاتي،";
				mes "وركز على حرارة التمرين.";
				mes "هيا، لنبدأ.";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[مفتول العضلات]";
				mes "انتهينا. يفترض الآن";
				mes "أن تزداد قدرتك على حمل الأغراض.";
				mes "لنر... ما زال بإمكاننا";
				mes "زيادة سعة حمل العناصر";
				mes "^FF00000" + .@remain_carry + "^000000 مرات أخرى إذا واصلنا";
				mes "بالتدريب معا بهذه الطريقة.";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[مفتول العضلات]";
				mes "اسمع، ماذا قلت؟";
				mes "لا بد من إحضار";
				mes "^FF0000بطاقة النادي الرياضي^000000 لبدء";
				mes "التمرين وتقوية عضلات";
				mes "حمل العناصر لديك.";
				close;
			}
		}
		else {
			mes "[مفتول العضلات]";
			mes "لا أظن أننا نستطيع";
			mes "تقوية عضلات حمل العناصر";
			mes "أكثر من ذلك.";
			mes "حتى محاولة ذلك ستكون";
			mes "خطيرة جدا على الجسد!";
			mes "لقد شرحت حدود التدريب.";
			close;
		}
	case 2:
		mes "[مفتول العضلات]";
		mes "آه، هذا مؤسف.";
		mes "يمكن العودة حين";
		mes "يتغير رأيك. وانشر";
		mes "الخبر بين أصدقائك:";
		mes "من يحتاج منهم إلى اللياقة";
		mes "سأساعده على استعادتها.";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[مفتول العضلات]";
			mes "ماذا حدث؟";
			mes "هل ضعفت عضلات حمل العناصر";
			mes "وضمرت؟ من حسن الحظ";
			mes "أن هناك ما يسمى";
			mes "ذاكرة العضلات. لن يستغرق";
			mes "بناؤها من جديد وقتا طويلا...";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[مفتول العضلات]";
			mes "ما رأيك؟";
			mes "عادت عضلات";
			mes "حمل العناصر إلى قوتها،";
			mes "وبهذه البساطة! لا تدعها";
			mes "تضعف مرة أخرى، اتفقنا؟";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[مفتول العضلات]";
			mes "همم...";
			mes "لم نتدرب";
			mes "معا من قبل.";
			mes "أنا متأكد من ذلك.";
			close;
		}
	}
}
