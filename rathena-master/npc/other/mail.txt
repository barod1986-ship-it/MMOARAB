// Floating mailbox npc that all NPCs duplicate from.
//============================================================
-	script	صندوق البريد#dummy::MailBox	-1,{
	mes "[صندوق البريد]";
	mes "لاستخدام خدمة صندوق البريد،";
	mes "يجب عليك دفع 130 زيني.";
	mes "هل ترغب في استخدام الخدمة؟";
	next;
	switch(select("نعم.:لا.")) {
	case 1:
		mes "[صندوق البريد]";
		if (Zeny < 130) {
			mes "أنا آسف، لكنك لا تملك مالا كافيا.";
			mes "لاستخدام خدمة صندوق البريد،";
			mes "يجب عليك دفع 130 زيني.";
			close;
		}
		mes "شكرا لك، يرجى العودة مرة أخرى.";
		set Zeny, Zeny - 130;
		close2;
		openmail;
		end;
	case 2:
		mes "[صندوق البريد]";
		mes "شكرا لك، يرجى العودة مرة أخرى.";
		close;
	}
}
