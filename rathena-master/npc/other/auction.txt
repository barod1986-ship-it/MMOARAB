function	script	F_AuctionWarper	{
	mes "[مرشد قاعة المزاد]";
	mes "مرحبا، هل تود";
	mes "دخول";
	mes "قاعة المزاد؟";
	next;
	if (select("نعم:لا") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "[مرشد قاعة المزاد]";
			mes "رائع! تفضل،";
			mes "أتمنى لك وقتا ممتعا";
			mes "في المزاد~";
		} else {
			mes "[مرشد قاعة المزاد]";
			mes "استمتع بوقتك في المزاد.";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "[مرشد قاعة المزاد]";
	mes "حسنا،";
	mes "أراك لاحقا.";
	mes "إن غيرت رأيك،";
	mes "فعد إلينا واستمتع";
	mes "بالمزادات~";
	close;
}
