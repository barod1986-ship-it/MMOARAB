// callfunc("F_GM_NPC"{,<password>,<0:int | 1:str>{,<min>,<max>}})
function	script	F_GM_NPC	{

	// لتعطيل جميع شخصيات إدارة GM، أزل التعليق عن السطر أدناه.
	//end;

	// لتحديد أدنى مستوى GM مسموح له باستخدام هذه الشخصيات، عدّل السطر أدناه.
	if (getgmlevel() < 99) {
		// سجّل المحاولة.
		getmapxy(.@map$,.@x,.@y,BL_NPC);
		logmes strcharinfo(0)+" حاول الوصول إلى شخصية إدارة GM "+strnpcinfo(0)+" ("+.@map$+","+.@x+","+.@y+").";
		end;
	}

	if (!getargcount()) return;

	// لتعطيل إدخال كلمات المرور، أزل التعليق عن السطر أدناه.
	//return 1;

	// قيم الإرجاع لفحص كلمة المرور:
	//  1: الإدخال صحيح
	//  0: الإدخال غير صحيح
	// -1: صفر/إلغاء (للأرقام فقط)
	// -2: خارج النطاق (للأرقام فقط)
	if (getarg(1) == 0) {
		// كلمة مرور رقمية
		if (getarg(3,0)) {
			.@error = input(.@input,getarg(2),getarg(3));
			if (.@error)
				return -2;
		} else
			input .@input;
		if (.@input == 0)
			return -1;
		else
			return (.@input == getarg(0))?1:0;
	} else {
		// كلمة مرور نصية
		input .@inputstr$;
		return (.@inputstr$ == getarg(0))?1:0;
	}
}
