-	script	تاجر المرتزقة#dummy::MercMerchant	-1,{
	mes "[تاجر مستلزمات المرتزقة]";
	mes "أهلا بك. أبيع مستلزمات";
	mes "مخصصة لاستخدام المرتزقة.";
	mes "هل تبحث عن";
	mes "شيء معين؟";
	mes "اختر ما تحتاج إليه.";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[تاجر مستلزمات المرتزقة]";
	mes getitemname(.@item[.@m]);
	mes "سعر القطعة الواحدة " + .@str_cost$ + " زيني.";
	mes "ما الكمية التي تريد شراءها؟";
	next;
	input .@input;
	if (!.@input) {
		mes "[تاجر مستلزمات المرتزقة]";
		mes "هل غيرت رأيك؟";
		mes "لا بأس. عد إلي";
		mes "متى احتجت إلى شراء";
		mes "مستلزمات لمرتزقتك.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[تاجر مستلزمات المرتزقة]";
		mes "عذرا، لا يمكنك شراء";
		mes "أكثر من 10,000 قطعة";
		mes "في العملية الواحدة.";
		mes "أدخل رقما صحيحا";
		mes "من 1 إلى 10,000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[تاجر مستلزمات المرتزقة]";
		mes "عذرا، لا تملك ما يكفي";
		mes "من الزيني لشراء";
		mes "هذه الكمية. سأبقى";
		mes "هنا حتى تستعد";
		mes "لشراء ما يحتاج إليه";
		mes "مرتزقك.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[تاجر مستلزمات المرتزقة]";
		mes "لن تتمكن من حمل";
		mes "هذه الكمية من الأغراض.";
		mes "أفرغ بعض المساحة";
		mes "في مخزونك، ثم";
		mes "عد لإتمام الشراء.";
		mes "لن أحملك ما يفوق قدرتك.";
		close;
	}
	mes "[تاجر مستلزمات المرتزقة]";
	if (.@input == 1) 
		mes "تفضل: " + getitemname(.@item[.@m]) + ".";
	else {
		mes "تفضل، هذه الكمية المطلوبة:";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "شكرا لك. عد متى";
	mes "احتجت إلى مزيد من";
	mes "مستلزمات المرتزقة.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[الفاحص]";
	mes "أدخل كلمة المرور.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[الفاحص]";
	if (.@i == -2)
		mes "القيمة خارج النطاق.";
	else if (.@i == 0)
		mes "الرقم غير صحيح.";
	else if (.@i == 1) {
		mes "اختر الإجراء المطلوب.";
		next;
		switch( select( "تعطيل شخصيات المرتزقة","تفعيل شخصيات المرتزقة" ) ) {
		case 1:
			mes "تم تعطيل شخصيات المرتزقة.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "تم تفعيل شخصيات المرتزقة.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
