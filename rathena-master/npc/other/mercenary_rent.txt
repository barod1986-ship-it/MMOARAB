-	script	تاجر المرتزقة#dummy::MercMerchant	-1,{
	mes "[تاجر بضائع المرتزقة]";
	mes "مرحبا، أنا أبيع بضائع";
	mes "يمكن للمرتزقة";
	mes "استخدامها. هل هناك شيء";
	mes "معين";
	mes "تبحث عنه؟";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[تاجر بضائع المرتزقة]";
	mes getitemname(.@item[.@m]);
	mes "تكلفة كل واحد منها " + .@str_cost$ + " زيني.";
	mes "كم واحدا تريد؟";
	next;
	input .@input;
	if (!.@input) {
		mes "[تاجر بضائع المرتزقة]";
		mes "هل غيرت رأيك؟";
		mes "حسنا، لا تتردد في";
		mes "العودة إلي متى أردت";
		mes "شراء أي جرعات للمرتزقة.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[تاجر بضائع المرتزقة]";
		mes "أنا آسف، لكن يمكنك";
		mes "شراء ما يصل إلى";
		mes "10,000 من هذه الجرعات";
		mes "في المرة الواحدة فقط. يرجى إدخال";
		mes "رقم من 1 إلى 10,000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[تاجر بضائع المرتزقة]";
		mes "أنا آسف، لكنك لا تملك";
		mes "ما يكفي من الزيني لشراء";
		mes "هذا العدد من الجرعات. حسنا، سأكون";
		mes "هنا عندما تكون مستعدا";
		mes "لشراء شيء";
		mes "لمرتزقتك.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[تاجر بضائع المرتزقة]";
		mes "إذا أعطيتك هذا العدد";
		mes "من الجرعات، فلن تكون";
		mes "قادرا على حملها معك.";
		mes "يرجى العودة بعد";
		mes "أن تفرغ بعض المساحة";
		mes "في مخزونك.";
		close;
	}
	mes "[تاجر بضائع المرتزقة]";
	if (.@input == 1) 
		mes "تفضل " + getitemname(.@item[.@m]) + " الخاص بك.";
	else {
		mes "تفضل، هذا بالضبط";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "شكرا لك، ويرجى العودة";
	mes "مرة أخرى عندما تحتاج إلى المزيد";
	mes "من الجرعات لمرتزقتك.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[الفاحص]";
	mes "يرجى إدخال كلمة المرور.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[الفاحص]";
	if (.@i == -2)
		mes "خطأ.";
	else if (.@i == 0)
		mes "رقم خاطئ";
	else if (.@i == 1) {
		mes "يرجى الاختيار.";
		next;
		switch( select( "إيقاف NPC المرتزقة","تشغيل NPC المرتزقة" ) ) {
		case 1:
			mes "تم إيقاف NPCs.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "تم تشغيل NPCs.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
