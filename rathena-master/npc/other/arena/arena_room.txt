function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[غيفو]";
	mes "هل تريد استبدال نقاط الحلبة بـ "+ getitemname(.@item_id) +"؟";
	mes "تكلف الدفعة ^3131FF"+ .@arena_point +" نقطة حلبة مقابل "+ .@amount +" من "+ getitemname(.@item_id) +"^000000.";
	mes "أدخل 0 للإلغاء، أو أدخل عدد ^3131FFالدفعات^000000 المطلوب استبدالها.";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[غيفو]";
		mes "تم إلغاء الطلب.";
		close;
	} else if (.@reward > 1500) {
		mes "[غيفو]";
		mes "تجاوز العدد الحد الأقصى المسموح به.";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[غيفو]";
		mes "لا تملك نقاط حلبة كافية.";
		mes "تحقق من رصيد نقاط الحلبة ثم حاول مجددا.";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[غيفو]";
	mes "شكرا لك. نراك مجددا.";
	close;
}
