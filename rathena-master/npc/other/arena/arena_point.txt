prt_are_in,103,11,3	script	مسؤول نقاط الحلبة::Arena Point Manager	732,{
	mes "[مسؤول نقاط الحلبة]";
	mes "آمل أنك تستمتع بوقتك";
	mes "في الحلبة. إذا ربحت نقاطا";
	mes "من مضمار التيربو بألديباران،";
	mes "فيمكنني تحويلها إلى";
	mes "نقاط حلبة.";
	next;
	switch(select("عرض الرصيد:تحويل النقاط:^660000معلومات عن التحويل^000000")) {
	case 1:
		mes "[مسؤول نقاط الحلبة]";
		mes "" + strcharinfo(0) + "،";
		mes "رصيدك الحالي:";
		mes "" + arena_point + " نقطة حلبة";
		mes "و" + tt_point + " نقطة من مضمار التيربو.";
		close;
	case 2:
		mes "[مسؤول نقاط الحلبة]";
		mes "" + strcharinfo(0) + "،";
		mes "رصيدك الحالي:";
		mes "" + arena_point + " نقطة حلبة";
		mes "و" + tt_point + " نقطة من مضمار التيربو.";
		next;
		mes "[مسؤول نقاط الحلبة]";
		mes "اختر طريقة تحويل";
		mes "نقاط مضمار التيربو إلى";
		mes "نقاط حلبة. عند تحويل أكثر من";
		mes "10 نقاط دفعة واحدة، يجب أن يكون";
		mes "العدد من ^4D4DFFمضاعفات 10^000000.";
		next;
		switch(select("نقطتا مضمار -> نقطة حلبة:4 نقاط مضمار -> نقطتا حلبة:6 نقاط مضمار -> 3 نقاط حلبة:8 نقاط مضمار -> 4 نقاط حلبة:10 نقاط مضمار فأكثر:إلغاء")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[مسؤول نقاط الحلبة]";
			mes "أدخل عدد مرات تحويل";
			mes "10 نقاط من مضمار التيربو";
			mes "إلى 5 نقاط حلبة.";
			mes "يمكن تنفيذ التحويل حتى";
			mes "20 مرة كحد أقصى.";
			mes "أدخل ''^3355FF0^000000'' للإلغاء.";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[مسؤول نقاط الحلبة]";
				mes "تم إلغاء";
				mes "خدمة";
				mes "التحويل.";
				close;
			}
			else if (.@input > 20) {
				mes "[مسؤول نقاط الحلبة]";
				mes "يتجاوز العدد المطلوب";
				mes "الحد الأقصى.";
				mes "أدخل قيمة لا تتجاوز 20.";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[مسؤول نقاط الحلبة]";
					mes "سيؤدي هذا التحويل";
					mes "إلى تجاوز الحد الأقصى";
					mes "لرصيد نقاط الحلبة.";
					mes "لا يمكن أن يتجاوز رصيدك";
					mes "29,000 نقطة حلبة.";
					mes "لن يتم تنفيذ التحويل.";
					next;
					mes "[مسؤول نقاط الحلبة]";
					mes "أنفق بعض نقاط الحلبة";
					mes "ثم أعد محاولة استخدام";
					mes "هذه الخدمة.";
					mes "شكرا لاستخدامك الخدمة.";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[مسؤول نقاط الحلبة]";
					mes "عذرا، لا يوجد في رصيدك";
					mes "ما يكفي من نقاط";
					mes "مضمار التيربو لإجراء";
					mes "عملية تحويل نقاط";
					mes "الحلبة المحددة.";
					close;
				}
				else {
					mes "[مسؤول نقاط الحلبة]";
					mes "تم تحويل";
					mes "10 نقاط من مضمار التيربو";
					mes "إلى 5 نقاط حلبة، بعدد " + .@input + " مرة.";
					mes "إجمالي النقاط المحولة: " + .@want_point1 + " من نقاط المضمار،";
					mes "" + .@want_point +" من نقاط الحلبة.";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[مسؤول نقاط الحلبة]";
					mes "" + strcharinfo(0) + "،";
					mes "رصيدك الآن:";
					mes "^4682B4" + arena_point + "^000000 نقطة حلبة";
					mes "و^00688B" + tt_point + "^000000 نقطة من مضمار التيربو.";
					mes "شكرا لاستخدامك الخدمة.";
					close;
				}
			}
		case 6:
			mes "[مسؤول نقاط الحلبة]";
			mes "تم إلغاء";
			mes "خدمة";
			mes "التحويل.";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[مسؤول نقاط الحلبة]";
		mes "سيؤدي هذا التحويل";
		mes "إلى تجاوز الحد الأقصى";
		mes "لرصيد نقاط الحلبة.";
		mes "لا يمكن أن يتجاوز رصيدك";
		mes "29,000 نقطة حلبة.";
		mes "لن يتم تنفيذ التحويل.";
		next;
		mes "[مسؤول نقاط الحلبة]";
		mes "أنفق بعض نقاط الحلبة";
		mes "ثم أعد محاولة استخدام";
		mes "هذه الخدمة.";
		mes "شكرا لاستخدامك الخدمة.";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[مسؤول نقاط الحلبة]";
		mes "" + strcharinfo(0) + "،";
		mes "تم تحويل "+ .@track_point +" من نقاط مضمار التيربو";
		mes "إلى "+ .@arena_point +" من نقاط الحلبة.";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[مسؤول نقاط الحلبة]";
		mes "" + strcharinfo(0) + "،";
		mes "رصيدك الآن:";
		mes "^4682B4" + arena_point + "^000000 نقطة حلبة";
		mes "و^00688B" + tt_point + "^000000 نقطة من مضمار التيربو.";
		mes "شكرا لاستخدامك الخدمة.";
		close;
	}
	else {
		mes "[مسؤول نقاط الحلبة]";
		mes "عذرا، لا توجد في رصيدك";
		mes "نقاط كافية من مضمار التيربو";
		mes "لإجراء التحويل المحدد.";
		close;
	}
}
