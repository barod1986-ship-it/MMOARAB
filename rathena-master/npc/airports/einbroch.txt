airport,143,43,5	script	موظف المطار#airport1a::airport1	90,{
	mes "[موظف المطار]";
	mes "مرحبا بك في";
	mes "مطار إينبروخ،";
	mes "حيث نوفر رحلات";
	mes "بلا توقف إلى مدن";
	mes "يونو ولايت هالزن وهويغل.";
	next;
	if (select("الصعود إلى المنطاد:إلغاء") == 1) {
		mes "[موظف المطار]";
		mes "تبلغ رسوم ركوب المنطاد";
		mes "1,200 زيني، ولكن إن كنت";
		mes "تملك تذكرة مجانية للمنطاد،";
		mes "فسنعفيك من الرسوم. هل";
		mes "ترغب في ركوب المنطاد؟";
		next;
		if (select("نعم:لا") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[موظف المطار]";
			mes "عذرا، لا تملك";
			mes "تذكرة مجانية للمنطاد،";
			mes "ولا تملك ما يكفي";
			mes "من الزيني لركوب";
			mes "المنطاد. تذكر أن";
			mes "رسوم الركوب هي 1,200 زيني.";
			close;
		}
	}
	mes "[موظف المطار]";
	mes "شكرا لك،";
	mes "ونتمنى لك يوما سعيدا.";
	close;
}
