-	script	::Airship_Staff_izlude	-1,{
	mes "[طاقم المنطاد]";
	mes "مرحبا! هل ترغب";
	mes "في ركوب المنطاد الذي";
	mes "يسلك الخط الدولي ويتوقف";
	mes "في يونو وراشيل؟";
	next;
	if (select("ركوب المنطاد إلى يونو/راشيل:إلغاء") == 1) {
		mes "[طاقم المنطاد]";
		mes "تبلغ رسوم ركوب المنطاد";
		mes "1,200 زيني. وإذا كنت";
		mes "تملك تذكرة مجانية للمنطاد،";
		mes "فسنعفيك من الرسوم. هل";
		mes "ما زلت ترغب في الركوب؟";
		next;
		if (select("نعم:لا") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[طاقم المنطاد]";
			mes "عذرا، لا تملك";
			mes "تذكرة مجانية للمنطاد ولا";
			mes "ما يكفي لدفع رسوم 1,200 زيني.";
			close;
		}
	}
	mes "[طاقم المنطاد]";
	mes "شكرا لك،";
	mes "نرجو أن تزورنا مجددا.";
	mes "وطاب يومك~";
	close;
}
