y_airport,143,43,5	script	موظف المطار#y_air1a::y_airport1	90,{
	mes "[موظف المطار]";
	mes "مرحبا بك في مطار يونو. نوفر رحلات محلية إلى إينبروخ ولايت هالزن وهويغل،";
	mes "ورحلات دولية إلى إزلود وراشيل.";
	mes "كيف يمكنني مساعدتك؟";
	next;
	if (select("ركوب المنطاد:إلغاء") == 1) {
		mes "[موظف المطار]";
		mes "تبلغ رسوم ركوب جميع";
		mes "الرحلات 1,200 زيني. ويمكنك";
		mes "استخدام تذكرة مجانية للمنطاد";
		mes "للإعفاء من الرسوم. هل تريد الركوب؟";
		next;
		if (select("نعم:لا") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[موظف المطار]";
			mes "عذرا، لا تملك تذكرة مجانية";
			mes "للمنطاد ولا تملك 1,200 زيني";
			mes "لدفع رسوم الركوب.";
			close;
		}
	}
	mes "[موظف المطار]";
	mes "شكرا لك،";
	mes "ونتمنى لك يوما سعيدا.";
	close;
}
