		mes "أسرار الحياة التي";
		mes "كانت يوما في متناولك...";
		next;
		if (select("نعم:لا") == 1) {
			mes "[كيميائي حيوي]";
			mes "أغمض عينيك";
			mes "ودع ذهنك يهدأ.";
			mes "سنعود بك إلى ماضيك";
			mes "لنستعيد منه";
			mes "شظايا ذكرياتك";

			mes "التي ترافق استخدام هذه الأسرار";
			mes "للغايات الصحيحة...";

			mes "افتح عينيك...";
			mes "والآن، بعد أن تذكرت";
			mes "كيفية خلق حياة";

			mes "منك سوى أن تعامل جميع";
			mes "مخلوقاتك باحترام.";
			close;
		}
		mes "[كيميائي حيوي]";
		mes "إن رغبت في";
		mes "استعادة ذكرياتك";
		mes "المفقودة، فعد";
		mes "إلي.";
		mes "فسر خلق الحياة";
		mes "ليس أمرا هينا...";
		close;
	}

		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[كيميائي حيوي]";
			mes "تهانينا.";
			mes "المجد للمحاربين.";
			close;
		}
		mes "[كيميائي حيوي]";
		mes "من الغريب أن يكون";
		mes "شخص مثلي هنا.";
		mes "لكن حتى من يتقن";
		mes "دروب العلم يستطيع";
		mes "أن يصبح بطلا.";

		mes "[كيميائي حيوي]";
		mes "في حالتي،";
		mes "لم تكن الوسائل التي";
		mes "استخدمتها هي ما";
		mes "منحني مكانا في فالهالا،";
		mes "بل الغايات التي قاتلت";
		mes "من أجلها...";
		close;

			mes "عد إلي حين تكتمل";
			mes "استعداداتك.";
			mes "المجد للمحاربين.";

		mes "[كيميائي حيوي]";
		mes "أتمنى لك";
		mes "الراحة والطمأنينة";
