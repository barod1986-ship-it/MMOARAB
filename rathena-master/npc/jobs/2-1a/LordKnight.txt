	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[سيد الفرسان]";
			mes "تهانينا.";
			mes "المجد للمحاربين.";
			close;
		}
		mes "[سيد الفرسان]";
		mes "نحن الفرسان نتحمل";
		mes "مسؤولية جسيمة...";
		mes "خدمة الآخرين وحمايتهم.";
		next;
		mes "[سيد الفرسان]";
		mes "حتى لو كلفنا ذلك";
		mes "حياتنا،";
		mes "فعلينا صون";
		mes "أمن رفاقنا وسلامتهم.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[سيد الفرسان]";
		mes "لقد حان وقت ارتقائك!";
		mes "فالعالم ما زال بحاجة إليك.";
		mes "واصل مسيرتك البطولية";
		mes "بهذه الهيئة الجديدة.";
		next;
		mes "[سيد الفرسان]";
		mes "هل ترغب";
		mes "في الارتقاء إلى رتبة سيد الفرسان؟";
		next;
		if (select("لا.:نعم.") == 1) {
			mes "[سيد الفرسان]";
			mes "عد متى اكتمل";
			mes "استعدادك.";
			mes "المجد للمحاربين.";
			close;
		}
		if (SkillPoint) {
			mes "[سيد الفرسان]";
			mes "ما زالت لديك نقاط مهارة غير مستخدمة.";
			mes "استخدمها جميعا قبل العودة إلي.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[سيد الفرسان]";
		mes "تهانينا.";
		mes "الآن وقد أصبحت سيد الفرسان،";
		mes "أتمنى أن تكون دائما";
		mes "في طليعة المعركة،";
		mes "وأن تقود حلفاءك إلى النصر.";
		close;
	}
	mes "[سيد الفرسان]";
	mes "مرحبا بك";
	mes "في فالهالا،";
	mes "قاعة الشرف.";
	next;
	mes "[سيد الفرسان]";
	mes "أتمنى لك";
	mes "الراحة والطمأنينة";
	mes "ما دمت هنا.";
	mes "المجد للمحاربين.";
	close;
}
