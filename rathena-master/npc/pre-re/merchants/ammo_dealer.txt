alberta,118,160,3	script	توني تاجر الرصاص#alb::bdt	4_M_04,{
	mes "[توني]";
	if (BaseClass == Job_Gunslinger) {
		mes "أنا توني، تاجر الرصاص.";
		mes "تعال إلي كلما نفدت";
		mes "ذخيرتك. فقط أحضر لي";
		mes "المواد، وسأصنع لك";
		mes "الرصاص الذي تحتاجه.";
		next;
		switch(select("كرة السم:كرة التوهج:كرة البرق:كرة العمى:كرة التجميد:إلغاء")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[توني]";
			mes "غيرت رأيك؟";
			mes "حسنا، إذا احتجت يوما";
			mes "إلى أي رصاص، سأكون هنا";
			mes "تماما. عد متى";
			mes "ظننت أنك تحتاج إلى مزيد من";
			mes "الذخيرة، يا رامي المسدسات.";
			close;
		}
	}
	mes "مرحبا، أنا توني. أنا";
	mes "مسؤول عن توزيع";
	mes "وصنع الرصاص";
	mes "لرامي المسدسات. هذه فقط";
	mes "الطريقة التي تفضلها نقابتنا";
	mes "في إنجاز الأمور.";
	next;
	mes "[توني]";
	mes "آسف إذا جئت";
	mes "إلى هنا لشراء بعض الرصاص.";
	mes "لا يمكنني التعامل إلا مع";
	mes "رامي مسدسات كامل الخبرة.";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[توني]";
	mes "مقابل كل مجموعة من";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "يجب أن تعطيني";
	mes "1 إمفيريتاركون،";
	mes "1 فراكون، و";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[توني]";
	mes "تذكر أنني أستطيع إعطاء";
	mes "500 مجموعة كحد أقصى من";
	mes "30 رصاصة في كل مرة. يرجى";
	mes "إدخال عدد مجموعات الرصاص";
	mes "التي تريدها. إذا أردت";
	mes "الإلغاء، فأدخل ''0.'' فقط";
	next;
	input .@amount;
	mes "[توني]";
	if (.@amount < 1 || .@amount > 500) {
		mes "مهلا، لا أستطيع أن أعطيك";
		mes "هذا العدد من الرصاص. لا";
		mes "تنس إدخال رقم";
		mes "لا يزيد عن 500";
		mes "إذا أردت مبادلة";
		mes "عناصرك ببعض الرصاص.";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "إيه؟ لا توجد في حقيبتك";
			mes "مساحة كافية لهذا";
			mes "العدد من الرصاص. عد لاحقا";
			mes "بعد أن توفر مساحة";
			mes "أكبر. حاول وضع بعض";
			mes "أغراضك في مخزن كافرا.";
		} else {
			mes "رائع، يبدو أن كل شيء";
			mes "على ما يرام. دعني آخذ";
			mes "هذه المواد، وهذه هي";
			mes "رصاصاتك. يسرني";
			mes "التعامل معك~";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "همم. يبدو أنك";
		mes "لا تملك مواد كافية";
		mes "لهذا العدد من الرصاص. حسنا،";
		mes "لا مشكلة. عد فقط";
		mes "بعد جمع كل ما";
		mes "تحتاج إليه، اتفقنا؟";
	}
	close;
}
