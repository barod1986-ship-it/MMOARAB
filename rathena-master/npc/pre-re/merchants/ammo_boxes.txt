	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[كيني]";
	mes "الآن، يمكنك مبادلة";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "و500 زيني مقابل 1 "+getitemname(.@pack_id)+"، لذا تأكد";
	mes "من أن لديك ما يكفي من الرصاص";
	mes "والزيني لهذه المبادلة.";
	next;
	mes "[كيني]";
	mes "يمكنك مبادلة "+callfunc("F_InsertPlural",500,getitemname(.@bullet_id))+" و500 زيني";
	mes "مقابل 1 "+getitemname(.@pack_id)+".";

	mes "[كيني]";
	mes "تذكر أنني لا أستطيع إعطاءك";
	mes "أكثر من 50 مخزنا";

	input .@amount;
	mes "[كيني]";
	if (.@amount > 50 || .@amount == 0) {
		mes "مهلا، لا أستطيع إعطاءك";
		mes "هذا العدد من المخازن.";
		mes "يرجى المحاولة مرة أخرى،";
		mes "وإدخال رقم لا";
		mes "يزيد عن 50.";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "مهلا، لديك الكثير";
				mes "من الأشياء المحشورة في";
				mes "حقيبتك. وفر بعض";

				mes "ببعض المخازن لاحقا، حسنا؟";

				mes "حسنا، هذه هي";
				mes "مخازنك. شكرا";
				mes "لزيارتك متجري، و";
				mes "آمل أن تستخدم كل";
				mes "ذخيرتك بحكمة.";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "آسف، لكنك لا";
			mes "تملك زيني كافيا";
			mes "لهذه مبادلة المخازن.";
			mes "عد إلى متجري";
			mes "بعد أن توفر";
			mes "بعض المال الإضافي.";
		}
	} else {
		mes "آسف، لكنك لا";
		mes "تملك رصاصا كافيا";

		mes "من حقيبتك مرة أخرى، ثم";
		mes "تعود إلي لاحقا.";
	}
