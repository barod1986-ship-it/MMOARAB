							mes "[مصفف شعر يوفوفيتش]";
							mes "الرسوم هي 1000 زيني. ألا تملك ما يكفي...؟";

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs

						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}

						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[مصفف شعر يوفوفيتش]";
			mes "عندما تشعر بالإحباط، عندما يتم التخلي عنك، عندما تريد إثارة إعجاب شخص ما، أو حتى عندما تريد فقط أن تتميز...";

			mes "لذلك المكان والوقت المميزين، ألا تريد تسريحة شعر خاصة بك؟ طالما لديك الصبغات المناسبة، سأجعل شعرك يبدو رائعا.";

			mes "[مصفف شعر يوفوفيتش]";
			mes "* الرسوم والمعلومات *";
			mes "- عنصر صبغة واحد بلون الشعر الذي تريده.";
			mes "- رسوم 1000 زيني.";
			close;

		case 3:
			mes "[مصفف شعر يوفوفيتش]";
			mes "رجالا أو نساء...";
