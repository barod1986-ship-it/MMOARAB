louyang,213,213,3	script	جندي#BA::LouGuide	825,{
	mes "[جندي]";
	mes "مرحبا بك في لويانغ،";
	mes "مدينة ذات تاريخ طويل";
	mes "وغني بالألوان.";
	next;
	mes "[جندي]";
	mes "لقد طورنا مؤخرا";
	mes "طريقا بحريا لاستيعاب";
	mes "التبادل الإيجابي مع";
	mes "الدول الأجنبية.";
	next;
	mes "[جندي]";
	mes "تشتهر لويانغ بتخصصات متنوعة إضافة إلى تاريخها الغني. يمكنك هنا العثور على الكثير من الأشياء الفريدة الخاصة بأرضنا.";
	next;
	mes "[جندي]";
	mes "خذ وقتك من فضلك";
	mes "وندعوك للاستمتاع";
	mes "برحلتك هنا في لويانغ.";
	next;
	switch(select("السؤال عن مواقع المباني.:إزالة كل العلامات من الخريطة المصغرة.:إلغاء.")) {
	case 1:
		mes "[جندي]";
		mes "إلى أين تود الذهاب؟";
		next;
		switch(select("قلعة التنين:عيادة الطبيب:دار المدينة:متجر الأسلحة:متجر الأدوات:الحانة:إلغاء")) {
		case 1:
			mes "[جندي]";
			mes "تقع قلعة التنين عند ^FF3355+^000000. إنها المكان الذي يقيم فيه كل النبلاء، بمن فيهم سيدنا.";
			next;
			mes "[جندي]";
			mes "بما أنك غريب، أظن أنه من المناسب أن تزور سيدنا أولا.";
			viewpoint 1,218,255,2,0xFF3355;
			break;
		case 2:
			mes "[جندي]";
			mes "لدينا طبيب ماهرة جدا.";
			mes "يمكنك العثور على عيادتها عند ^CE6300+^000000.";
			next;
			mes "[جندي]";
			mes "يقال إنه";
			mes "لا يوجد مرض لا تستطيع علاجه. حسنا، لا أستطيع ضمان إن كان هذا صحيحا أم لا.";
			viewpoint 1,263,94,3,0xCE6300;
			break;
		case 3:
			mes "[جندي]";
			mes "لدينا دار مدينة حيث تعمل الحكومة الاتحادية.";
			mes "تقع عند ^A5BAAD+^000000.";
			next;
			mes "[جندي]";
			mes "إذا كانت لديك أي مشاكل،";
			mes "فعليك التحدث مع";
			mes "الموظفين في دار المدينة.";
			viewpoint 1,309,80,4,0xA5BAAD;
			break;
		case 4:
			mes "[جندي]";
			mes "يقع متجر الأسلحة عند ^55FF33+^000000.";
			next;
			mes "[جندي]";
			mes "سترى";
			mes "أسلحة رائعة صنعها";
			mes "حدادو لويانغ";
			mes "ذوو الخبرة الكبيرة.";
			viewpoint 1,145,174,5,0x55FF33;
			break;
		case 5:
			mes "[جندي]";
			mes "يقع متجر الأدوات عند ^3355FF+^000000.";
			next;
			mes "[جندي]";
			mes "معرفة عدوك";
			mes "نصف المعركة!";
			mes "ومن الآمن أيضا أن تستعد بدل أن تندم لاحقا. لم لا تذهب لتتفقد مؤنهم؟";
			viewpoint 1,135,98,6,0x3355FF;
			break;
		case 6:
			mes "[جندي]";
			mes "عندما تتعب أثناء رحلتك، أقترح أن تزور الحانة. إنها تقع عند ^00FF00+^000000.";
			next;
			mes "[جندي]";
			mes "الحانة مكان جيد";
			mes "لمقابلة سياح آخرين، وكذلك لسماع أي أخبار قد يكون من المفيد معرفتها.";
			viewpoint 1,280,167,7,0x00FF00;
			break;
		case 7:
			mes "[جندي]";
			mes "إذا أردت إزالة كل العلامات";
			mes "من خريطتك المصغرة، يرجى اختيار 'إزالة كل العلامات من الخريطة المصغرة.' من القائمة.";
		}
		break;
	case 2:
		viewpoint 2,218,255,2,0xFF00FF;
		viewpoint 2,263,94,3,0xFF00FF;
		viewpoint 2,309,80,4,0xFF00FF;
		viewpoint 2,145,174,5,0xFF00FF;
		viewpoint 2,135,98,6,0xFF00FF;
		viewpoint 2,280,167,7,0xFF00FF;
		mes "[جندي]";
		mes "ها قد محوت كل العلامات من خريطتك المصغرة. لا تتردد في سؤالي عن مواقع المباني متى احتجت إلى ذلك.";
		break;
	case 3:
		mes "[جندي]";
		mes "أظن أن الاستكشاف";
		mes "بمفردك يكون ممتعا";
		mes "أحيانا. اعتن بنفسك.";
	}
	close;
}
