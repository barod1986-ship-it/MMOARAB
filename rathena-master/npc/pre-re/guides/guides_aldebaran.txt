aldebaran,139,63,4	script	جندي#alde::Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[حارس ألديباران]";
	mes "أنا مجرد حارس عادي";
	mes "يمكنك أن تجده في أي مدينة أخرى. لا أظن أن لدي اسما حتى...";
	next;
	mes "[حارس ألديباران]";
	mes "أنا مسؤول عن مرشدي الخدمة من حامية ألديباران. دعني أرشدك";
	mes "في أنحاء بلدتنا!";
	next;
	if (select("الحصول على دليل المواقع.:إنهاء المحادثة.") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> مكتب كافرا الرئيسي ";
		mes "^0000FF+^000000 -> متجر الأسلحة ";
		mes "^00FFFF+^000000 -> نقابة السحرة (مغلقة)";
		mes "^515151+^000000 -> الحانة";
		mes "^3355FF+^000000 -> متجر العناصر";
		mes "^FF5555+^000000 -> نقابة الخيميائيين";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[حارس ألديباران]";
	mes "لقد أقسمنا على حماية ألديباران! لعل قوى الشر تسحق دائما بواسطة";
	mes "قبضة الخير الصالحة!";
	close2;
	cutin "prt_soldier",255;
	end;
}

aldebaran,133,108,4	script	جندي#2alde::Soldier#2alde	105,{
	cutin "prt_soldier",2;
	mes "[حارس ألديباران]";
	mes "أنا مجرد";
	mes "حارس عادي،";
	mes "من النوع الذي يمكنك";
	mes "أن تجده في أي مدينة أخرى.";
	next;
	mes "[حارس ألديباران]";
	mes "عندما لا أكون مشغولا جدا";
	mes "بحماية سكان";
	mes "ألديباران، أكون هنا لأرشد المغامرين";
	mes "أمثالك.";
	next;
	switch(select("مكتب كافرا الرئيسي :متجر الأسلحة :نقابة السحرة :الحانة :متجر العناصر :نقابة الخيميائيين :إنهاء المحادثة ")) {
	case 1:
		viewpoint 1,61,229,0,0xFF6633;
		mes "^FF6633+^000000 -> مكتب كافرا الرئيسي ";
		break;
	case 2:
		viewpoint 1,72,197,1,0x0000FF;
		mes "^0000FF+^000000 -> متجر الأسلحة ";
		break;
	case 3:
		viewpoint 1,223,222,2,0x00FFFF;
		mes "^00FFFF+^000000 -> نقابة السحرة (مغلقة)";
		break;
	case 4:
		viewpoint 1,233,105,3,0x515151;
		mes "^515151+^000000 -> الحانة ";
		break;
	case 5:
		viewpoint 1,197,70,4,0x3355FF;
		mes "^3355FF+^000000 -> متجر العناصر ";
		break;
	case 6:
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF5555+^000000 -> نقابة الخيميائيين";
		break;
	case 7:
		mes "[حارس ألديباران]";
		mes "لقد أقسمنا على";
		mes "حماية ألديباران!";
		mes "لعل قوى الخير";
		mes "تنتصر دائما على الشر~";
		break;
	}
	close2;
	cutin "prt_soldier",255;
	end;
}
