rachel,138,146,5	script	دليل راشيل::Rachel Guide	934,{
	mes "[دليل راشيل]";
	mes "مرحبا بك في عاصمة";
	mes "أرونافلتز، راشيل، حيث يصل";
	mes "النفس الدافئ للإلهة فريا.";
	mes "إن كانت هذه أول مرة لك";
	mes "في استخدام خدمات المرشد، فلم";
	mes "لا تتحقق من قائمة ملاحظات أولا؟";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("دليل القرية:إزالة علامات الخريطة المصغرة:ملاحظات:إلغاء")) {
		case 1:
			mes "[دليل راشيل]";
			mes "يمكنني إخبارك بموقع أي مبنى ما دام داخل راشيل.";
			mes "إلى أين تريد الذهاب؟";
			if (.@compass_check == 0) {
				mes "هل تريد مني";
				mes "أن أحدد المواقع";
				mes "على خريطتك المصغرة؟";
				next;
				if (select("نعم:لا") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("تشيسهرومنير:النزل:متجر الأسلحة:متجر الأدوات:المطار:إلغاء")) {
				case 1:
					mes "[دليل راشيل]";
					mes "تشيسهرومنير أرض مقدسة يقيم فيها البابا، تجسيد الإلهة فريا.";
					mes "اسلك الطريق نحو الشمال لتجد المبنى.";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[دليل راشيل]";
					mes "يمكنك أن تستريح من تعب الرحلة في النزل.";
					mes "المبنى الموجود على يساري هو نزل راشيل.";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[دليل راشيل]";
					mes "هل تريد إلقاء نظرة على الأسلحة التي تباع في راشيل؟";
					mes "يقع متجر الأسلحة بالقرب من البوابة الغربية.";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[دليل راشيل]";
					mes "يبيع متجر أدوات راشيل جرعات بأفضل جودة.";
					mes "يقع بالقرب من البوابة الغربية.";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[دليل راشيل]";
					mes "يقع المطار خارج البوابة الشرقية.";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[دليل راشيل]";
					mes "إذا أردت التخلص من كل علامات المواقع على خريطتك المصغرة،";
					mes "فاسألني مرة أخرى، واختر قائمة إزالة علامات الخريطة المصغرة.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[دليل راشيل]";
			mes "حسنا، لقد اختفت الآن. إن كانت لديك مواقع أخرى تريد السؤال عنها، فأخبرني فقط.";
			break;
		case 3:
			mes "[دليل راشيل]";
			mes "عندما تستخدم قائمة دليل القرية،";
			mes "تأكد من أن مواقع المباني ستحدد على الخريطة المصغرة في الجهة العلوية اليمنى من شاشتك.";
			mes "إذا لم تتمكن من رؤية الخريطة المصغرة، فاستخدم مفتاح الاختصار ctrl+tab أو اضغط زر الخريطة في نافذة معلوماتك الأساسية، حسنا؟";
			mes "ويمكنك أيضا تصغير عرض الخريطة المصغرة باستخدام زر - إذا لم تتمكن من رؤية خريطة القرية بالكامل.";
			break;
		case 4:
			mes "[دليل راشيل]";
			mes "آمل أن تحظى برحلة رائعة";
			mes "في أرونافلتز.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
