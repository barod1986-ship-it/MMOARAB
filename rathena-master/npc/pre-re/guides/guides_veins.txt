veins,210,345,5	script	دليل فينز#1::ve_guide	934,{
	mes "[دليل فينز]";
	mes "ترحب مدينة الصحراء فينز بالمغامرين الباحثين عن ملجأ من العواصف الرملية القاسية.";
	mes "إذا كانت هذه أول مرة تستخدم فيها خدمات المرشد، فلم لا تلقي نظرة على...";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("دليل القرية:إزالة العلامات من الخريطة المصغرة:ملاحظة:إلغاء")) {
		case 1:
			mes "[دليل فينز]";
			mes "يمكنني إخبارك بموقع أي مبنى ما دام داخل فينز.";
			mes "إلى أين تريد أن تذهب؟";
			if (.@compass_check == 0) {
				mes "هل تريد مني";
				mes "وضع علامات للمواقع";
				mes "على خريطتك المصغرة؟";
				next;
				if (select("نعم:لا") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("المعبد:النزل:متجر الأسلحة:متجر الأدوات:المنطاد:الحانة:معهد الأبحاث الجيولوجية:إلغاء")) {
				case 1:
					mes "[دليل فينز]";
					mes "يقع معبدنا في الشمال،";
					mes "وهو دائما مزدحم بأتباع الإلهة فريا المخلصين.";
					if (.@compass_check == 1) viewpoint 1,196,258,1,0xFF0000;
					break;
				case 2:
					mes "[دليل فينز]";
					mes "يمكنك إراحة تعب رحلتك في النزل.";
					mes "المبنى الموجود على يساري هو نزل فينز.";
					if (.@compass_check == 1) viewpoint 1,128,266,2,0xFF00FF;
					break;
				case 3:
					mes "[دليل فينز]";
					mes "نعم، عليك حماية نفسك من الخطر بنفسك.";
					mes "اشتر أسلحة عالية الجودة بأسعار مناسبة.";
					mes "يقع متجر أسلحة فينز في الغرب.";
					if (.@compass_check == 1) viewpoint 1,150,175,3,0x99FFFF;
					break;
				case 4:
					mes "[دليل فينز]";
					mes "هل جهزت ما يكفي من الضروريات لمغامرتك؟";
					mes "إذا لم تفعل، أقترح أن ترى ما يمكن لمركز فينز أن يقدمه لك.";
					if (.@compass_check == 1) viewpoint 1,230,161,4,0x0000FF;
					break;
				case 5:
					mes "[دليل فينز]";
					mes "يرجى العلم أن فينز تشغل مناطيد الشحن فقط.";
					if (.@compass_check == 1) viewpoint 1,273,285,5,0x00FF00;
					break;
				case 6:
					mes "[دليل فينز]";
					mes "إذا أردت تكوين صداقات مع";
					mes "أهل البلدة، أقترح أن";
					mes "تذهب لتناول شراب في الحانة";
					mes "إلى الغرب.";
					if (.@compass_check == 1) viewpoint 1,150,217,6,0x00FF00;
					break;
				case 7:
					mes "[دليل فينز]";
					mes "هل أنت مهتم بدراسة الجيولوجيا؟";
					mes "إذن من الأفضل أن تزور";
					mes "معهد الأبحاث الجيولوجية في";
					mes "الطابق الثاني من متجر الأسلحة.";
					if (.@compass_check == 1) viewpoint 1,150,175,7,0x00FF00;
					break;
				case 8:
					mes "[دليل فينز]";
					mes "إذا أردت التخلص من كل علامات المواقع على خريطتك المصغرة،";
					mes "فقط اسألني مرة أخرى، واختر قائمة 'إزالة العلامات من الخريطة المصغرة'.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,196,258,1,0xFF0000;
			viewpoint 2,128,266,2,0xFF00FF;
			viewpoint 2,150,175,3,0x99FFFF;
			viewpoint 2,230,161,4,0x0000FF;
			viewpoint 2,273,285,5,0x00FF00;
			viewpoint 2,150,217,6,0x00FF00;
			viewpoint 2,150,175,7,0x00FF00;
			mes "[دليل فينز]";
			mes "حسنا، لقد اختفت الآن. إذا أردت السؤال عن مواقع أخرى، فأخبرني فقط.";
			mes "استمتع بإقامتك في فينز.";
			break;
		case 3:
			mes "[دليل فينز]";
			mes "عند استخدام قائمة ''دليل القرية''،";
			mes "تأكد من أن مواقع المباني ستوضع عليها علامات في الخريطة المصغرة أعلى يمين شاشتك.";
			mes "إذا لم تتمكن من رؤية الخريطة المصغرة، فاستخدم مفتاح الاختصار ''ctrl+tab'' أو اضغط زر ''Map'' في نافذة المعلومات الأساسية لديك، حسنا؟";
			mes "كما يمكنك تصغير عرض الخريطة المصغرة باستخدام زر ''-'' إذا لم تتمكن من رؤية خريطة القرية كاملة.";
			break;
		case 4:
			mes "[دليل فينز]";
			mes "استمتع بإقامتك في فينز.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}

veins,189,101,5	duplicate(ve_guide)	دليل فينز#2::Veins Guide#2	934
