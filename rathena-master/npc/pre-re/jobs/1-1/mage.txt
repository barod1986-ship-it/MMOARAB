							if (countitem(909) > 0) set .@mixitem1_1,.@mixitem1_1 + .@input;
							set .@progress,2;
							break;
						}
					}
					break;
				case 2:
					while(1) {
						input .@input;
						if (.@input > 10000) {
							next;
							mes "[آلة الخلط]";
							mes "خطأ: تم تجاوز حد العناصر. يرجى إدخال قيم أقل من 10,000 والمحاولة مرة أخرى.";
							next;
						}
						else {
							if (countitem(914) > 0) set .@mixitem1_2,.@mixitem1_2 + .@input;
							set .@progress,2;
							break;
						}
					}
					break;
				case 3:
					while(1) {
						input .@input;
						if (.@input > 10000) {
							next;
							mes "[آلة الخلط]";
							mes "خطأ: تم تجاوز حد العناصر. يرجى إدخال قيم أقل من 10,000 والمحاولة مرة أخرى.";

						}
						else {
						if (countitem(519) > 0) set .@mixitem1_3,.@mixitem1_3 + .@input;
							set .@progress,2;
							break;
						}
					}
					break;

				case 4:
					if (.@progress != 2)
						set .@progress,1;
					set .@continue,1;
					next;
				}
				if (.@continue) break;
			}
		}
		mes "[آلة الخلط]";
		mes "يرجى إدخال ";
		mes "الرقم التسلسلي";
		mes "للمسحوق السحري.";

					next;
					if (select("نعم.:لا.") == 1) {
						break;
					}
				}
				else {
					mes "رقم تسلسلي غير صالح.";
					mes "يرجى المحاولة مرة أخرى.";
					next;
				}
			}
			else {
				mes "[آلة الخلط]";
				mes "الرقم التسلسلي هو #" + .@input + "، صحيح؟";
				next;
				if (select("تأكيد.:إلغاء.") == 1) {
					if (.@input == 8472)
						set .@magic_powder,1;
					else if (.@input == 3735)
						set .@magic_powder,2;
					else if (.@input == 2750)
						set .@magic_powder,3;
					else if (.@input == 5429)
						set .@magic_powder,4;
					else
						set .@magic_powder,5;
				}
				break;
			}
		}
		mes "[آلة الخلط]";
		mes "اختر";
		mes "حجر المحفز.";
		next;
		switch(select("جوهرة صفراء.:جوهرة حمراء.:جوهرة زرقاء.:ألماسة 1 قيراط.:تخطي.")) {

		case 2: set .@mixitem3,2; break;
		case 3: set .@mixitem3,3; break;
		case 4: set .@mixitem3,4; break;
		case 5: break;
		}
		mes "[آلة الخلط]";
		mes "كل شيء جاهز.";
		mes "بدء";
		mes "عملية الخلط.";
		mes "يرجى الانتظار.";
		next;
		mes "[آلة الخلط]";
		mes "- مثل اليوم -";
		switch(rand(1,5)) {
		case 1:
			mes "العين بالعين: عندما تأخذ من شخص، يجب أن تعوضه أو ترد له بطريقة ما.";
			break;
		case 2:
			mes "المصداقية عملة الرجل: للثقة الصادقة قيمة لا يمكن قياسها.";
			break;
		case 3:
			mes "كما تدين تدان: في النهاية، ستعامل بالطريقة التي تعامل بها الآخرين.";
			mes "يعني ذلك 'عندما تؤذي شخصا آخر، سيؤذيك في موقف لا مفر منه'.";
			break;
		case 4:
			mes "الجار الطيب أفضل من الأخ البعيد: عندما تحتاج إلى المساعدة، يمكنك الاعتماد على القريبين منك.";
			break;
		default:
			mes "الطيور على أشكالها تقع: يمكنك النظر إلى أصدقاء الشخص كمؤشر على شخصيته.";
		}
		next;
		if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 1 && .@mixitem2 == 1 && .@mixitem3 == 1 && .@magic_powder == 1) {
			mes "[آلة الخلط]";
			mes "محلول اختبار الساحر رقم 1.";
			getitem 1071,1; //Mage_Test_1
			next;
		}
		else if (.@mixitem1_1 == 3 && .@mixitem1_2 == 1 && .@mixitem1_3 == 1 && .@mixitem2 == 0 && .@mixitem3 == 2 && .@magic_powder == 2) {
			mes "[آلة الخلط]";
			mes "محلول اختبار الساحر رقم 2.";
			getitem 1085,1; //Mage_Test_2
			next;
		}
		else if (.@mixitem1_1 == 6 && .@mixitem1_2 == 1 && .@mixitem1_3 == 0 && .@mixitem2 == 1 && .@mixitem3 == 3 && .@magic_powder == 3) {
			mes "[آلة الخلط]";
			mes "محلول اختبار الساحر رقم 3.";
			getitem 1086,1; //Mage_Test_3
			next;
		}
		else if (.@mixitem1_1 == 2 && .@mixitem1_2 == 3 && .@mixitem1_3 == 0 && .@mixitem2 == 2 && .@mixitem3 == 4 && .@magic_powder == 4) {
			mes "[آلة الخلط]";
			mes "محلول اختبار الساحر رقم 4.";
			getitem 1087,1; //Mage_Test_4
			next;
		}
		else {
			mes "[آلة الخلط]";
			mes "حدث";
			mes "خطأ غير متوقع.";
			getitem 1090,1; //Mage_Test_Etc
			next;
		}
		mes "[آلة الخلط]";
		mes "اكتمل الخلط.";
		mes "شكرا لك.";
		close;
	}
	close;
}
