			else {
				mes "مهلا.";
				mes "تبدو كمتقدم للمريد. هل أنا محق؟";
				next;
				mes "[الأب يوسكي]";
				mes "ليس سيئا أبدا، لقد وصلت إلى هنا من برونتيرا. إذن ما اسمك، يا فتى؟";
				next;
				mes "[الأب يوسكي]";
				mes "" + strcharinfo(0) + "، هاه؟ لماذا ليس اسمك في قائمتي؟";
				next;
				mes "[الأب يوسكي]";
				mes "ربما ارتكبت خطأ. عد إلى المعبد وتحقق مع أسقف.";
				close;
			}
		}
		else {
			mes "أنت...";
			mes "أيها المبتدئ.";
			mes "هل هناك شيء";
			mes "تريد إخباري به؟";
			close;
		}
	}
	else if (BaseJob == Job_Acolyte) callfunc "F_FatherYos";
	else {
		if (BaseJob == Job_Priest) {
			mes "مهلا...";
			next;
			mes "[الأب يوسكي]";
			mes "إذا أردت، تعال واجلس هنا معي وتأمل الحقائق العظيمة. إن جلال الله ملهم حقا...";
			close;
		}
		else {
			mes "هل لديك ما تقوله؟ لأنه لسوء حظك،";
			mes "ليست لدي أي ردود.";
			close;
		}
	}
