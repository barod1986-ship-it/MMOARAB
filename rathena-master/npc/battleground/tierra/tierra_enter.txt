bat_room,124,178,5	script	ضابط وادي تييرا#01a::Tierra Gorge Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- لحظة من فضلك! -";
		mes "- أنت تحمل حاليا -";
		mes "- عددا كبيرا من العناصر. -";
		mes "- خفف حمولتك -";
		mes "- ثم حاول مرة أخرى. -";
		close;
	}
	mes "[ضابط جيش غيوم]";
	mes "لنر جنود كروا البغيضين قوة جيش غيوم!";
	next;
	switch(select("أريد الانضمام إلى جيشكم!:إنهاء المحادثة")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[ضابط جيش غيوم]";
			mes "يسعدني رغبتك في الانضمام إلينا ومحاربة جيش كروا، لكن لا يمكنني إرسال مقاتل مبتدئ مثلك إلى الموت في ساحة المعركة القاسية.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[ضابط جيش غيوم]";
				mes "يسعدني رغبتك في الانضمام إلينا ومحاربة جيش كروا، لكن لا يمكنني إرسال مقاتل مبتدئ مثلك إلى الموت في ساحة المعركة القاسية.";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[ضابط جيش غيوم]";
					mes "يبدو أنك عدت للتو من ساحة المعركة.";
					mes "ما زال الوقت مبكرا على عودتك. استرح قليلا واترك الأعداء لنا!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "[ضابط جيش غيوم]";
						mes "وصلني تقرير بأن فرقة نخبة أرسلت بالفعل إلى ساحة المعركة لمواجهة جيش كروا.";
						mes "انتظر قليلا في هذه الأثناء. اشحذ أسلحتك وجهز مؤنك حتى يحين دورك.";
					}
					else {
						if ($@TierraBG1) {
							mes "[ضابط جيش غيوم]";
							mes "هناك فرقة نخبة تستعد حاليا للانتشار في ساحة المعركة.";
							mes "انتظر قليلا في هذه الأثناء. اشحذ أسلحتك وجهز مؤنك حتى يحين دورك.";
						}
						else {
							mes "[ضابط جيش غيوم]";
							mes "تبدو مستعدا تماما للمعركة!";
							mes "اذهب وأر جنود كروا معنى الخوف الحقيقي!";
							mes "اليوم ستتردد صيحة نصرنا في أرجاء ساحة المعركة!";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[ضابط جيش غيوم]";
		mes "اليوم، سنحقق النصر!";
		break;
	}
	close;
}

bat_room,127,178,5	script	فارس غيوم#1::Guillaume Knight#1	417,{
	mes "[فارس غيوم]";
	mes "يمتد وادي تييرا بين منحدرين شديدين متقابلين، وتوجد مخازن مؤن جيشي غيوم وكروا باتجاهي الساعة 11 والساعة 7.";
	next;
	mes "[فارس غيوم]";
	mes "تبدأ المعركة عند سفينة جيشك، والهدف هو التقدم وتدمير مستودع مؤن العدو قبل أن يدمروا مستودعكم.";
	next;
	mes "[فارس غيوم]";
	mes "سيعيد معالج المعركة الجنود إلى القتال مرة كل 25 ثانية.";
	mes "الجيش الذي يستولي على راية الحياد في وسط ساحة المعركة سيحصل على نقطة عودة إضافية، فتعود قواته من موقع أقرب ويحصل على أفضلية.";
	next;
	mes "[فارس غيوم]";
	mes "يربط بين مستودعي المؤن ممران قصيران وضيّقان، لكن الحواجز تسد البوابات.";
	mes "فكر وتحرك بأسرع ما تستطيع. يعتمد نصر جيشك على مساهمتك.";
	next;
	mes "[فارس غيوم]";
	if (getbattleflag("feature.bgqueue"))
		mes "عندما تصبح مستعدا، قدم طلبك بالنقر على ^0000FFزر المعركة في الواجهة^000000.";
	else {
		mes "تذكر أن الانضمام إلى المعركة يضيف إلى سجل مهامك مؤقتا يبين مدة الانتظار قبل المشاركة مجددا في النوع نفسه من المعارك.";
		mes "يمكنك الاطلاع على المدة المتبقية بفتح سجل المهام عبر Alt+U.";
	}
	next;
	mes "[فارس غيوم]";
	if (getbattleflag("feature.bgqueue"))
		mes "تذكر أنك إذا ألغيت الطلب، ^0000FFفلن تتمكن من التقديم مجددا لمدة دقيقة واحدة^000000!";
	else
		mes "هل أنت مستعد للمعركة؟ قدم طلبك لدى مسؤول التجنيد المجاور لي!";
	close;
}
