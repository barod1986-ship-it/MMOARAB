bat_room,133,178,5	script	ضابط فلافيوس#01a::Flavius Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- لحظة من فضلك! -";
		mes "- أنت تحمل حاليا -";
		mes "- عددا كبيرا من العناصر. -";
		mes "- خفف حمولتك -";
		mes "- ثم حاول مرة أخرى. -";
		close;
	}
	mes "[ضابط جيش غيوم]";
	mes "لنر جنود كروا البغيضين قوة جيش غيوم!";
	next;
	switch(select("أريد الانضمام إلى جيشكم!:إنهاء المحادثة")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[ضابط جيش غيوم]";
			mes "يسعدني رغبتك في الانضمام إلينا ومحاربة جيش كروا، لكن لا يمكنني إرسال مقاتل مبتدئ مثلك إلى الموت في ساحة المعركة القاسية.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[ضابط جيش غيوم]";
				mes "يسعدني رغبتك في الانضمام إلينا ومحاربة جيش كروا، لكن لا يمكنني إرسال مقاتل مبتدئ مثلك إلى الموت في ساحة المعركة القاسية.";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[ضابط جيش غيوم]";
					mes "يبدو أنك عدت للتو من ساحة المعركة.";
					mes "ما زال الوقت مبكرا على عودتك. استرح قليلا واترك الأعداء لنا!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[ضابط جيش غيوم]";
						mes "وصلني تقرير بأن فرقة نخبة أرسلت بالفعل إلى ساحة المعركة لمواجهة جيش كروا.";
						mes "انتظر قليلا في هذه الأثناء. اشحذ أسلحتك وجهز مؤنك حتى يحين دورك.";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[ضابط جيش غيوم]";
							mes "هناك فرقة نخبة تستعد حاليا للانتشار في ساحة المعركة.";
							mes "انتظر قليلا في هذه الأثناء. اشحذ أسلحتك وجهز مؤنك حتى يحين دورك.";
						}
						else {
							mes "[ضابط جيش غيوم]";
							mes "تبدو مستعدا تماما للمعركة!";
							mes "اذهب وأر جنود كروا معنى الخوف الحقيقي!";
							mes "اليوم ستتردد صيحة نصرنا في أرجاء ساحة المعركة!";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[ضابط جيش غيوم]";
		mes "اليوم، سنحقق النصر!";
		break;
	}
	close;
}

bat_room,135,178,5	script	فارس غيوم#3::Guillaume Knight#3	417,{
	mes "[فارس غيوم]";
	mes "هدف معركة فلافيوس هو إحراز نقطتين قبل العدو عبر تدمير كريستالته.";
	next;
	mes "[فارس غيوم]";
	mes "كريستالتا الجيشين محميتان بحاجزين خاصين لا يمكن تدميرهما بالهجمات المباشرة.";
	next;
	mes "[فارس غيوم]";
	mes "يمكن إزالة الحاجزين بتدمير الحارسين اللذين يحميان قاعدة جيش العدو.";
	next;
	mes "[فارس غيوم]";
	mes "الكريستالات محصنة ضد جميع المهارات؛ ولا يمكن تدمير كريستالة العدو إلا بالهجمات الجسدية.";
	next;
	mes "[فارس غيوم]";
	if (getbattleflag("feature.bgqueue"))
		mes "عندما تصبح مستعدا، قدم طلبك بالنقر على ^0000FFزر المعركة في الواجهة^000000.";
	else {
		mes "تذكر أنك عند الانضمام إلى المعركة ستتلقى رمزا مؤقتا يحدد المدة التي لا يمكنك خلالها المشاركة مجددا في النوع نفسه من المعارك.";
		mes "يمكنك معرفة المدة المتبقية بالضغط على Alt+U.";
	}
	next;
	mes "[فارس غيوم]";
	if (getbattleflag("feature.bgqueue"))
		mes "تذكر أنك إذا ألغيت الطلب، ^0000FFفلن تتمكن من التقديم مجددا لمدة دقيقة واحدة^000000!";
	else
		mes "هل أنت مستعد للمعركة؟ قدم طلبك لدى المجند المجاور لي!";
	close;
}
