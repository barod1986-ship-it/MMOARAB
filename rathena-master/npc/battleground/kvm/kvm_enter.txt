// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	ضابط مرتزقة غيوم#1::KVM Mercenary Officer#1	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[ضابط مرتزقة كروا]";
		mes "لقد انضممت بالفعل إلى صفوف كروا. عد إليهم؛ فلن نقبل خائنا يقاتل معنا.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[ضابط مرتزقة غيوم]";
		mes "لقد شاركت مؤخرا في معركة KVM. عليك الانتظار 5 دقائق قبل التسجيل مجددا.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[ضابط مرتزقة غيوم]";
	mes "أرهم القوة الحقيقية لجيش غيوم!";
	next;
	switch(select("سأنضم إلى صفوفكم.:إنهاء المحادثة.")) {
	case 1:
		mes "[ضابط مرتزقة غيوم]";
		mes "أرهم مدى قوة جيشنا.";
		mes "اليوم، سيسمع الجميع هتاف انتصار غيوم!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[ضابط مرتزقة غيوم]";
		mes "سننتصر!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	فارس غيوم#kvm::Guillaume Knight#kvm	417,{
	mes "[فارس غيوم]";
	mes "مرحبا بك.";
	mes "ما الذي تريد معرفته؟";
	next;
	switch(select("التسجيل في KVM.:ما هي KVM؟:كيف أشارك في KVM؟:عرض رصيد نقاط KVM.")) {
	case 1:
		mes "[فارس غيوم]";
		if (getbattleflag("feature.bgqueue"))
			mes "إذا كنت مستعدا للقتال، فانقر على ^0000FFزر المعركة في الواجهة^000000 للتسجيل في معركة KVM!";
		else {
			mes "التسجيل غير متاح حاليا.";
			mes "للتسجيل، تحدث إلى ضابط مرتزقة KVM.";
		}
		close;
	case 2:
		mes "[فارس غيوم]";
		mes "KVM اختصار لعبارة كرايغر فون ميدغارد.";
		mes "أيها المغامر، هل تعلم أن الطريق إلى العالم الجديد أصبح مفتوحا؟";
		next;
		mes "[فارس غيوم]";
		mes "لدى قيادة غيوم عدة خطط للتقدم نحو العالم الجديد.";
		mes "وساحة KVM إحدى هذه الخطط.";
		next;
		mes "[فارس غيوم]";
		mes "فنحن إحدى دول قارة ميدغارد،";
		mes "وقد قررنا تجنيد عدد كبير من المغامرين للانطلاق فورا نحو العالم الجديد.";
		next;
		mes "[فارس غيوم]";
		mes "ولهذا نقيم منافسات KVM لاختيار أفضل المغامرين.";
		mes "ونتوقع مشاركة عدد كبير منهم.";
		next;
		mes "[فارس غيوم]";
		mes "وسيحصل المشاركون على مكافآت أيضا!";
		mes "للاطلاع على المكافآت، تحدث إلى ضابط لوجستيات KVM.";
		close;
	case 3:
		mes "[فارس غيوم]";
		if (getbattleflag("feature.bgqueue")) {
			mes "معارك KVM هي في الأساس مواجهات 3 ضد 3.";
			mes "يمكنك التسجيل بالنقر على ^0000FFزر المعركة في الواجهة^000000، ضمن مجموعة أو بصورة فردية.";
		} else {
			mes "معارك KVM هي في الأساس مواجهات 5 ضد 5.";
			mes "سجل أولا لدى موظف استقبال KVM، ضمن مجموعة أو بصورة فردية.";
		}
		next;
		mes "[فارس غيوم]";
		if (getbattleflag("feature.bgqueue"))
			mes "يستخدم التسجيل الجماعي عندما تريد دخول KVM مع أعضاء فريقك أو نقابتك،";
		else
			mes "يستخدم التسجيل الجماعي عندما تريد دخول KVM مع أعضاء فريقك،";
		mes "أما التسجيل الفردي فيتيح لك دخول KVM بمفردك.";
		next;
		mes "[فارس غيوم]";
		if (getbattleflag("feature.bgqueue"))
			mes "انتبه: إذا ألغيت طلبك، ^0000FFفلن تتمكن من التسجيل مجددا لمدة دقيقة واحدة^000000.";
		else {
			mes "سجل لدى ضابط KVM، وسيتواصل معك لاحقا عندما تكون في برونتيرا.";
			next;
			mes "[فارس غيوم]";
			mes "وعند دخولك، اتبع التعليمات داخل ساحة المعركة.";
			next;
			mes "[فارس غيوم]";
			mes "لكن تذكر أنه لن يستطيع التواصل معك ما لم تكن داخل مكتب KVM.";
		}
		close;
	case 4:
		mes "[فارس غيوم]";
		mes "رصيد نقاط KVM لديك:";
		mes ""+kvm_point+".";
		close;
	}
}
