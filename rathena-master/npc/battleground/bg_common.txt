		mes "[إيرونديك]";

			mes "ألغيت عملية الاستبدال.";

			mes "تم الاستبدال بنجاح.";

		else mes "لا تملك ما يكفي من "+getitemname(.@cost)+".";

	}
	close;

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[أكسل روز]";
			mes "المجد لغيوم!";
		} else if (.@team$ == "Croix") {
			mes "[سواندري]";
			mes "المجد لكروا!";
		}
		mes "لنحتفل بنصرنا المجيد!";
		mes strcharinfo(0)+"، هذه الشارة رمز لانتصارك.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[أكسل روز]";
			mes "رغم الهزيمة، أثبت إخلاصك في هذه المعركة.";
			mes "هذه مكافأة من غيوم مارولو تقديرا لتفانيك العظيم!";
			mes "اجعل من هذه الهزيمة درسا، وستنتصر حتما في المرة القادمة.";
		} else if (.@team$ == "Croix") {
			mes "[سواندري]";
			mes "يا "+strcharinfo(0)+"، لا تحزن.";
			mes "لم ننتصر، لكننا بذلنا كل ما في وسعنا.";
			mes "هذه هدية ملكية من كروا. لا تنس هذه المعركة؛ سننتصر في المرة القادمة.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
