gef_fild10,242,202,0	script	شظية الصدع البعدي::Dimensional Gorge Piece	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Orc's Memory";
	set .@md_name_display$,"ذاكرة الأورك";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "لا يمكن دخول هذه الزنزانة إلا للاعبين من المستوى ^ff000030 إلى 80^000000.";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "تم التحقق من حالة المجموعة. هل تريد حجز الدخول إلى "+.@md_name_display$+"؟";
			next;
			switch(select("حجز "+.@md_name_display$+":دخول الزنزانة:إلغاء")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "اسم المجموعة: "+ getpartyname(.@party_id);
					mes "قائد المجموعة: "+strcharinfo(0);
					mes "^0000ff"+.@md_name_display$+" ^000000 - فشل الحجز.";
					close;
				}
				mes "^0000ffجار حجز مدخل إلى "+.@md_name_display$+"...^000000";
				mes "بعد إتمام الحجز، اختر 'دخول الزنزانة' من القائمة لدخول "+.@md_name_display$+".";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name_display$ + " دخول الزنزانة التذكارية:إلغاء") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "يمكنك دخول الزنزانة إن كانت قد أنشئت.";
		next;
		if(select("دخول الزنزانة "+.@md_name_display$+":إلغاء") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000ffحذفت جميع السجلات والآثار المرتبطة بزنزانة ذاكرة الأورك. يمكنك الآن إنشاء الزنزانة مجددا أو العودة إليها.^000000";
		erasequest 12059;
		close;
	}
	mes "لا يمكن إنشاء الزنزانة إلا إذا كنت قائد المجموعة وكان فيها عضوان على الأقل.";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "حدث خطأ غير معروف.";
		close;
	case IE_NOINSTANCE:
		mes "زنزانة ذاكرة الأورك التذكارية غير موجودة.";
		mes "إما أن قائد المجموعة دمر الزنزانة التذكارية أو أن مهلتها انتهت. حاول مجددا بعد ساعتين.";
		close;
	case IE_NOMEMBER:
		mes "لا يمكن دخول الزنزانة التذكارية إلا لأعضاء المجموعة.";
		close;
	case IE_OK:
		mapannounce "gef_fild10", "دخل عضو مجموعة "+ getpartyname( getcharid(1) ) +"، "+strcharinfo(0)+"، إلى ذاكرة الأورك.",bc_map,"0x00ff99";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}

	end;

OnInstanceInit:
	disablenpc instance_npcname("#Resurrect Monsters1");
	end;

OnDisable:
	stopnpctimer;
	killmonster instance_mapname("1@orcs"),instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnEnable:
	enablenpc instance_npcname("#Resurrect Monsters1");
	monster instance_mapname("1@orcs"),0,0,"محارب أورك",1023,30,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	end;

OnMyMobDead:
	set .@map$, instance_mapname("1@orcs");
	set .@mob_dead_num, 30 - mobcount(.@map$,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead");
	if (.@mob_dead_num > 0) {
		set .@mob_ran,rand(1,30);
		if (.@mob_ran > 29) {
			monster .@map$,0,0,"محارب أورك",1023,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
		}
		else if ((.@mob_ran > 28) && (.@mob_ran < 30)) {
			monster .@map$,0,0,"أورك عال",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "أورك عال: نحتاج إلى تعزيز الدفاعات! أحضروا مزيدا من الجنود!",bc_map,"0xff4444";
			}
		}
		else if ((.@mob_ran > 26) && (.@mob_ran < 29)) {
			areamonster .@map$,41,91,51,81,"أورك عال",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,10) == 9) {
				mapannounce .@map$, "أين الأورك العالون؟! أرسلوهم لإيقاف الأعداء!",bc_map,"0xff4444";
			}
		}
		else {
			areamonster .@map$,17,187,27,177,"أورك عال",1213,.@mob_dead_num,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
			if (rand(1,5) == 3) {
				mapannounce .@map$, "تحذير: بدأ الجيش يحتشد في المنطقة رقم 4.",bc_map,"0x77ff77";
			}
			if (rand(1,100) == 50) {
				initnpctimer;
			}
		}
	}
	end;

OnTimer10:
	mapannounce instance_mapname("1@orcs"), "صيحات قائد حرس الأورك: يبدو أن الأمر سيستغرق أطول مما توقعنا. استدعوا غولمات الصواعد!",bc_map,"0xff4444";
	end;

	set .@map$, instance_mapname("1@orcs");
	mapannounce .@map$, "تشق غولمات الصواعد طريقها من أعماق الأرض.",bc_map,"0x77ff77";
	areamonster .@map$,17,187,27,177,"غولم الصواعد",1278,20,instance_npcname("#Resurrect Monsters1")+"::OnMyMobDead";
	stopnpctimer;
	end;
