			callsub L_Enter,1,1;
		case 2:

			close2;
			warp "alberta",223,36;
			end;
		case 3:
			end;
		}
	case 0:
	case 1:
		if (.@etower_timer2 < 2 && getcharid(1) == etower_partyid) {
			mes "إذا سبق أن أنشأت الزنزانة، فيمكنك دخولها.";
			next;
			switch(select("دخول "+.@md_name_display$+":العودة إلى ألبرتا:إلغاء")) {
			case 1:
				callsub L_Enter,0,0;
			case 2:
				mes "سأنقلك إلى ألبرتا.";
				close2;
				warp "alberta",223,36;
				end;
			case 3:
				close;
			}
		} else {
			set .@dun_lim_time,etower_timer+604800; // 1 week
			// set .@dun_lim_time2,etower_timer+14400; // 4 hours

			set .@dun_h,(.@dun_ent_t / 3600);
			set .@dun_m,(.@dun_ent_t - (.@dun_h * 3600)) / 60;

			setquest 60201;

		if (getarg(0) == 0) close;
		else end;
	}
}
