function	script	F_Custom_Kafra	{

	.npcName$ = "[Kafra Employee]";
	.storageFee = 40; // Personal storage fee in Zeny
	.warehouseFee = 1000; // Guild Warehouse fee in Zeny

	cutin getarg(4), 2;
	mes .npcName$;
	mes "Welcome to the Kafra Corporation.";
	mes "The Kafra Services are always here to support you.";
	mes "So how can I be of service today?";
	next;

	.@menu$ = "^1A66FFSave^000000:Open Storage ^727272[" + callfunc("F_InsertComma", .storageFee) + " z]^000000";
	.@menu$ += ":Open Guild Warehouse ^727272[" + callfunc("F_InsertComma", .warehouseFee) + " z]^000000";
	.@menu$ += ":Rental Service:^727272Cancel^000000";

	switch(select(.@menu$)) {
		case 1:
			mes .npcName$;
			if (getarg(2)) {
				savepoint getarg(1), getarg(2), getarg(3);
				mes "Your Respawn Point has been saved here in the " + getarg(0) + ".";
				mes "Thank you.";
			} else {
				mes "I'm sorry but you can't set your respawn point in here.";
			}
			break;
		case 2:
			callfunc "F_KafStor", 0, .storageFee, 0;
			break;
		case 3:
			if (Zeny < .warehouseFee) {
				mes .npcName$;
				mes "I am afraid you do not have enough money to settle the fee.";
				mes "The fee is " + callfunc("F_InsertComma", .warehouseFee) + " z.";
				close3;
			}

			if (Zeny < .warehouseFee) {
				// Cheat prevention
				end;
			}

			mes .npcName$;
			mes "I will open the guild storage for you then. Have a memorable time!";
			close2;
			cutin "", 255;

			switch(guildopenstorage()) {
				case GSTORAGE_STORAGE_ALREADY_OPEN:
					mes .npcName$;
					mes "You are using your storage.";
					mes "Please wait a while, then come back.";
					close3;
				case GSTORAGE_ALREADY_OPEN:
					mes .npcName$;
					mes "The Guild Warehouse is being used right now.";
					mes "Please wait a while, then come back.";
					close3;
				case GSTORAGE_NO_GUILD:
					mes .npcName$;
					mes "You can't use this service if you're not in a guild!";
					close3;
				case GSTORAGE_NO_STORAGE:
					mes .npcName$;
					mes "The guild does not have a guild storage.";
					close3;
				case GSTORAGE_NO_PERMISSION:
					mes .npcName$;
					mes "You don't have permission to use the guild storage!";
					close3;
				default:
					Zeny -= .warehouseFee;
					end;
			}
			break;
		case 4:
			if (ismounting()) {
				message strcharinfo(0), "I must remove my mount first.";
				cutin "",255;
				end;
			}

			// Cart Rental
			if (!checkcart() && getskilllv("MC_PUSHCART")) {
				.@cartFee = 1000;
				mes "[Rental Service]";
				mes "Rent a cart for ^FF3355" + callfunc("F_InsertComma", .@cartFee) + "^000000 z?";
				next;
				switch(select("Yes:No")) {
					case 1:
						mes "[Rental Service]";
						if (Zeny < .@cartFee) {
							mes "You need ^FF3355" + callfunc("F_InsertComma", .@cartFee) + "^000000 z to rent a cart.";
							close3;
						}
						Zeny -= .@cartFee;
						setcart;
						specialeffect2 EF_CARTREVOLUTION;
						mes "Cart has been rented!";
						close3;
					case 2:
						mes "[Rental Service]";
						mes "Okay then.";
						close3;

				}
				close3;
			}

			// Falcon Rental
			if (!checkfalcon() && getskilllv("HT_FALCON")) {
				.@falconFee = 2500;
				mes "[Rental Service]";
				mes "Rent a falcon for ^FF3355" + callfunc("F_InsertComma", .@falconFee) + "^000000 z?";
				next;
				switch(select("Yes:No")) {
					case 1:
						mes "[Rental Service]";
						if (Zeny < .@falconFee) {
							mes "You need ^FF3355" + callfunc("F_InsertComma", .@falconFee) + "^000000 z to rent a falcon.";
							close3;
						}
						Zeny -= .@falconFee;
						setfalcon;
						specialeffect2 EF_DETECTING;
						mes "Falcon has been rented!";
						close3;
					case 2:
						mes "[Rental Service]";
						mes "Okay then.";
						close3;

				}
				close3;
			}

			// Peco Peco Rental
			if (!checkriding() && getskilllv("KN_RIDING")) {
				if (BaseJob == Job_Knight) {
					.@pecoFee = 2500;
				} else if (BaseJob == Job_Crusader) {
					.@pecoFee = 3500;
				} else {
					mes "[Rental Service]";
					mes "Only Knights and Crusader can rent a Peco Peco.";
					close3;
				}
				
				mes "[Rental Service]";
				mes "Rent a Peco Peco for ^FF3355" + callfunc("F_InsertComma", .@pecoFee) + "^000000 z?";
				next;
				switch(select("Yes:No")) {
					case 1:
						mes "[Rental Service]";
						if (Zeny < .@pecoFee) {
							mes "You need ^FF3355" + callfunc("F_InsertComma", .@pecoFee) + "^000000 z to rent a mount.";
							close3;
						}
						Zeny -= .@pecoFee;
						setriding;
						specialeffect2 EF_MAGNUMBREAK;
						mes "Peco peco has been rented!";
						close3;
					case 2:
						mes "[Rental Service]";
						mes "Okay then.";
						close3;

				}
				close3;
			}

			mes .npcName$;
			mes "You do not meet requirements to rent.";
			break;
		case 5:
			mes .npcName$;
			mes "^727272*Whew...*^000000";
			mes "Great, because I'm actually on my break.";
			break;
	}
	close3;
}
