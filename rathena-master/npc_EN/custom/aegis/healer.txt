//===== rAthena Script ===========================================
//= Healer & Buffer NPC
//===== By: ======================================================
//= wrywndp (https://www.youtube.com/@wrywndp)
//===== Description: =============================================
//= Advanced healer NPC script (Pre-Renewal).
//= Features a progressive scaling system where healing and buff 
//= fee increase every Base Levels.
//= Includes a dynamic buff fee calculation based on the number of 
//= active buff requested by the player.
//===== Additional Comments: =====================================
//= This script is made for Ragnarok Offline Pre-Renewal Pack
//= https://www.youtube.com/watch?v=aN5OCJdj2qA
//================================================================

-	script	Healer	-1,{

	disable_items;

	//=====  SETTINGS  =====//
	.delay			= 5;	// NPC delay (in seconds)
	.itemBuffDelay	= 30;	// Item-based buff cooldown (seconds)

	.healFee		= true; 	// Toggle payment requirement for Heal functions
    .buffFee		= true; 	// Toggle payment requirement for Buff functions
	.dyBuffFee		= true;		// Dynamic buff fee based on active buff groups
	.bDuration		= 10;		// Buff duration (in minutes)

	.ibDuration		= false;	// Toggle item-based buff duration (false = Use .bDuration, true = Original)

	//===== if (.ibDuration) == true =====//
	// This duration based on original duration (in minutes)
	.aspersioDuration	= 3;	// Priest (Aspersio)
	.cpDuration			= 10;	// Alchemist (Chemical Protection)
	.berserkDuration	= 15;	// Alchemist (Berserk Pitcher)
	.elemDuration		= 30;	// Sage (Elemental Weapon)

	.repair			= true;	// Automatic equipment repair
	.identify		= true;	// Automatic identify
	.removeDebuff	= true;	// Remove negative status

	.healHp			= 100;	// HP heal percentage per pulse (0–100)
	.healSp			= 100;	// SP heal percentage per pulse (0–100)
	.healCount		= 3;	// Number of heal pulses
	.healInterval	= 5;	// Delay between each heal pulse (in seconds)

	.foodBuff		= false;	// Food buff (All stats)
	.foodBuffLevel	= 10;		// Food buff level

	//=====  BUFF TOGGLES  =====//
	.enableAcolyte			= true;	// Acolyte basic
	.enablePriest			= true;	// Priest basic
	.enablePrAspersio		= true;	// Priest (Aspersio)
	.enablePrSacramenti		= true;	// Priest (B.S Sacramenti)
	.enableHighPriest		= true;	// High Priest (Assumptio)
	.enableBlacksmith		= true;	// Blacksmith basic
	.enableBsOverThrust		= true;	// Blacksmith (Over Thrust)
	.enableAlProtection		= true;	// Alchemist (Chemical Protection)
	.enableAlBerserk		= true;	// Alchemist (Berserk Pitcher)
	.enableCrusader			= true;	// Crusader (Providence)
	.enableSniper			= true;	// Sniper (Wind Walk)
	.enableSoulLinker		= true;	// Soul Linker (Spirit)
	.enableSoulLinkerKa		= true;	// Soul Linker (Ka)
	.enableAssassin			= true;	// Assassin (Enchant Poison)
	.enableSageElemental	= true;	// Sage (Elemental Weapon)



	// Convert millisecond to minute [DO NOT TOUCH THESE!]
	.bDuration	*= 60000;
	.aspDur		= .ibDuration ? .aspersioDuration	* 60000 : .bDuration;
	.cpDur		= .ibDuration ? .cpDuration			* 60000 : .bDuration;
	.berserkDur	= .ibDuration ? .berserkDuration	* 60000 : .bDuration;
	.elemDur	= .ibDuration ? .elemDuration		* 60000 : .bDuration;

	HealerMenu:
		//=====  BUFF DATABASE  =====//
		// Verify if the player has active buffs in the database
		.@buffs = configBuff;

		// Crusader & Paladin cannot receive Providence, clean up if job changed
		if (configBuff & 512 && (BaseJob == Job_Crusader || BaseJob == Job_Paladin)) {
			configBuff &= ~512;
			.@buffs = configBuff;
		}

		// Retrieve player buff configuration using bit fields
		if (.@buffs) {
			if (.enableAcolyte)			.@acolyte		= configBuff & 1;		// Acolyte basic
			if (.enablePriest)			.@priest		= configBuff & 2;		// Priest basic
			if (.enablePrAspersio)		.@prAspersio	= configBuff & 4;		// Priest (Aspersio)
			if (.enablePrSacramenti)	.@prSacramenti	= configBuff & 8;		// Priest (B.S Sacramenti)
			if (.enableHighPriest)		.@highPriest	= configBuff & 16;		// High Priest (Assumptio)
			if (.enableBlacksmith)		.@blacksmith	= configBuff & 32;		// Blacksmith basic
			if (.enableBsOverThrust)	.@bsOverThrust	= configBuff & 64;		// Blacksmith (Over Thrust)
			if (.enableAlProtection)	.@alProtection	= configBuff & 128;		// Alchemist (Chemical Protection)
			if (.enableAlBerserk)		.@alBerserk		= configBuff & 256;		// Alchemist (Berserk Pitcher)
			if (.enableCrusader)		.@crusader		= configBuff & 512;		// Crusader (Providence)
			if (.enableSniper)			.@sniper		= configBuff & 1024;	// Sniper (Wind Walk)
			if (.enableSoulLinker)		.@soulLinker	= configBuff & 2048;	// Soul Linker (Spirit)
			if (.enableSoulLinkerKa)	.@soulLinkerKa	= configBuff & 4096;	// Soul Linker (Ka)
			if (.enableAssassin)		.@assassin		= configBuff & 8192;	// Assassin (Enchant Poison)
			if (.enableSageElemental)	.@sageElemental	= configBuff & 16384;	// Sage (Elemental Weapon)
		}



		//=====  FEE CALCULATION  =====//
		if (.healFee) {
			.@healFee = 50 * (BaseLevel + 1);
		}

		if (.buffFee) {
			if (.@buffs) {
				if (.dyBuffFee) {
					.@tmp = configBuff;
					.@activeGroups = 0;
					while (.@tmp) {
						.@activeGroups += (.@tmp & 1);
						.@tmp >>= 1;
					}

					// Crusader & Paladin cannot receive Providence
					if (.@crusader && (BaseJob == Job_Crusader || BaseJob == Job_Paladin)) {
						.@crusader = 0;
						.@activeGroups--;
						if (.@activeGroups < 0) .@activeGroups = 0;
					}

					.@feePerGroup = 10 * (BaseLevel + 1);
					.@buffFee = .@feePerGroup * .@activeGroups;
				} else {
					.@buffFee = 50 * (BaseLevel + 1);
				}
			} else {
				.@buffFee = 0;
			}
		}

		.@totalFee = .@healFee + .@buffFee;



		//=====  MAIN SCRIPT  =====//
		.@mainMenu$ = "^1A66FFHeal" + ((.@buffs) ? " + Buffs" : "") + "^000000" + ((.@totalFee) ? " ^727272[" + callfunc("F_InsertComma", .@totalFee) + " z]^000000" : "");
		.@mainMenu$ += ":Settings";
		.@mainMenu$ += ":^727272Cancel^000000";

		switch(select(.@mainMenu$)) {
			case 1:
				//=====  DELAY  =====//
				if (@hd > gettimetick(2)) {
					.@timeLeft = @hd - gettimetick(2);
					message strcharinfo(0), "Please wait " + callfunc("F_InsertPlural", .@timeLeft, "second") + " to use Healer again.";
					end;
				}

				// Initialize trackers for item-based buff
				.@missingCount = 0;
				.@delayCount = 0;

				if (Zeny < .@totalFee) {
					message strcharinfo(0), "Healing" + ((.@buffs) ? " and Buffs" : "") + " costs " + callfunc("F_InsertComma", .@totalFee) + " Zeny.";
					end;
				}

				Zeny -= .@totalFee;

				// Automatic equipment repair
				if (.repair) {
					repairall;
				}

				// Auto-Identify
				.@identifyTemp = identifyall(false);
				if (.identify && .@identifyTemp) {
					message strcharinfo(0), callfunc("F_InsertPlural", .@identifyTemp, "item") + " identified.";
					identifyall;
				}

				// Cleanse negative status
				if (.removeDebuff) {
					setarray .@debuffList,
						SC_BERSERK,
						SC_BLEEDING,
						SC_BLIND,
						SC_CHANGEUNDEAD,
						SC_CONFUSION,
						SC_CRITICALWOUND,
						SC_CURSE,
						SC_DECREASEAGI,
						SC_FREEZE,
						SC_HALLUCINATION,
						SC_HELLPOWER,
						SC_LEXAETERNA,
						SC_ORCISH,
						SC_POISON,
						SC_SILENCE,
						SC_SLEEP,
						SC_SLOWCAST,
						SC_SLOWDOWN,
						SC_STONE,
						SC_STRIPARMOR,
						SC_STRIPHELM,
						SC_STRIPSHIELD,
						SC_STRIPWEAPON;

					for (.@i = 0; .@i < getarraysize(.@debuffList); .@i++) {
						sc_end .@debuffList[.@i];
					}
				}

				// Buff
				if (.@buffs) {
					// Food buff (All Stats)
					if (.foodBuff) {
						specialeffect2 EF_FOOD01;
						sc_start SC_STRFOOD, .bDuration, .foodBuffLevel;
						sc_start SC_AGIFOOD, .bDuration, .foodBuffLevel;
						sc_start SC_VITFOOD, .bDuration, .foodBuffLevel;
						sc_start SC_INTFOOD, .bDuration, .foodBuffLevel;
						sc_start SC_DEXFOOD, .bDuration, .foodBuffLevel;
						sc_start SC_LUKFOOD, .bDuration, .foodBuffLevel;
					}

					// Acolyte
					if (.@acolyte) {
						specialeffect2 EF_INCAGIDEX;
						sc_start SC_BLESSING, .bDuration, 10;
						sc_start SC_INCREASEAGI, .bDuration, 10;
						sc_start SC_ANGELUS, .bDuration, 10;
					}

					// Priest
					if (.@priest) {
						specialeffect2 EF_MAGNIFICAT;
						sc_start SC_MAGNIFICAT, .bDuration, 5;
						sc_start SC_GLORIA, .bDuration, 5;
						sc_start SC_IMPOSITIO, .bDuration, 5;
						sc_start SC_SLOWPOISON, .bDuration, 4;
						sc_start SC_SUFFRAGIUM, .bDuration, 3;

						// Kyrie Eleison will not be activated if Assumptio also activated
						if (!.@highPriest) {
							specialeffect2 EF_KYRIE;
							sc_start SC_KYRIE, .bDuration, 10;
						}
					}

					// Priest (Aspersio)
					if (.@prAspersio) {
						if (@aspersioDelay > gettimetick(2)) {
							.@aspersioDelay = @aspersioDelay - gettimetick(2);
							.@dNames$[.@delayCount] = "Aspersio (" + .@aspersioDelay + "s)";
							.@delayCount++;
						} else {
							.holyWaterId = 523; // Holy Water Item ID

							if (countitem(.holyWaterId)) {
								specialeffect2 EF_ASPERSIO;
								sc_start SC_ASPERSIO, .aspDur, 5;
								delitem .holyWaterId, 1;
								@aspersioDelay = gettimetick(2) + .itemBuffDelay;
							} else {
								.@mNames$[.@missingCount] = getitemname(.holyWaterId);
								.@missingCount++;
							}
						}
					}

					// Priest (B.S Sacramenti)
					if (.@prSacramenti) {
						specialeffect2 EF_BENEDICTIO;
						sc_start SC_BENEDICTIO, .bDuration, 5;
					}

					// High Priest (Assumptio)
					if (.@highPriest) {
						specialeffect2 EF_ASSUMPTIO2;
						sc_start SC_ASSUMPTIO, .bDuration, 5;
					}

					// Blacksmith
					if (.@blacksmith) {
						specialeffect2 EF_HASTEUP;
						sc_start SC_ADRENALINE, .bDuration, 5;
						sc_start SC_ADRENALINE2, .bDuration, 5;
						sc_start SC_WEAPONPERFECTION, .bDuration, 5;
					}

					// Blacksmith (Over Thrust)
					if (.@bsOverThrust) {
						specialeffect2 EF_OVERTHRUST;
						sc_start SC_OVERTHRUST, .bDuration, 5;
					}

					// Alchemist (Chemical Protection)
					if (.@alProtection) {
						if (@cpDelay > gettimetick(2)) {
							.@cpDelay = @cpDelay - gettimetick(2);
							.@dNames$[.@delayCount] = "Chemical Protection (" + .@cpDelay + "s)";
							.@delayCount++;
						} else {
							.glisteningCoatId = 7139; // Glistening Coat

							if (countitem(.glisteningCoatId)) {
								specialeffect2 EF_CHEMICALPROTECTION;
								sc_start SC_CP_HELM, .cpDur, 5;
								sc_start SC_CP_ARMOR, .cpDur, 5;
								sc_start SC_CP_WEAPON, .cpDur, 5;
								sc_start SC_CP_SHIELD, .cpDur, 5;
								delitem .glisteningCoatId, 1;
								@cpDelay = gettimetick(2) + .itemBuffDelay;
							} else {
								.@mNames$[.@missingCount] = getitemname(.glisteningCoatId);
								.@missingCount++;
							}
						}
					}

					// Alchemist (Berserk Pitcher)
					if (.@alBerserk) {
						if (@berserkDelay > gettimetick(2)) {
							.@berserkDelay = @berserkDelay - gettimetick(2);
							.@dNames$[.@delayCount] = "Berserk Pitcher (" + .@berserkDelay + "s)";
							.@delayCount++;
						} else {
							.berserkId = 657; // Berserk Potion
							.berserkReq = 2; // How many Berserk Potion needed

							if (countitem(.berserkId) >= .berserkReq) {
								specialeffect2 EF_POTION_BERSERK;
								sc_start SC_ASPDPOTION2, .berserkDur, 0;
								delitem .berserkId, .berserkReq;
								@berserkDelay = gettimetick(2) + .itemBuffDelay;
							} else {
								.@berserkLack = .berserkReq - countitem(.berserkId);
								.@mNames$[.@missingCount] = callfunc("F_InsertPlural", .@berserkLack, getitemname(.berserkId));
								.@missingCount++;
							}
						}
					}

					// Crusader (Providence)
					if (.@crusader && BaseJob != Job_Crusader && BaseJob != Job_Paladin) {
						specialeffect2 EF_PROVIDENCE;
						sc_start SC_PROVIDENCE, .bDuration, 5;  
					}

					// Sniper (Wind Walk)
					if (.@sniper) {
						specialeffect2 EF_PORTAL4;
						sc_start SC_WINDWALK, .bDuration, 10;
					}

					// Soul Linker (Spirit)
					if (.@soulLinker) {
						switch (BaseJob) {
							case Job_Knight:			.@spirit = 452; break;
							case Job_Wizard:			.@spirit = 453; break;
							case Job_Priest:			.@spirit = 454; break;
							case Job_Assassin:			.@spirit = 457; break;
							case Job_Hunter:			.@spirit = 460; break;
							case Job_Blacksmith:		.@spirit = 458; break;
							case Job_Crusader:			.@spirit = 450; break;
							case Job_Sage:				.@spirit = 449; break;
							case Job_Monk:				.@spirit = 447; break;
							case Job_Rogue:				.@spirit = 456; break;
							case Job_Bard:
							case Job_Dancer:			.@spirit = 455; break;
							case Job_Alchemist:			.@spirit = 445; break;
							case Job_Soul_Linker:		.@spirit = 461; break;
							case Job_Star_Gladiator:	.@spirit = 448; break;
							case Job_SuperNovice:		.@spirit = 451; break;
							default:
								if (Upper == 1 && BaseLevel < 70) {
									.@spirit = 494;
								}
						}

						if (.@spirit) {
							specialeffect2 EF_SOULLINK;
							sc_start2 SC_SPIRIT, .bDuration, 5, .@spirit;
						}
					}

					// Soul Linker (Ka)
					if (.@soulLinkerKa) {
						specialeffect2 EF_KAIZEL;
						sc_start SC_KAIZEL, .bDuration, 7;
						sc_start SC_KAAHI, .bDuration, 7;
						sc_start SC_KAUPE, .bDuration, 3;

						// Kaite will not be activated if Assumptio also activated
						if (!.@highPriest) {
							sc_start SC_KAITE, .bDuration, 7;
						}
					}

					// Assassin (Enchant Poison)
					if (.@assassin) {
						specialeffect2 EF_ENCHANTPOISON;
						sc_start SC_ENCPOISON, .bDuration, 10;
					}

					// Sage (Elemental Weapon)
					if (.@sageElemental) {
						if (@sageElementalDelay > gettimetick(2)) {
							.@sageElementalDelay = @sageElementalDelay - gettimetick(2);
							.@dNames$[.@delayCount] = "Elemental Weapon (" + .@sageElementalDelay + "s)";
							.@delayCount++;
						} else {
							switch (sageElement) {
								case 1: .@elemItemId = 990; break; // Red Blood (Fire)
								case 2: .@elemItemId = 991; break; // Crystal Blue (Water)
								case 3: .@elemItemId = 992; break; // Wind of Verdure (Wind)
								case 4: .@elemItemId = 993; break; // Green Live (Earth)
							}

							if (countitem(.@elemItemId)) {
								switch (sageElement) {
									case 1: specialeffect2 EF_FLAMELAUNCHER; sc_start SC_FIREWEAPON, .elemDur, 1; break;
									case 2: specialeffect2 EF_FROSTWEAPON; sc_start SC_WATERWEAPON, .elemDur, 1; break;
									case 3: specialeffect2 EF_LIGHTNINGLOADER; sc_start SC_WINDWEAPON, .elemDur, 1; break;
									case 4: specialeffect2 EF_SEISMICWEAPON; sc_start SC_EARTHWEAPON, .elemDur, 1; break;
								}
								delitem .@elemItemId, 1;
								@sageElementalDelay = gettimetick(2) + .itemBuffDelay;
							} else {
								setarray .elemNames$[1],
									"Red Blood",
									"Crystal Blue",
									"Wind of Verdure",
									"Green Live";
								.@mNames$[.@missingCount] = .elemNames$[sageElement];
								.@missingCount++;
							}
						}
					}

					// Notify player of any item-based buffs that couldn't be applied due to missing items
					if (.@missingCount) {
						.@msg$ = "Missing item" + ((.@missingCount > 1 || .@berserkLack > 1) ? "s" : "") + ": " + .@mNames$[0];
						
						for (.@i = 1; .@i < .@missingCount; .@i++) {
							if (.@i == .@missingCount - 1) {
								.@msg$ += " & " + .@mNames$[.@i];
							} else {
								.@msg$ += ", " + .@mNames$[.@i];
							}
						}

						// Refund fee for each buff that couldn't be applied due to missing items
						if (.buffFee && .dyBuffFee) {
							Zeny += .@feePerGroup * .@missingCount;
						}

						message strcharinfo(0), .@msg$ + ".";
					}

					// Notify player of any item-based buffs that couldn't be applied due to active cooldown
					if (.@delayCount) {
						.@dMsg$ = .@dNames$[0];
						for (.@i = 1; .@i < .@delayCount; .@i++) {
							if (.@i == .@delayCount - 1) {
								.@dMsg$ += " & " + .@dNames$[.@i];
							} else {
								.@dMsg$ += ", " + .@dNames$[.@i];
							}
						}
						
						// Refund fee for each buff that couldn't be applied due to active cooldown
						if (.buffFee && .dyBuffFee) {
							Zeny += .@feePerGroup * .@delayCount;
						}

						message strcharinfo(0), "Delay: " + .@dMsg$ + ".";
					}
				}

				// Schedule heal pulses with interval delay
				for (.@i = 0; .@i < .healCount; .@i++) {
					addtimer .@i * (.healInterval * 1000), strnpcinfo(3) + "::OnStart";
				}

				// Start delay count
				if (.delay) {
					@hd = gettimetick(2) + .delay;
				}

				end;
			case 2:
				.onText$	= " ^727272[^1A66FFON^000000^727272]^000000  ";
				.offText$	= "^727272[^FF3355OFF^000000^727272]^000000 ";

				ClassBuffMenu:
					.@classIndex = 0;
					.@sageIndex = -1;
					.@assassinIndex = -1;
					if (.enableAcolyte)      { .@classBits[.@classIndex] = 1;     .@classNames$[.@classIndex++] = " Acolyte"; }
					if (.enablePriest)       { .@classBits[.@classIndex] = 2;     .@classNames$[.@classIndex++] = " Priest"; }
					if (.enablePrAspersio)   { .@classBits[.@classIndex] = 4;     .@classNames$[.@classIndex++] = " Priest  ^727272(Aspersio)^000000"; }
					if (.enablePrSacramenti) { .@classBits[.@classIndex] = 8;     .@classNames$[.@classIndex++] = " Priest  ^727272(B.S Sacramenti)^000000"; }
					if (.enableHighPriest)   { .@classBits[.@classIndex] = 16;    .@classNames$[.@classIndex++] = " High Priest  ^727272(Assumptio)^000000"; }
					if (.enableBlacksmith)   { .@classBits[.@classIndex] = 32;    .@classNames$[.@classIndex++] = " Blacksmith"; }
					if (.enableBsOverThrust) { .@classBits[.@classIndex] = 64;    .@classNames$[.@classIndex++] = " Blacksmith  ^727272(Over Thrust)^000000"; }
					if (.enableAlProtection) { .@classBits[.@classIndex] = 128;   .@classNames$[.@classIndex++] = " Alchemist  ^727272(Chemical Protection)^000000"; }
					if (.enableAlBerserk)    { .@classBits[.@classIndex] = 256;   .@classNames$[.@classIndex++] = " Alchemist  ^727272(Berserk Pitcher)^000000"; }
					if (.enableCrusader && BaseJob != Job_Crusader && BaseJob != Job_Paladin) {
						.@classBits[.@classIndex] = 512;
						.@classNames$[.@classIndex++] = " Crusader  ^727272(Providence)^000000";
					}
					if (.enableSniper)       { .@classBits[.@classIndex] = 1024;  .@classNames$[.@classIndex++] = " Sniper  ^727272(Wind Walk)^000000"; }
					if (.enableSoulLinker)   { .@classBits[.@classIndex] = 2048;  .@classNames$[.@classIndex++] = " Soul Linker  ^727272(Spirit)^000000"; }
					if (.enableSoulLinkerKa) { .@classBits[.@classIndex] = 4096;  .@classNames$[.@classIndex++] = " Soul Linker  ^727272(Ka)^000000"; }
					if (.enableAssassin) {
						.@assassinIndex = .@classIndex;
						.@classBits[.@classIndex] = 8192;
						.@classNames$[.@classIndex++] = " Assassin  ^727272(Enchant Poison)^000000";
					}
					if (.enableSageElemental)   {
						.@classBits[.@classIndex] = 16384;
						.@sageIndex = .@classIndex;
						.@classNames$[.@classIndex++] = " Sage  ^727272(Elemental Weapon)^000000";
					}

					.@menu$ = "";

					for (.@i = 0; .@i < getarraysize(.@classNames$); .@i++) {
						.@bit = .@classBits[.@i];
						.@isEnabled = (configBuff & .@bit);

						// Show selected element next to Sage label
						if (.@i == .@sageIndex && .@isEnabled) {
							setarray .elemLabels$[1],
								"^FF3355Fire^000000",
								"^1A66FFWater^000000",
								"^00A000Wind^000000",
								"^C1440EEarth^000000";
							.@menu$ += (.onText$) + .@classNames$[.@i] + "  " + .elemLabels$[sageElement] + ":";
						} else {
							.@menu$ += (.@isEnabled ? .onText$ : .offText$) + .@classNames$[.@i] + ":";
						}
					}

					.@menu$ += "^1A66FFEnable All Buffs^000000";
					.@menu$ += ":^FF3355Disable All Buffs^000000";
					.@menu$ += ":^727272Back to Main Menu^000000";

					.@choice = select(.@menu$);

					if (.@choice >= 1 && .@choice <= getarraysize(.@classNames$)) {
						.@index = .@choice - 1;
						.@bit = .@classBits[.@index];
						.@isEnabled = (configBuff & .@bit);

						if (.@isEnabled) {
							// If Sage, show sub-menu: change element or disable
							if (.@index == .@sageIndex) {
								.@sageChoice = select(
									"^1A66FFChange Element^000000",
									"^FF3355Disable^000000",
									"^727272Back^000000"
								);

								if (.@sageChoice == 1) {
									.@elemChoice = select(
										"^FF3355Fire  ^727272(Red Blood)^000000",
										"^1A66FFWater  ^727272(Crystal Blue)^000000",
										"^00A000Wind  ^727272(Wind of Verdure)^000000",
										"^C1440EEarth  ^727272(Green Live)^000000",
										"^727272Back^000000"
									);

									if (.@elemChoice < 5) {
										sageElement = .@elemChoice;
									}
								} else if (.@sageChoice == 2) {
									configBuff &= ~.@bit;
									sageElement = 0;
								}
							} else {
								configBuff &= ~.@bit;
							}
						} else {
							// Aspersio: Deactivate Enchant Poison and Sage elemental weapon
							if (.@classBits[.@index] == 4) {
								configBuff |= .@bit;
								configBuff &= ~8192;	// Enchant Poison
								configBuff &= ~16384;	// Sage elemental weapon
								sageElement = 0;

							// Enchant Poison: Deactivate Aspersio and Sage elemental weapon
							} else if (.@index == .@assassinIndex) {
								configBuff |= .@bit;
								configBuff &= ~4;		// Aspersio
								configBuff &= ~16384;	// Sage elemental weapon
								sageElement = 0;

							// Sage elemental weapon: Deactivate Aspersio and Enchant Poison, prompt element first
							} else if (.@index == .@sageIndex) {
								.@elemChoice = select(
									"^FF3355Fire  ^727272(Red Blood)^000000",
									"^1A66FFWater  ^727272(Crystal Blue)^000000",
									"^00A000Wind  ^727272(Wind of Verdure)^000000",
									"^C1440EEarth  ^727272(Green Live)^000000",
									"^727272Back^000000"
								);

								if (.@elemChoice < 5) {
									sageElement = .@elemChoice;
									configBuff |= .@bit;
									configBuff &= ~4;		// Aspersio
									configBuff &= ~8192;	// Enchant Poison
								}

							} else {
								configBuff |= .@bit;
							}
						}

						goto ClassBuffMenu;
					}

					// Enable all
					if (.@choice == getarraysize(.@classNames$) + 1) {
						for (.@i = 0; .@i < getarraysize(.@classNames$); .@i++) {
							if (.@i != .@sageIndex && .@i != .@assassinIndex) {
								configBuff |= .@classBits[.@i];
							}
						}
						goto ClassBuffMenu;
					}

					// Disable all
					if (.@choice == getarraysize(.@classNames$) + 2) {
						configBuff = 0;
						sageElement = 0;
						goto ClassBuffMenu;
					}

					// Back to Main Menu
					if (.@choice == getarraysize(.@classNames$) + 3) {
						goto HealerMenu;
					}

					end;
			case 3:
				end;
		}

OnStart:
	specialeffect2 EF_HEAL;
	percentheal .healHp, .healSp;
	end;
}
