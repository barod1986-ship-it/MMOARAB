		end;

	function FeeLabel {
		.@fee = (getarg(1, false)) ? getarg(0) : getarg(0) * @feeMod;
		if (.@fee == 0)
			return "";
		return " ^727272[" + callfunc("F_InsertComma", .@fee) + " z]^000000";
	}

	function Go {
		if (getarg(0) == strcharinfo(3)) {
			message strcharinfo(0), "I'm already in here.";
			end;
		}

		if (Zeny < .warpFee * @feeMod  && getarg(0) != "prontera" && getarg(0) != strcharinfo(3)) {
			message strcharinfo(0), "I don't have enough zeny.";
			end;
		}

		lastwarp$ = getarg(0);
		lastwarpx = getarg(1,0);
		lastwarpy = getarg(2,0);

		if (getarg(0) == "prontera") {
			lastwarpz = 0;
		} else if (getarg(0) != strcharinfo(3)) {
			lastwarpz = .warpFee;
			Zeny -= .warpFee * @feeMod ;
		}

		// Heal if player warp to aegis_pvp
		if (lastwarp$ == "aegis_pvp") {
			percentheal 100, 100;
		}

		warp getarg(0), getarg(1,0), getarg(2,0);
		end;
	}

	function Disp {
		if (getargcount() < 3)
			@menu$ = getarg(0);
		else {
			@menu$ = "";
			for (.@i = getarg(1); .@i <= getarg(2); .@i++)
				@menu$ += getarg(0) + " " + .@i + ":";
		}
		return;
	}

	function Pick {
		if ((@f && .onlyFirstFld) || (@d && .onlyFirstDun)) {
			.@select = 1;
		} else {
			.@select = select(@menu$);
		}
		if (getarg(0) == "") {
			.@i = .@select;
			.@map$ = getarg(.@i);
		} else {
			.@i = .@select - getarg(1,0);
			.@map$ = getarg(0) + ((.@i < 10) ? "0" : "") + .@i;
		}
		.@x = @c[.@i * 2];
		.@y = @c[.@i * 2 + 1];
		deletearray @c[0], getarraysize(@c);
		@f = false;
		@d = false;
		Go(.@map$, .@x, .@y);
	}
