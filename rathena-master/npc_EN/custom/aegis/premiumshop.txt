aegis,63,90,6	script	Premium Shop	721,{

	cutin "kafra_07", 2;
	mes .npcName$;
	mes "Welcome to the premium shop.";
	mes "Which catalog would you like to browse today?";

	for(.@i = 1; .@i < getarraysize(.shops$); .@i++) {
		set .@menu$, .@menu$ + .@i + " - " + .shops$[.@i] + ":";
	}

	.@selection = select(.@menu$);

	clear;
	mes .npcName$;
	mes "Certainly. I am opening the ^1A66FF" + .shops$[.@selection] + "^000000 catalog for you now.";
	close2;
	cutin "", 255;
	callshop "Premium Shop#" + .@selection, 1;
	end;

OnInit:
	.npcName$ = "[Premium Shop]";
	setarray .shops$[1],
		"Tools",
		"Combat Supplements",
		"Battle Scriptures",
		"Logistics Gear",
		"Bestiary Relics",
		"Rookie Relics",
		"Lucky Draw",
		"Gods' Relics";
	end;
}
