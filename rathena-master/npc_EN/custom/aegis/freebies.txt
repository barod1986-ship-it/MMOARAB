aegis,112,73,4	script	Freebies	49,{

	.@freeCash = 1000; // Amount of cash

	if (!#freebies)  {
		mes "[Freebies]";
		mes "Welcome, this is your free coin";
		#CASHPOINTS += .@freeCash;
		#freebies = 1;
		dispbottom "You've got " + callfunc("F_InsertComma", .@freeCash) + " Cash Points. Total: " + callfunc("F_InsertComma", #CASHPOINTS) + " Cash Points.";
		close;
	} else {
		mes "[Freebies]";
		mes "You have claim the free cash already.";
		close;
	}

OnInit:
	waitingroom "Freebies Here",0;
	end;
}
