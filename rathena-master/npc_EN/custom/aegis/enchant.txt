aegis_works,28,26,4	script	Enchanter	804,{

	disable_items;

	mes .npcName$;
	mes "Do you want to enchant your equipment?";
	mes "Catalyst: ^1A66FF" + callfunc("F_InsertPlural", .catalyst[1], getitemname(.catalyst[0])) + "^000000";
	mes "Enchant Fee: ^00A000" + callfunc("F_InsertComma", .enchantFee) + " ^727272z^000000";
	next;

	switch(select("^1A66FFYes^000000:What is this?:^727272Cancel^000000")) {
		case 1:
			setarray .@indices[1],
				EQI_HEAD_TOP,
				EQI_ARMOR,
				EQI_HAND_L,
				EQI_HAND_R,
				EQI_GARMENT,
				EQI_SHOES,
				EQI_ACC_L,
				EQI_ACC_R;

			for (.@i = 1; .@i < getarraysize(.@indices); .@i++) {
				if (getequipisequiped(.@indices[.@i])) {
					.@hasEnchant = (getequiprandomoption(.@indices[.@i], 0, 0) > 0);

					if (!.enchantMode && .@hasEnchant) {
						.@menu$ += ":";
						continue;
					}

					if (!(.@indices[.@i] == EQI_HAND_L && getiteminfo(getequipid(.@indices[.@i]), ITEMINFO_TYPE) == IT_ARMOR)) {
						.@equipSlot = (getequipcardid(.@indices[.@i],0) == 255) ? 0 : getitemslots(getequipid(.@indices[.@i]));
						set .@menu$, .@menu$
							+ "^727272" + F_getpositionname(.@indices[.@i]) + " "
							+ (getequiprefinerycnt(.@indices[.@i]) ? "^FF3355+" + getequiprefinerycnt(.@indices[.@i]) + " " : "")
							+ ((.@hasEnchant) ? "^00A000" : "^1A66FF") + getequipname(.@indices[.@i])
							+ ((.@equipSlot) ? " ^727272[" + .@equipSlot + "]" : "")
							+ "^000000";
						.@equipped = 1;
					}
				}
				.@menu$ += ":";
			}

			if (!.@equipped) {
				mes .npcName$;
				mes "You're not wearing any equip that I can enchant.";
				close;
			}

			.@part = .@indices[select(.@menu$)];

			if (!getequipisequiped(.@part)) {
				mes .npcName$;
				mes "Your not wearing anything there.";
				close;
			}
			clear;

			// Equipment data
			.@equipId = getequipid(.@part);
			.@equipSlot = (getequipcardid(.@part,0) == 255) ? 0 : getitemslots(.@equipId);
			.@equipRefine = getequiprefinerycnt(.@part);
			.@hasEnchant = (getequiprandomoption(.@part, 0, 0) > 0);
			setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

			getinventorylist;
			for (.@i = 0; .@i < @inventorylist_count; .@i++) {
				if (@inventorylist_expire[.@i]) {
					if (.@equipId == @inventorylist_id[.@i]) {
						mes .npcName$;
						mes "Your equipment ^FF3355" + getitemname(.@equipId) + "^000000 has ^FF3355expired time^000000. I can't enchant this equipment.";
						close;
					}
				}
			}

			mes .npcName$;
			mes "You're going to enchant " + ((.@equipRefine) ? "^FF3355+" + .@equipRefine + " " : "") + ((.@hasEnchant) ? "^00A000" : "^1A66FF") + getequipname(.@part) + ((.@equipSlot) ? " ^727272[" + .@equipSlot + "]" : "") + "^000000.";
			mes "Catalyst: ^1A66FF" + callfunc("F_InsertPlural", .catalyst[1], getitemname(.catalyst[0])) + "^000000.";
			mes "Enchant Fee: ^00A000" + callfunc("F_InsertComma", .enchantFee) + "^727272 z^000000.";
			mes "Do you want to continue?";
			next;

			if (select("^1A66FFYes, I am^000000:^727272Never mind^000000") == 2) {
				mes .npcName$;
				mes "Very well. Return at once if you seek my services.";
				close;
			}

			if (Zeny < .enchantFee || countitem(.catalyst[0]) < .catalyst[1]) {
				mes .npcName$;
				mes "You don't meet the requirements I've requested.";
				mes "I need ^00A000" + callfunc("F_InsertComma", .enchantFee) + "^727272 z^000000 and ^1A66FF" + callfunc("F_InsertPlural", .catalyst[1], getitemname(.catalyst[0])) + "^000000.";
				close;
			}

			switch (.@part) {
				// Go to db\import\item_randomopt_group.yml for random option group database
				case EQI_HEAD_TOP:
				case EQI_ARMOR: randomoptgroup(1); break;
				case EQI_GARMENT:
				case EQI_SHOES: randomoptgroup(2); break;
				case EQI_ACC_L:
				case EQI_ACC_R: randomoptgroup(3); break;
				case EQI_HAND_R:
				case EQI_HAND_L:
					mes .npcName$;
					mes "What kind of enchant do you want?";
					mes " ";
					mes "^FF3355Note: Special is combination of physical & magic.^000000";
					next;
					switch(select("^FF3355Melee^000000:^1A66FFRange^000000:^FF69B4Magic^000000:^806C00Special^000000:^727272Cancel^000000")) {
						case 1:
							randomoptgroup(4);
							break;
						case 2:
							randomoptgroup(5);
							break;
						case 3:
							randomoptgroup(6);
							break;
						case 4:
							randomoptgroup(7);
							break;
						case 5:
							mes .npcName$;
							mes "Very well. Return at once if you seek my services.";
							close;
					}
					break;
			}

			specialeffect2 EF_REFINEOK;
			Zeny -= .enchantFee;
			delitem .catalyst[0], .catalyst[1];
			delequip(.@part);
			getitem3 .@equipId,1,1,.@equipRefine,0,.@card[0],.@card[1],.@card[2],.@card[3],.@opt_id,.@opt_value,.@opt_param;

			mes .npcName$;
			mes "The process was a success.";
			mes "Here is your equipment.";
			mes "Farewell.";
			close;
		case 2:
			mes .npcName$;
			mes "I can imbue your equipment with random stats and unique effects.";
			mes "I work with ^1A66FFHeadgears^000000, ^1A66FFArmor^000000, ^1A66FFWeapons^000000, ^1A66FFGarments^000000, ^1A66FFFootwear^000000, and ^1A66FFAccessories^000000.";
			if (!.enchantMode) {
				next;
				mes .npcName$;
				mes "However, be warned: I only work on ^FF3355clean items^000000.";
				mes "Once an item has been imbued with power, I cannot touch it again.";
			}
			close;
		case 3:
			clear;
			mes .npcName$;
			mes "Very well. Return at once if you seek my services.";
			close;
	}
	close;

OnInit:
	.npcName$ = "[Enchanter]";
	setarray .catalyst, 714, 1; // Item ID required for enchantment (Default: Emperium) and the quantity.
	.enchantMode = 1; // Set to 1 to allow multiple enchants on the same item; Set to 0 to only allow fresh items.
	.enchantFee = 1000000; // Base cost for enchanting (in Zeny).
}
