aegis,104,90,4	script	Stylist	122,{

	mes .npcName$;
	mes .feeMsg$;
	mes .selectMsg$;
	next;

	setarray .@styles[1],
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color"),
		getbattleflag("max_cloth_color");

	setarray .@look[1],
		LOOK_HAIR,
		LOOK_HAIR_COLOR,
		LOOK_CLOTHES_COLOR;

	.@menu = select("Hair Style:Hair Color:Cloth Color:^FF3355Reset Looks^000000");

	if (.@menu == 4) {
		mes .npcName$;
		mes "Are you sure you want to reset your looks?";
		next;
		switch(select("Yes:No")) {
			case 1:
				for (.@i = 1; .@i <= 3; .@i++) {
					setlook .@look[.@i], 1;
				}
				specialeffect2 EF_STEAL;
				mes .npcName$;
				mes "Done! I've reset your look.";
				close;
			case 2:
				mes .npcName$;
				mes "Okay then.";
				close;
		}
		close;
	}

	.@style = .@revert = (getlook(.@look[.@menu]) > 0) ? getlook(.@look[.@menu]) : 1;

	while (1) {
		changelook .@look[.@menu], .@style;
		message strcharinfo(0), "This is style #" + .@style + ".";
		
		.@next		= ((.@style != .@styles[.@menu]) ? .@style + 1 : 1);
		.@previous	= ((.@style != 1) ? .@style - 1 : .@styles[.@menu]);
		
		.@nMenu$	= "Next [^FF3355" + ((.@style != .@styles[.@menu]) ? .@style + 1 : 1)+"^000000]";
		.@pMenu$	= "Previous [^FF3355" + ((.@style != 1) ? .@style - 1 : .@styles[.@menu]) + "^000000]";

		switch(prompt(
			(.@order) ? .@pMenu$ : .@nMenu$,
			(.@order) ? .@nMenu$ : .@pMenu$,
			"^00A000Jump to ...^000000",
			"^FF3355Revert to original^000000 [^FF3355" + .@revert + "^000000]",
			"^1A66FFUse this style^000000")) {
			case 1:
				.@style = (.@order) ? .@previous : .@next;
				break;
			case 2:
				.@style = (.@order) ? .@next : .@previous;
				.@order = !.@order;
				break;
			case 3:
				clear;
				mes .npcName$;
				mes "Choose a style between ^FF33551^000000 - ^FF3355" + .@styles[.@menu] + "^000000.";
				mes "^FF33550^000000 = Random";
				next;

				input .@style, 0, .@styles[.@menu];
				if (!.@style) .@style = rand(1, .@styles[.@menu]);

				clear;
				mes .npcName$;
				mes .feeMsg$;
				mes .selectMsg$;
				break;
			case 4:
				.@style = .@revert;
				setlook .@look[.@menu], .@revert;
				break;
			case 5:
				clear;
				mes .npcName$;
				if (Zeny < .stylistFee) {
					mes "I'm sorry, you don't have enough Zeny.";
					changelook .@look[.@menu], .@revert;
					close;
				}

				if (.@style != .@revert) {
					Zeny -= .stylistFee;
					setlook .@look[.@menu], .@style;
					specialeffect2 EF_STEAL;
					mes "Good choice.";
				} else {
					mes "No changes made?";
					mes "Come back and see me if you change your mind.";
				}

				close;
			case 255:
				.@style = .@revert;
				changelook .@look[.@menu], .@revert;
				end;
		}
	}

OnInit:
	.stylistFee = 100000;

	.npcName$ = "[Stylist]";
	.feeMsg$ = "You can browse the selection for free, but you must select ^1A66FFUse this style^000000 to finalize your choice for ^FF3355" + callfunc("F_InsertComma", .stylistFee) + "^000000 ^727272z^000000.";
	.selectMsg$ = "Do you want to change your style?";
	end;
}
