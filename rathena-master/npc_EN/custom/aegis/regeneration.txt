//= Regeneration NPC script (Pre-Renewal).
//= Faster HP & SP generation when not overburden.
//= Default: 5s per-tick.
//===== Additional Comments: =====================================
//= This script is made for Ragnarok Offline Pre-Renewal Pack
//= https://www.youtube.com/watch?v=aN5OCJdj2qA
//============================================================

function	script	F_FasterRegen	{
	.@hp = 2;		// HP heal percentage per pulse (1–100)
	.@sp = 2;		// SP heal percentage per pulse (1–100)
	.@range = 200;	// Range effect coordinate

	getmapxy .@map$, .@x, .@y, BL_NPC, strnpcinfo(3);
	
	.@size = getareaunits(BL_PC, .@map$, (.@x - (.@range + 3)), .@y - .@range, (.@x + (.@range + 3)), .@y + .@range, .@aid);
	
	if (.@size > 0) {
		for (.@i = 0; .@i < .@size; .@i++) {
			if (attachrid(.@aid[.@i])) {
				if (Hp && ((Hp != MaxHp) || (Sp != MaxSp)) && !getstatus(SC_WEIGHT50) && !getstatus(SC_WEIGHT90)) {
					percentheal .@hp, .@sp;
				}
				detachrid;
			}
		}
	}
	return;
}



//============================================================
// Duplicates
//============================================================
aegis,84,84,6	script	FasterRegen::custom_regeneration_aegis	-1,{
OnTimer5000:
	callfunc "F_FasterRegen", "aegis";
	initnpctimer;
	end;

OnInit:
	initnpctimer;
	end;
}
