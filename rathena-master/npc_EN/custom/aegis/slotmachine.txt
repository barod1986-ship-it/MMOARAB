-	script	SlotMachine	-1,{
	mes "[Slot Machine]";
	mes "Welcome!";
	mes "Would you like to try your luck?";
	mes "Cost: ^1A66FF" + callfunc("F_InsertComma", .price) + "^000000 Zeny per pull.";
	mes "Reward: ^1A66FF" + callfunc("F_InsertComma", .reward) + "^000000 Zeny.";

	if (select("^1A66FFLet's play!^000000:^727272No thanks.^000000") == 2) {
		mes "[Slot Machine]";
		mes "Come back when you're feeling lucky!";
		close;
	}

	if (Zeny < .price) {
		mes "[Slot Machine]";
		mes "You don't have enough Zeny to play.";
		close;
	}

	Zeny -= .price;

	.@slot1 = rand(10);
	.@slot2 = rand(10);
	// --- Near-Miss ---
	// A "near-miss" (e.g., 7-7-8) tricks the player's brain into thinking they "almost won."
	// This is more stimulating than a "total loss" (e.g., 7-7-2).
	// It only trigger this if the first two reels already match.
	if (.@slot1 == .@slot2) {
		// 40% chance to artificially "force" a near-miss result
		if (rand(100) < 40) {
			// Take the winning number and add or subtract 1
			// Example: If slot 1 and 2 are [7][7], this makes slot 3 either 6 or 8
			.@slot3 = .@slot1 + (rand(2) ? 1 : -1);
			// Safety Check: Ensure the number stays within the 0-9 slot range
			if (.@slot3 > 9) @slot3 = 8;
			if (.@slot3 < 0) @slot3 = 1;
		} else {
			// 60% chance to just let the third reel roll naturally
			.@slot3 = rand(10);
		}
	} else {
		// If the first two didn't match, the third reel rolls normally
		.@slot3 = rand(10);
	}

	for (.@stage = 0; .@stage < 3; .@stage++) {
		for (.@i = 0; .@i < 10; .@i++) {
			clear;
			mes " ";
			mes " ";
			mes " ";
			.@r1$ = (.@stage >= 1) ? "^1A66FF"+.@slot1+"^000000" : ""+rand(10);
			.@r2$ = (.@stage >= 2) ? "^1A66FF"+.@slot2+"^000000" : ""+rand(10);
			.@r3$ = ""+rand(10);
			mes "                              [ " + .@r1$ + " ] [ " + .@r2$ + " ] [ " + .@r3$ + " ]";
			sleep2 100;
		}
		specialeffect2 EF_STEAL;
	}

	if (.@slot1 == .@slot2 && .@slot2 == .@slot3) {
		specialeffect2 EF_RG_COIN;
		clear;
		mes " ";
		mes " ";
		mes " ";
		mes "                              [ ^00A000" + .@slot1 + "^000000 ] [ ^00A000" + .@slot2 + "^000000 ] [ ^00A000" + .@slot3 + "^000000 ]";
		mes " ";
		mes "                  JACKPOT! All three match!";
		Zeny += .reward;
		announce strcharinfo(0) + " just hit the Jackpot on the Slot Machine!", bc_all;
	} else {
		clear;
		mes " ";
		mes " ";
		mes " ";
		mes "                              [ ^FF3355" + .@slot1 + "^000000 ] [ ^FF3355" + .@slot2 + "^000000 ] [ ^FF3355" + .@slot3 + "^000000 ]";
		mes " ";
		mes "                      Better luck next time!";
	}
	close;

OnInit:
	.price = 10000; // Cost to play (Zeny)
	.reward = .price * 100; // Reward (Zeny)
	end;
}

aegis,68,110,4	duplicate(SlotMachine)	Slot Machine#1	563
