function Chk;
function GetTierCount;
function GetTierArraySize;
function GetTierMonster;
function GetMonsterName;

	mes "[Mission Board]";
	mes "Are you ready for your next mission?";
	mes "Browse the board below.";
	next;
	switch(select(
		((!Mission0) ? "^1A66FFNew Mission^000000::" : ":Mission Status:^FF0000Abandon Mission^000000") +
		":Information:^727272Cancel^000000"
	)) {
	case 1:
		mes "[Mission Board]";
		if (#Mission_Count) {
			mes "You've started a mission on another character.";
			if (!@hm_char_del_check) {
				query_sql("SELECT 1 FROM `char_reg_num` WHERE `key` = 'Mission0' AND `char_id` IN(SELECT `char_id` FROM `char` WHERE `account_id` = " + getcharid(3) + ")", .@i);
				if (!.@i) {
					next;
					mes "[Mission Board]";
					mes "I can't seem to find any records for that character, though...";
					#Mission_Count = 0;
				}
				@hm_char_del_check = true;
			}
			close;
		}

		if (#Mission_Delay > gettimetick(2)) {
			mes "I'm afraid you'll have to wait";
			mes "^FF0000" + callfunc("Time2Str", #Mission_Delay) + "^000000";
			mes "before taking another mission.";
			close;
		}

		// Build tier selection menu based on player base level
		.@available_tiers = GetTierCount(BaseLevel);
		.@menu$ = "";
		for (.@i = 1; .@i <= .@available_tiers; .@i++) {
			.@min_lv = (.@i - 1) * 10 + 1;
			.@max_lv =  (.@i == 10) ? 99 : .@i * 10;
			.@menu$ = .@menu$ + "Mission Level " + .@i + " ^727272(Lv. " + .@min_lv + " ~ " + .@max_lv + ")^000000:";
		}
		.@menu$ = .@menu$ + ":^727272Cancel^000000";
		mes "Choose your mission:";
		if (.@available_tiers < 10) {
			mes " ";
			mes "^727272Reach level " + (.@available_tiers * 10 + 1) + " to unlock Mission Level " + (.@available_tiers + 1) + ".^000000";
		}
		next;

		.@tier_choice = select(.@menu$);
		if (.@tier_choice > .@available_tiers) {
			mes "[Mission Board]";
			mes "Maybe next time!";
			close;
		}

		// Calculate info for confirmation screen
		.@tier_size = GetTierArraySize(.@tier_choice);
		if (.@tier_size == 0) {
			mes "[Mission Board]";
			mes "No monsters are available for this tier.";
			close;
		}

		.@min_lv = (.@tier_choice - 1) * 10 + 1;
		.@max_lv =  (.@tier_choice == 10) ? 99 : .@tier_choice * 10;
		.@actual_quests = (.Quests < .@tier_size) ? .Quests : .@tier_size;
		.@abandon_cost  = .Reset * .@tier_choice;

		// Confirmation screen
		mes "[Mission Board]";
		mes "^1A66FFMission Level " + .@tier_choice + "^000000 ^727272(Lv. " + .@min_lv + " ~ " + .@max_lv + ")^000000";
		mes " ^727272>^000000 Hunt " + callfunc("F_InsertPlural", .@actual_quests, "type", 0, "^1A66FF%d^000000 %s") + " of monsters.";
		mes " ^727272>^000000 Kill ^1A66FF" + .Count + "^000000 of " + ((.@actual_quests == 1) ? "the monster." : "each monster type.");
		if (.Delay)
			mes " ^727272>^000000 After completing a mission, you must wait ^1A66FF" + callfunc("F_InsertPlural", .Delay, "hour") + "^000000 before taking a new one.";
		if (.Reset < 0)
			mes " ^727272>^000000 If you abandon, you must wait ^FF0000" + callfunc("F_InsertPlural", (.Reset * -1), "hour") + "^000000 before taking another mission.";
		else if (.Reset > 0)
			mes " ^727272>^000000 Abandoning this mission costs ^FF0000" + F_InsertComma(.@abandon_cost) + "^000000 ^727272z^000000.";
		else
			mes " ^727272>^000000 Abandoning this mission has ^1A66FFno penalty^000000.";
		next;
		switch(select("Accept:^727272Cancel^000000")) {
		case 2:
			mes "[Mission Board]";
			mes "Maybe next time!";
			close;
		}

		// Pick random monsters from chosen tier array
		deletearray .@picked[0], getarraysize(.@picked);
		deletearray .@pool[0], getarraysize(.@pool);
		for (.@i = 0; .@i < .@tier_size; .@i++)
			.@pool[.@i] = GetTierMonster(.@tier_choice, .@i);

		for (.@i = 0; .@i < .@actual_quests; .@i++) {
			.@remaining = .@tier_size - .@i;
			if (.@remaining == 1) {
				.@picked[.@i] = .@pool[.@i];
			} else {
				.@r = rand(.@remaining);
				.@picked[.@i] = .@pool[.@r + .@i];
				.@tmp = .@pool[.@i];
				.@pool[.@i] = .@pool[.@r + .@i];
				.@pool[.@r + .@i] = .@tmp;
			}
		}

		for (.@i = 0; .@i < .@actual_quests; .@i++) {
			setd "Mission" + .@i, .@picked[.@i];
			setd "Mission" + .@i + "_", 0;
		}

		// Clear leftover quest slots
		for (.@i = .@actual_quests; .@i < .Quests; .@i++) {
			setd "Mission" + .@i, 0;
			setd "Mission" + .@i + "_", 0;
		}

		#Mission_Count	= .Count;
		#Mission_Tier	= .@tier_choice;
		#Mission_Quests	= .@actual_quests;
		mes "[Mission Board]";
		callsub Mission_Status;
		next;
		mes "[Mission Board]";
		mes "Report back when you've finished.";
		mes "Good luck!";
		close;

	case 2:
		mes "[Mission Board]";
		callsub Mission_Status;
		close;

	case 3:
		mes "[Mission Board]";

		if (.Reset < 0)
			mes "Abandoning your mission will set a ^FF0000" + callfunc("F_InsertPlural", (.Reset * -1), "hour") + "^000000 delay before you can take a new one.";
		else if (.Reset > 0)
			mes "Abandoning your mission will cost you ^FF0000" + F_InsertComma(.Reset * #Mission_Tier) + "^000000 ^727272z^000000.";
		else
			mes "Abandoning your mission has ^1A66FFno penalty^000000.";

		mes "Are you sure you want to abandon?";
		next;
		switch(select("^FF0000Abandon Mission^000000:^727272Cancel^000000")) {
		case 1:
			if (.Reset < 0) {
				#Mission_Delay = gettimetick(2) + ((.Reset * -1) * 3600);
				specialeffect2 EF_DECAGILITY;
			} else if (.Reset > 0) {
				if (Zeny < (.Reset * #Mission_Tier)) {
					mes "[Mission Board]";
					mes "You don't have enough Zeny to drop this mission.";
					close;
				}

				Zeny -= (.Reset * #Mission_Tier);
				specialeffect2 EF_COIN;
			} else {
				specialeffect2 EF_STORMKICK4;
			}

			for (.@i = 0; .@i < #Mission_Quests; .@i++) {
				setd "Mission" + .@i, 0;
				setd "Mission" + .@i + "_", 0;
			}

			#Mission_Count	= 0;
			#Mission_Tier	= 0;
			#Mission_Quests	= 0;

			mes "[Mission Board]";
			mes "Alright, I've dropped your current mission.";
			close;
		case 2:
			mes "[Mission Board]";
			mes "I knew you were kidding!";
			mes "Keep up the good work.";
			close;
		}

	case 4:
		mes "[Mission Board]";
		mes "Welcome to ^1A66FFMission Board^000000!";
		mes "I'll assign you a random hunt based on the ^1A66FFtier you choose^000000.";
		mes "^FF0000Higher tiers^000000 mean tougher monsters but ^00AA00better rewards^000000!";
		next;
		mes "[Mission Board]";
		mes "Each tier has its own ^1A66FFset of monsters^000000 to hunt.";
		mes "New tiers unlock every ^1A66FF10 base levels^000000.";
		next;
		mes "[Mission Board]";
		mes "Mission progress" + (.Delay ? " and delay time are" : " is") + " ^1A66FFshared across all your characters^000000.";
		mes "^FF0000Only one mission^000000 can be active at a time.";
		next;
		mes "[Mission Board]";
		mes (.Delay ? "You may take a new mission every ^1A66FF" + callfunc("F_InsertPlural", .Delay, "hour") + "^000000." : "You may take a new mission ^1A66FFwhenever you want^000000.");
		mes "Would you like to view the monster list for a specific tier?";
		next;
		switch(select("View Monster List:^727272No Thanks^000000")) {
			case 2:
				mes "[Mission Board]";
				mes "Good luck out there!";
				close;
		}
		callsub Monster_List;
		mes "[Mission Board]";
		mes "Good luck out there!";
		close;

		mes "[Mission Board]";
		mes "Nothing? Okay...";
		close;

// -------------------------------------------------------
// Mission_Status subroutine
// -------------------------------------------------------
Mission_Status:
	@mission_incomplete = false;
	.@active_quests = #Mission_Quests;
	.@tier = #Mission_Tier;

	// Show which tier the player is on
	if (.@tier > 0) {
		.@min_lv = (.@tier - 1) * 10 + 1;
		.@max_lv = (.@tier == 10) ? 99 : .@tier * 10;
		mes "^1A66FFMission Level " + .@tier + " (Lv. " + .@min_lv + " ~ " + .@max_lv + ")^000000";
	}

	deletearray .@j[0], getarraysize(.@j);
	for (.@i = 0; .@i < .@active_quests; .@i++) {
		.@j[.@i] = getd("Mission" + .@i);
		.@total_mob_lv   += getmonsterinfo(.@j[.@i], MOB_LV);
		.@total_base_exp += getmonsterinfo(.@j[.@i], MOB_BASEEXP);
		.@total_job_exp  += getmonsterinfo(.@j[.@i], MOB_JOBEXP);
		mes " ^727272>^000000 " + Chk(getd("Mission" + .@i + "_"), #Mission_Count) + "[" + getd("Mission" + .@i + "_") + "/" + #Mission_Count + "] " + GetMonsterName(.@j[.@i]) + " ^727272(" + getmonsterinfo(.@j[.@i], MOB_ID) + ")^000000";
	}

	// Reward formulas
	.@Base_Exp		= #Mission_Count * .@total_base_exp * .Reward[1] * .@tier;
	.@Job_Exp		= #Mission_Count * .@total_job_exp  * .Reward[2] * .@tier;
	.@Zeny			= #Mission_Count * .@total_mob_lv * .Reward[3] * .@tier * .@active_quests * 5;
	.@CashPoints	= .cp_reward[.@tier] * .Reward[4];

	next;
	mes "[Mission Board]";
	mes "Mission rewards:";
	mes " ^727272>^000000 Base Experience: ^1A66FF" + F_InsertComma(.@Base_Exp) + "^000000";
	mes " ^727272>^000000 Job Experience: ^1A66FF" + F_InsertComma(.@Job_Exp) + "^000000";
	mes " ^727272>^000000 Zeny: ^1A66FF" + F_InsertComma(.@Zeny) + "^000000";
	mes " ^727272>^000000 Cash Points: ^1A66FF" + F_InsertComma(.@CashPoints) + "^000000";

	if (@mission_incomplete) {
		@mission_incomplete = false;
		return;
	}
	next;
	mes "[Mission Board]";
	mes "Oh, you're done!";
	mes "Good work.";
	mes "Here's your reward.";
	specialeffect2 EF_ANGEL;
	specialeffect2 EF_TRUESIGHT;
	BaseExp		+= .@Base_Exp;
	JobExp		+= .@Job_Exp;
	Zeny		+= .@Zeny;
	#CASHPOINTS += .@CashPoints;
	for (.@i = 0; .@i < #Mission_Quests; .@i++) {
		setd "Mission" + .@i, 0;
		setd "Mission" + .@i + "_", 0;
	}
	#Mission_Count		= 0;
	#Mission_Tier		= 0;
	#Mission_Quests		= 0;
	if (.Delay)
		#Mission_Delay = gettimetick(2) + (.Delay * 3600);
	close;


// -------------------------------------------------------
// Monster_List subroutine
// -------------------------------------------------------
Monster_List:
	while (true) {
		mes "[Mission Board]";
		mes "Select a tier to view its monster list:";
		next;
		.@menu$ = "";
		for (.@i = 1; .@i <= 10; .@i++) {
			.@min_lv = (.@i - 1) * 10 + 1;
			.@max_lv = (.@i == 10) ? 99 : .@i * 10;
			.@menu$ = .@menu$ + "Mission Level " + .@i + " ^727272(Lv. " + .@min_lv + " ~ " + .@max_lv + ")^000000:";
		}
		.@menu$ = .@menu$ + ":^727272Close^000000";
		.@view_tier = select(.@menu$);
		if (.@view_tier > 10) return;
		.@view_size = GetTierArraySize(.@view_tier);
		.@min_lv = (.@view_tier - 1) * 10 + 1;
		.@max_lv = (.@view_tier == 10) ? 99 : .@view_tier * 10;
		mes "[Mission Board]";
		mes "^1A66FFMission Level " + .@view_tier + " (Lv. " + .@min_lv + " ~ " + .@max_lv + ")^000000";
		mes "Monster list:";
		for (.@i = 0; .@i < .@view_size; .@i++) {
			.@mob_id = GetTierMonster(.@view_tier, .@i);
			mes " ^727272[" + .@mob_id + "]^000000 " + GetMonsterName(.@mob_id);
		}
		next;
	}
	return;


// -------------------------------------------------------
// Helper: returns how many tiers are available for a given level
// -------------------------------------------------------
function GetTierCount {
	.@lv = getarg(0);
	.@tiers = (.@lv - 1) / 10 + 1;
	if (.@tiers > 10) .@tiers = 10;
	return .@tiers;
}


// -------------------------------------------------------
// Helper: returns size of the monster array for a given tier
// -------------------------------------------------------
function GetTierArraySize {
	.@tier = getarg(0);
	switch (.@tier) {
		case 1:  return getarraysize(.lv1);
		case 2:  return getarraysize(.lv2);
		case 3:  return getarraysize(.lv3);
		case 4:  return getarraysize(.lv4);
		case 5:  return getarraysize(.lv5);
		case 6:  return getarraysize(.lv6);
		case 7:  return getarraysize(.lv7);
		case 8:  return getarraysize(.lv8);
		case 9:  return getarraysize(.lv9);
		case 10: return getarraysize(.lv10);
		default: return 0;
	}
}


// -------------------------------------------------------
// Helper: returns a monster ID from a tier array by index
// -------------------------------------------------------
function GetTierMonster {
	.@tier	= getarg(0);
	.@idx	= getarg(1);
	switch (.@tier) {
		case 1:  return .lv1[.@idx];
		case 2:  return .lv2[.@idx];
		case 3:  return .lv3[.@idx];
		case 4:  return .lv4[.@idx];
		case 5:  return .lv5[.@idx];
		case 6:  return .lv6[.@idx];
		case 7:  return .lv7[.@idx];
		case 8:  return .lv8[.@idx];
		case 9:  return .lv9[.@idx];
		case 10: return .lv10[.@idx];
		default: return 0;
	}
}

// -------------------------------------------------------
// Chk function: color-codes kill progress
// -------------------------------------------------------
function Chk {
	if (getarg(0) < getarg(1)) {
		@mission_incomplete = true;
		return "^FF0000";
	} else
		return "^00A000";
}


// -------------------------------------------------------
// Kill tracking
// -------------------------------------------------------
OnNPCKillEvent:
	if (!getcharid(1) || !.Party) {
		if (!#Mission_Count || !Mission0) end;
		for (.@i = 0; .@i < #Mission_Quests; .@i++) {
			if (killedrid == getd("Mission" + .@i)) {
				if (getd("Mission" + .@i + "_") < #Mission_Count) {
					dispbottom "[Hunting Mission] Killed "
						+ (set(getd("Mission" + .@i + "_"), getd("Mission" + .@i + "_") + 1)) + " of " + #Mission_Count
						+ " " + GetMonsterName(killedrid) + ".", 0xFFFF63;
					end;
				}
			}
		}
	} else if (.Party) {
		.@mob = killedrid;
		getmapxy(.@map1$, .@x1, .@y1);
		getpartymember getcharid(1), 1;
		getpartymember getcharid(1), 2;
		for (.@i = 0; .@i < $@partymembercount; .@i++) {
			if (isloggedin($@partymemberaid[.@i], $@partymembercid[.@i])) {
				set .@Mission_Count, getvar(#Mission_Count, $@partymembercid[.@i]);
				set .@Mission0, getvar(Mission0, $@partymembercid[.@i]);
				set .@Mission_Quests, getvar(#Mission_Quests, $@partymembercid[.@i]);
				set .@HP, readparam(HP, $@partymembercid[.@i]);

				if (.@Mission_Count && .@Mission0 && .@HP > 0) {
					getmapxy(.@map2$, .@x2, .@y2, BL_PC, rid2name($@partymemberaid[.@i]));
					if ((.@map1$ == .@map2$ || .Party == 1) && (distance(.@x1,.@y1,.@x2,.@y2) <= 30 || .Party < 3)) {
						for (.@j = 0; .@j < .@Mission_Quests; .@j++) {
							.@my_mob_id = getvar(getd("Mission" + .@j), $@partymembercid[.@i]);
							.@my_count = getvar(getd("Mission" + .@j + "_"), $@partymembercid[.@i]);
							if (.@mob == .@my_mob_id) {
								if (.@my_count < .@Mission_Count) {
									setd "Mission" + .@j + "_", (.@my_count + 1), $@partymembercid[.@i];
									dispbottom "[Hunting Mission] Killed " + (.@my_count + 1) + " of " + .@Mission_Count + " " + GetMonsterName(killedrid) + ".", 0xFFFF63, $@partymembercid[.@i];
									break;
								}
							}
						}
					}
				}
			}
		}
	}
	end;
