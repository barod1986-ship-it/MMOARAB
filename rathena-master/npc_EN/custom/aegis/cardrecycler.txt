aegis,112,74,4	script	Card Recycler	89,{

	disable_items;

	.cardNeeded = 100; // Amount of cards required to recycle
	.zenyCost = 10000000; // Zeny cost to recycle

	mes "[Card Recycler]";
	mes "I can recycle your unused cards.";
	mes "Give me ^FF0000" + .cardNeeded + "^000000 cards, and ^FF0000" + callfunc("F_InsertComma", .zenyCost) + "^000000 ^727272 Zeny^000000";
	mes "and I'll give you ^1A66FF1^000000 random card in return.";
	next;

	callfunc("CountCards");

	if (@cardCount < .cardNeeded) {
		mes "[Card Recycler]";
		mes "You only have ^FF0000" + @cardCount + "^000000 eligible cards.";
		mes "Come back when you have at least ^1A66FF" + .cardNeeded + "^000000 cards.";
		close;
	}

	if (Zeny < .zenyCost) {
		mes "[Card Recycler]";
		mes "You don't have enough Zeny.";
		mes "You need at least ^FF0000" + callfunc("F_InsertComma", .zenyCost) + "^000000 ^727272 Zeny^000000.";
		close;
	}

	mes "[Card Recycler]";
	mes "You have ^1A66FF" + @cardCount + "^000000 eligible cards.";
	mes "Proceed with recycling ^FF0000" + .cardNeeded + "^000000 of them for ^FF0000" + callfunc("F_InsertComma", .zenyCost) + "^000000 ^727272 Zeny^000000?";
	next;

	if (select("Yes:No") == 2) close;

	// Deduct cards and Zeny
	callfunc("RemoveCards", .cardNeeded);
	Zeny -= .zenyCost;

	// Give 1 random card (excluding 4446: It doesn't exist in Pre-Renewal)
	do {
		.@newcard = rand(4001, 4453);
	} while (.@newcard == 4446);

	getitem .@newcard, 1;

	mes "[Card Recycler]";
	mes "Here is your new card!";
	mes "Card ID : ^00A000" + .@newcard + "^000000";
	mes "Card Name : ^00A000" + getitemname(.@newcard) + "^000000";
	close;
}

// Function: CountCards
function	script	CountCards	{
	@cardCount = 0;
	for (.@i = 4001; .@i <= 4453; .@i++) {
		if (.@i == 4446) continue;
		@cardCount += countitem(.@i);
	}
	return;
}

// Function: RemoveCards
function	script	RemoveCards	{
	.@toRemove = getarg(0);
	for (.@i = 4001; .@i <= 4453; .@i++) {
		if (.@i == 4446) continue;
		.@have = countitem(.@i);
		if (.@have == 0) continue;

		if (.@have >= .@toRemove) {
			delitem .@i, .@toRemove;
			break;
		} else {
			delitem .@i, .@have;
			.@toRemove -= .@have;
		}

		if (.@toRemove <= 0) break;
	}
	return;
}
