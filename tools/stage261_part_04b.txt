    for rel in all_paths:
        b = before.get(rel, "")
        a = after.get(rel, "")
        if b and a:
            status = "دون تغيير" if b == a else "معدل"
            if status == "معدل":
                changed_existing.append(rel)
        elif a:
            status = "جديد"
            new_files.append(rel)
        else:
            status = "محذوف"
            deleted_files.append(rel)
        status_counts[status] += 1
        comparison_rows.append([rel, status, b, a])

    expected_changed = {MAIN_REL, TRACK_MD_REL, TRACK_CSV_REL, COMPLETED_CSV_REL, REMAINING_CSV_REL}
    if set(changed_existing) != expected_changed:
        raise AssertionError(f"Unexpected modified existing files: {set(changed_existing) ^ expected_changed}")
    if deleted_files:
        raise AssertionError(f"Unexpected deletions: {deleted_files}")

    with (STAGE_DIR / "تقرير_المرحلة_261_مقارنة_الحزمة.csv").open("w", encoding="utf-8-sig", newline="") as f:
        csv.writer(f, lineterminator="\n").writerows(comparison_rows)

    result = {

        "changed_existing_files": len(changed_existing),
        "changed_existing_paths": changed_existing,
        "new_files": len(new_files),

        "status_counts": status_counts,
        "main_sha256": sha256(final_bytes),
    }
    write_utf8_lf(STAGE_DIR / "نتيجة_المرحلة_261.json", json.dumps(result, ensure_ascii=False, indent=2) + "\n")
