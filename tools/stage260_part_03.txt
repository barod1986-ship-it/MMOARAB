    STAGE_DIR.mkdir(parents=True, exist_ok=True)

    completed_csv_path = p(COMPLETED_CSV_REL)
    remaining_csv_path = p(REMAINING_CSV_REL)

    required = [
        main_path, en_path, navi_path, track_md_path, track_csv_path,
        completed_csv_path, remaining_csv_path,
    ]
    missing = [str(x.relative_to(ROOT)) for x in required if not x.exists()]
    if missing:
        raise FileNotFoundError(f"Missing required files: {missing}")

    original_main = main_path.read_bytes()
    english = en_path.read_bytes()
    navi_before = navi_path.read_bytes()

    if original_main.startswith(b"\xef\xbb\xbf"):
        raise AssertionError("Main file unexpectedly has UTF-8 BOM")
    if b"\r\n" in original_main:
        raise AssertionError("Main file line endings are not LF")
    original_text = original_main.decode("utf-8")
    english_text = english.decode("utf-8")

    if git_blob_sha(english) != OFFICIAL_BLOB:
        raise AssertionError(
            f"Internal English reference blob mismatch: {git_blob_sha(english)} != {OFFICIAL_BLOB}"
        )
    if len(original_text.splitlines()) != 56 or len(english_text.splitlines()) != 56:
        raise AssertionError("Unexpected line count in main/internal reference")

    backup_rel = "تقارير_المراجعة_الحوارية/المرحلة_260/نسخة_احتياطية_قبل_المرحلة_260_guildpvp.txt"
    official_rel = "تقارير_المراجعة_الحوارية/المرحلة_260/مرجع_rAthena_الرسمي_guildpvp.txt"
    p(backup_rel).write_bytes(original_main)
    p(official_rel).write_bytes(english)

    revised = original_text
    applied: list[tuple[str, str]] = []
    for old, new in REPLACEMENTS:
        count = revised.count(old)
        if count != 1:
            raise AssertionError(f"Expected one occurrence of {old!r}, found {count}")
        revised = revised.replace(old, new, 1)
        applied.append((old, new))

    revised_bytes = revised.encode("utf-8")
    if len(revised.splitlines()) != 56:
        raise AssertionError("Line count changed unexpectedly")
    main_path.write_bytes(revised_bytes)

    # Update Markdown tracker.
    track_md, track_md_bom, track_md_nl = read_text_preserve(track_md_path)
    replacements_md = [

         "- إجمالي التعديلات المسجلة: **62,435**"),
        ("- ملفات NPC المراجعة بعمق: **260 من 555**",
         "- ملفات NPC المراجعة بعمق: **261 من 555**"),
    ]
    for old, new in replacements_md:
        if old not in track_md:
            raise AssertionError(f"Tracker marker missing: {old}")
        track_md = track_md.replace(old, new, 1)
