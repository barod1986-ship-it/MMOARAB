def p(rel: str) -> Path:

def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def git_blob_sha(data: bytes) -> str:
    return hashlib.sha1(f"blob {len(data)}\0".encode("ascii") + data).hexdigest()


def read_text_preserve(path: Path) -> tuple[str, bool, str]:
    data = path.read_bytes()
    bom = data.startswith(b"\xef\xbb\xbf")
    newline = "\r\n" if b"\r\n" in data else "\n"
    return data.decode("utf-8-sig"), bom, newline


def write_text_preserve(path: Path, text: str, bom: bool, newline: str) -> None:
    normalized = text.replace("\r\n", "\n").replace("\r", "\n")
    if newline == "\r\n":
        normalized = normalized.replace("\n", "\r\n")
    data = normalized.encode("utf-8")
    if bom:
        data = b"\xef\xbb\xbf" + data
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)


def write_utf8_lf(path: Path, text: str, bom: bool = False) -> None:
    data = text.replace("\r\n", "\n").replace("\r", "\n").encode("utf-8")
    if bom:
        data = b"\xef\xbb\xbf" + data
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)


def csv_line(values: list[object]) -> str:
    stream = io.StringIO(newline="")
    csv.writer(stream, lineterminator="").writerow(values)
    return stream.getvalue()


def snapshot_repo() -> dict[str, str]:
    result: dict[str, str] = {}
    for file in ROOT.rglob("*"):
        if not file.is_file():
            continue
        rel = file.relative_to(ROOT).as_posix()
        if rel.startswith(".git/"):
            continue
        if rel == ".github/workflows/stage261.yml" or rel == "tools/apply_stage_261.py" or rel.startswith("tools/stage261_part_"):
            continue
        result[rel] = sha256(file.read_bytes())
    return result


def add_check(checks: list[dict[str, str]], name: str, condition: bool, details: str) -> None:
    checks.append({
        "test": name,
        "status": "PASS" if condition else "FAIL",
        "details": details,
    })


def _cell(value: object, style: int = 0) -> str:
    if isinstance(value, (int, float)):
        return f'<c s="{style}"><v>{value}</v></c>'
    return f'<c s="{style}" t="inlineStr"><is><t>{escape(str(value))}</t></is></c>'


def _sheet_xml(rows: list[list[object]]) -> str:
    body = []
    for i, row in enumerate(rows, 1):
        cells = "".join(_cell(v, 1 if i == 1 else 0) for v in row)
        body.append(f'<row r="{i}">{cells}</row>')
    return (
        '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
        '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" '
        'xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">'
        '<sheetData>' + "".join(body) + '</sheetData></worksheet>'
    )
